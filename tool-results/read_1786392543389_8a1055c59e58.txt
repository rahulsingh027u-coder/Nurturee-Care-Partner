   184→   184→function loadDocsFromStorage(): VerificationDoc[] {
   185→   185→  try {
   186→   186→    // Version check — if storage format is outdated, clear and start fresh
   187→   187→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   188→   188→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   189→   189→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   190→   190→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   191→   191→      return getInitialDocs();
   192→   192→    }
   193→   193→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   194→   194→    if (stored) {
   195→   195→      const parsed = JSON.parse(stored);
   196→   196→      if (Array.isArray(parsed) && parsed.length === 4) {
   197→   197→        // Validate structure — reject data from old format
   198→   198→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   199→   199→        if (isValid) return parsed;
   200→   200→      }
   201→   201→    }
   202→   202→  } catch {
   203→   203→    // fall through
   204→   204→  }
   205→   205→  return getInitialDocs();
   206→   206→}
   207→   207→
   208→   208→function saveDocsToStorage(docs: VerificationDoc[]) {
   209→   209→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   210→   210→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   211→   211→}

   231→   231→
   232→   232→const OPTIONAL_DEADLINE_DAYS = 15;
   233→   233→const REGISTERED_AT_KEY = 'nurturee_caregiver_registered_at';
   234→   234→
   235→   235→function getDaysRemaining(): number | null {
   236→   236→  try {
   237→   237→    const registeredAt = localStorage.getItem(REGISTERED_AT_KEY);
   238→   238→    if (!registeredAt) return null;
   239→   239→    const registered = new Date(registeredAt).getTime();
   240→   240→    const deadline = registered + OPTIONAL_DEADLINE_DAYS * 24 * 60 * 60 * 1000;
   241→   241→    const remaining = Math.ceil((deadline - Date.now()) / (24 * 60 * 60 * 1000));
   242→   242→    return Math.max(0, remaining);
   243→   243→  } catch {
   244→   244→    return null;
   245→   245→  }
   246→   246→}

   250→   250→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   251→   251→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   252→   252→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   253→   253→  const [formFields, setFormFields] = useState<Record<string, string>>({});

   257→   257→  const [submitting, setSubmitting] = useState(false);
   258→   258→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   259→   259→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   260→   260→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   261→   261→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   262→   262→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   263→   263→
   264→   264→  const currentYear = new Date().getFullYear();
   265→   265→
   266→   266→  // Clean up timers on unmount
   267→   267→  useEffect(() => {
   268→   268→    return () => {
   269→   269→      Object.values(reviewTimersRef.current).forEach((timers) => {
   270→   270→        timers.forEach(clearTimeout);
   271→   271→      });
   272→   272→    };
   273→   273→  }, []);
   274→   274→
   275→   275→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   276→   276→  useEffect(() => {
   277→   277→    if (submittedType) {
   278→   278→      const doc = docs.find((d) => d.type === submittedType);
   279→   279→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   280→   280→        setSubmittedType(null);
   281→   281→        setReviewingType(null);
   282→   282→      }
   283→   283→    }
   284→   284→    if (reviewingType) {
   285→   285→      const doc = docs.find((d) => d.type === reviewingType);
   286→   286→      if (!doc || doc.status !== 'under_review') {
   287→   287→        setReviewingType(null);
   288→   288→      }
   289→   289→    }
   290→   290→  }, [docs, submittedType, reviewingType]);
   291→   291→
   292→   292→  // Persist docs to localStorage whenever they change
   293→   293→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   294→   294→    setDocs((prev) => {
   295→   295→      const updated = updater(prev);
   296→   296→      saveDocsToStorage(updated);
   297→   297→      return updated;
   298→   298→    });
   299→   299→  }, []);
   300→   300→
   301→   301→  // ─── Backend Review Simulation ───────────────────────────────────────────
   302→   302→
   303→   303→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   304→   304→    // Clear any existing timers for this type
   305→   305→    if (reviewTimersRef.current[type]) {
   306→   306→      reviewTimersRef.current[type].forEach(clearTimeout);
   307→   307→    }
   308→   308→    reviewTimersRef.current[type] = [];
   309→   309→
   310→   310→    const doc = docs.find((d) => d.type === type);
   311→   311→    if (!doc) return;
   312→   312→
   313→   313→    // Timer 1: After 3s → under_review
   314→   314→    const t1 = setTimeout(() => {
   315→   315→      setReviewingType(type);
   316→   316→      updateDocs((prev) =>
   317→   317→        prev.map((d) =>
   318→   318→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   319→   319→        )
   320→   320→      );
   321→   321→    }, 3000);
   322→   322→
   323→   323→    // Timer 2: After 5-7s total → resolve
   324→   324→    const resolveDelay = 5000 + Math.random() * 2000;
   325→   325→    const t2 = setTimeout(() => {
   326→   326→      const currentDoc = docs.find((d) => d.type === type);
   327→   327→      if (!currentDoc) return;
   328→   328→
   329→   329→      const currentAttempts = currentDoc.attempts.length;
   330→   330→      let willApprove: boolean;
   331→   331→
   332→   332→      if (type === 'aadhaar') {
   333→   333→        // Aadhaar always approve on first attempt (demo-friendly)
   334→   334→        willApprove = true;
   335→   335→      } else {
   336→   336→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   337→   337→        if (attemptNumber === 1) {
   338→   338→          willApprove = Math.random() < 0.4;
   339→   339→        } else if (attemptNumber === 2) {
   340→   340→          willApprove = Math.random() < 0.7;
   341→   341→        } else {
   342→   342→          willApprove = true;
   343→   343→        }
   344→   344→      }
   345→   345→
   346→   346→      if (willApprove) {
   347→   347→        // Approve
   348→   348→        setApprovingType(type);
   349→   349→        updateDocs((prev) =>
   350→   350→          prev.map((d) => {
   351→   351→            if (d.type !== type) return d;
   352→   352→            return {
   353→   353→              ...d,
   354→   354→              status: 'approved' as VerificationStatus,
   355→   355→              attempts: d.attempts.map((a) =>
   356→   356→                a.attemptNumber === attemptNumber
   357→   357→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   358→   358→                  : a
   359→   359→              ),
   360→   360→            };
   361→   361→          })
   362→   362→        );
   363→   363→        setReviewingType(null);
   364→   364→
   365→   365→        // If Aadhaar approved and onComplete exists, call it after delay
   366→   366→        if (type === 'aadhaar' && onComplete) {
   367→   367→          setTimeout(() => {
   368→   368→            setApprovingType(null);
   369→   369→            setSubmittedType(null);
   370→   370→            onComplete();
   371→   371→          }, 1000);
   372→   372→        } else {
   373→   373→          // For optional: show success briefly, then allow going back
   374→   374→          setTimeout(() => {
   375→   375→            setApprovingType(null);
   376→   376→            setSubmittedType(null);
   377→   377→            setActiveType(null);
   378→   378→          }, 2000);
   379→   379→        }
   380→   380→      } else {
   381→   381→        // Reject
   382→   382→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   383→   383→        updateDocs((prev) =>
   384→   384→          prev.map((d) => {
   385→   385→            if (d.type !== type) return d;
   386→   386→            const updatedAttempts = d.attempts.map((a) =>
   387→   387→              a.attemptNumber === attemptNumber
   388→   388→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   389→   389→                : a
   390→   390→            );
   391→   391→            // Check if max attempts reached
   392→   392→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   393→   393→            return {
   394→   394→              ...d,
   395→   395→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   396→   396→              attempts: updatedAttempts,
   397→   397→              currentDraft: { documents: [], formFields: {} },
   398→   398→            };
   399→   399→          })
   400→   400→        );
   401→   401→        setReviewingType(null);
   402→   402→        // Transition from submitted view to allow re-upload
   403→   403→        setTimeout(() => {
   404→   404→          setSubmittedType(null);
   405→   405→        }, 300);
   406→   406→      }
   407→   407→    }, resolveDelay);
   408→   408→
   409→   409→    reviewTimersRef.current[type] = [t1, t2];
   410→   410→  }, [docs, onComplete, updateDocs]);
   411→   411→
   412→   412→  // ─── Handlers ────────────────────────────────────────────────────────────
   413→   413→
   414→   414→  const handleSelectType = (type: VerificationType) => {
   415→   415→    const doc = docs.find((d) => d.type === type);
   416→   416→    if (!doc) return;
   417→   417→    setActiveType(type);
   418→   418→    // Restore draft for editable docs; approved/locked docs show read-only
   419→   419→    const isFrozenDoc = doc.status === 'max_attempts_reached' || doc.status === 'approved';
   420→   420→    if (!isFrozenDoc) {
   421→   421→      setFormFields(doc.currentDraft?.formFields || {});
   422→   422→      if (type === 'aadhaar') {
   423→   423→        const docs = doc.currentDraft?.documents || [];
   424→   424→        setUploadedFront(docs.filter(f => f.id.startsWith('front-')));
   425→   425→        setUploadedBack(docs.filter(f => f.id.startsWith('back-')));
   426→   426→        setUploadedFiles([]);
   427→   427→      } else {
   428→   428→        setUploadedFiles(doc.currentDraft?.documents || []);
   429→   429→        setUploadedFront([]);
   430→   430→        setUploadedBack([]);
   431→   431→      }
   432→   432→    } else {
   433→   433→      setFormFields({});
   434→   434→      setUploadedFiles([]);
   435→   435→      setUploadedFront([]);
   436→   436→      setUploadedBack([]);
   437→   437→    }
   438→   438→    setSubmittedType(null);
   439→   439→  };
   440→   440→
   441→   441→  const handleBack = () => {
   442→   442→    // Save draft
   443→   443→    if (activeType) {
