    36→const avatarColors = ['blue', 'violet', 'emerald', 'amber', 'rose', 'cyan'] as const;
    37→
    38→type AvatarColor = (typeof avatarColors)[number];
    39→
    40→const avatarColorMap: Record<AvatarColor, { bg: string; text: string }> = {
    41→  blue: { bg: 'bg-blue-100', text: 'text-blue-700' },
    42→  violet: { bg: 'bg-violet-100', text: 'text-violet-700' },
    43→  emerald: { bg: 'bg-emerald-100', text: 'text-emerald-700' },
    44→  amber: { bg: 'bg-amber-100', text: 'text-amber-700' },
    45→  rose: { bg: 'bg-rose-100', text: 'text-rose-700' },
    46→  cyan: { bg: 'bg-cyan-100', text: 'text-cyan-700' },
    47→};
    48→
    49→function getAvatarColor(name: string): AvatarColor {
    50→  let hash = 0;
    51→  for (let i = 0; i < name.length; i++) {
    52→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    53→  }
    54→  return avatarColors[Math.abs(hash) % avatarColors.length];
    55→}
    56→
    57→function getInitials(name: string): string {
    58→  return name
    59→    .split(' ')
    60→    .map((part) => part[0])
    61→    .join('')
    62→    .toUpperCase()
    63→    .slice(0, 2);
    64→}
    65→
    66→function formatDate(dateStr: string): string {
    67→  const date = new Date(dateStr);
    68→  return date.toLocaleDateString('en-IN', {
    69→    day: 'numeric',
    70→    month: 'short',
    71→    year: 'numeric',
    72→  });
    73→}
    74→
    75→function careTypeLabel(careType: Patient['careType']): string {
    76→  switch (careType) {
    77→    case 'elder_care':
    78→      return 'Elder Care';
    79→    case 'child_care':
    80→      return 'Child Care';
    81→    default:
    82→      return careType;
    83→  }
    84→}
    85→
    86→function careTypeBadgeClasses(careType: Patient['careType']): string {
    87→  if (careType === 'elder_care') {
    88→    return 'bg-violet-100 text-violet-700 border-violet-200';
    89→  }
    90→  if (careType === 'child_care') {
    91→    return 'bg-rose-100 text-rose-700 border-rose-200';
    92→  }
    93→  return 'bg-gray-100 text-gray-600 border-gray-200';
    94→}

   159→          {patient.conditions.map((condition) => (

   161→              key={condition}

   172→        <span className="text-xs text-gray-500">
   173→          {patient.totalSessions} sessions

   175→        <span className="text-sm text-blue-600 font-medium">
   176→          View Details →
   177→        </span>
   178→      </div>
   179→    </div>
   180→  );
   181→}
