     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react';
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→  DialogDescription,
    10→  DialogFooter,
    11→} from '@/components/ui/dialog';
    12→import { Input } from '@/components/ui/input';
    13→import { Button } from '@/components/ui/button';
    14→import { Badge } from '@/components/ui/badge';
    15→import {
    16→  MapPin,
    17→  ShieldCheck,
    18→  Loader2,
    19→  AlertCircle,
    20→  CheckCircle,
    21→  Camera,
    22→  Clock,
    23→  ArrowLeft,
    24→  Copy,
    25→  IndianRupee,
    26→  Eye,
    27→  XCircle,
    28→  RotateCcw,
    29→  Phone,
    30→} from 'lucide-react';
    31→
    32→import type { Shift } from '@/lib/care-partner-data';
    33→
    34→const NURTUREE_HELPLINE = 'tel:+918045678900';
    35→
    36→// ── Types ────────────────────────────────────────────────────────────
    37→type CheckInStep =
    38→  | 'intro'
    39→  | 'selfie'
    40→  | 'selfie_processing'
    41→  | 'location_check'
    42→  | 'location_passed'
    43→  | 'location_failed'
    44→  | 'otp_enter'
    45→  | 'otp_verifying'
    46→  | 'success'
    47→  | 'failed';
    48→
    49→interface ShiftCheckInProps {
    50→  shift: Shift;
    51→  onStart: (shiftId: string) => void;
    52→  onCancel: () => void;
    53→}
    54→
    55→// ── Helpers ──────────────────────────────────────────────────────────
    56→const DEMO_OTP = '1234';
    57→const RESEND_COOLDOWN = 30;
    58→
    59→function formatTime(time24: string): string {
    60→  const [hours, minutes] = time24.split(':').map(Number);
    61→  const period = hours >= 12 ? 'PM' : 'AM';
    62→  const hours12 = hours % 12 || 12;
    63→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    64→}
    65→
    66→function getFamilyLabel(category: string): string {
    67→  return category === 'child_care'
    68→    ? "Child's Parent"
    69→    : "Elder's Family Member";
    70→}
    71→
    72→// ── Component ────────────────────────────────────────────────────────
    73→export default function ShiftCheckIn({
    74→  shift,
    75→  onStart,
    76→  onCancel,
    77→}: ShiftCheckInProps) {
    78→  const [step, setStep] = useState<CheckInStep>('intro');
    79→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
    80→  const [otp, setOtp] = useState(['', '', '', '']);
    81→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
    82→  const [error, setError] = useState('');
    83→  const [copiedOtp, setCopiedOtp] = useState(false);
    84→  const [useDemoMode, setUseDemoMode] = useState(false);
    85→
    86→  const videoRef = useRef<HTMLVideoElement | null>(null);
    87→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    88→  const streamRef = useRef<MediaStream | null>(null);
    89→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
    90→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    91→
    92→  const familyLabel = getFamilyLabel(shift.serviceCategory);
    93→
    94→  // ── Cleanup camera stream on unmount ─────────────────────────────
    95→  useEffect(() => {
    96→    return () => {
    97→      if (streamRef.current) {
    98→        streamRef.current.getTracks().forEach((t) => t.stop());
    99→      }
   100→      if (timerRef.current) clearInterval(timerRef.current);
   101→    };
   102→  }, []);
   103→
   104→  // ── Auto-start location check when step reaches location_check ──
   105→  useEffect(() => {
   106→    if (step === 'location_check') {
   107→      startLocationCheck();
   108→    }
   109→  }, [step]); // intentionally only depend on step to trigger once per entry
   110→
   111→  // ── Resend countdown ───────────────────────────────────────────────
   112→  useEffect(() => {
   113→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   114→
   115→    timerRef.current = setInterval(() => {
   116→      setResendTimer((prev) => {
   117→        if (prev <= 1) {
   118→          if (timerRef.current) clearInterval(timerRef.current);
   119→          return 0;
   120→        }
   121→        return prev - 1;
   122→      });
   123→    }, 1000);
   124→
   125→    return () => {
   126→      if (timerRef.current) clearInterval(timerRef.current);
   127→    };
   128→  }, [step, resendTimer]);
   129→
   130→  // ── Reset OTP inputs ──────────────────────────────────────────────
   131→  useEffect(() => {
   132→    if (step === 'otp_enter') {
   133→      setOtp(['', '', '', '']);
   134→      setError('');
   135→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   136→    }
   137→  }, [step]);
   138→
   139→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   140→  const startSelfieCapture = useCallback(async () => {
   141→    setStep('selfie');
   142→    setError('');
   143→
   144→    try {
   145→      const stream = await navigator.mediaDevices.getUserMedia({
   146→        video: { facingMode: 'user', width: 480, height: 640 },
   147→        audio: false,
   148→      });
   149→      streamRef.current = stream;
   150→
   151→      if (videoRef.current) {
   152→        videoRef.current.srcObject = stream;
   153→        await videoRef.current.play();
   154→      }
   155→    } catch {
   156→      // Camera not available — allow skip in demo
   157→      if (useDemoMode) {
   158→        setStep('location_check');
   159→      } else {
   160→        setError('Camera access denied. Please allow camera access.');
   161→      }
   162→    }
   163→  }, [useDemoMode]);
   164→
   165→  const captureSelfie = useCallback(() => {
   166→    if (!videoRef.current || !canvasRef.current) return;
   167→
   168→    const video = videoRef.current;
   169→    const canvas = canvasRef.current;
   170→    canvas.width = video.videoWidth;
   171→    canvas.height = video.videoHeight;
   172→
   173→    const ctx = canvas.getContext('2d');
   174→    if (!ctx) return;
   175→
   176→    // Mirror the image (selfie style)
   177→    ctx.translate(canvas.width, 0);
   178→    ctx.scale(-1, 1);
   179→    ctx.drawImage(video, 0, 0);
   180→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   181→
   182→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   183→    setSelfieDataUrl(dataUrl);
   184→
   185→    // Stop camera stream
   186→    if (streamRef.current) {
   187→      streamRef.current.getTracks().forEach((t) => t.stop());
   188→      streamRef.current = null;
   189→    }
   190→
   191→    setStep('selfie_processing');
   192→    // Simulate liveness check processing
   193→    setTimeout(() => {
   194→      setStep('location_check');
   195→    }, 1500);
   196→  }, []);
   197→
   198→  const skipSelfie = useCallback(() => {
   199→    if (streamRef.current) {
   200→      streamRef.current.getTracks().forEach((t) => t.stop());
   201→      streamRef.current = null;
   202→    }
   203→    setStep('location_check');
   204→  }, []);
   205→
   206→  const retakeSelfie = useCallback(() => {
   207→    setSelfieDataUrl(null);
   208→    startSelfieCapture();
   209→  }, [startSelfieCapture]);
   210→
   211→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   212→  const startLocationCheck = useCallback(async () => {
   213→    setStep('location_check');
   214→    setError('');
   215→
   216→    try {
   217→      if (useDemoMode) {
   218→        // Demo mode — simulate GPS fetch
   219→        await new Promise((r) => setTimeout(r, 2000));
   220→        setStep('location_passed');
   221→        return;
   222→      }
   223→
   224→      // Real high-accuracy GPS check
   225→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   226→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   227→          enableHighAccuracy: true,
   228→          timeout: 15000,
   229→          maximumAge: 0,
   230→        });
   231→      });
   232→
   233→      const accuracy = position.coords.accuracy;
   234→      if (accuracy <= 100) {
   235→        setStep('location_passed');
   236→      } else {
   237→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   238→        setStep('location_failed');
   239→      }
   240→    } catch (err) {
   241→      setError('Location check failed. Please enable location services and try again.');
   242→      setStep('location_failed');
   243→    }
   244→  }, [useDemoMode]);
   245→
   246→  // ── Step 3: OTP Verification ──────────────────────────────────────
   247→  const handleRequestOTP = useCallback(() => {
   248→    setResendTimer(RESEND_COOLDOWN);
   249→    setStep('otp_enter');
   250→  }, []);
   251→
   252→  const handleOtpChange = useCallback(
   253→    (index: number, value: string) => {
   254→      const digit = value.replace(/\D/g, '');
   255→
   256→      if (digit.length > 1) {
   257→        // Handle paste: fill as many slots as possible
   258→        const digits = digit.slice(0, 4);
   259→        const newOtp = [...otp];
   260→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   261→          newOtp[index + i] = digits[i];
   262→        }
   263→        setOtp(newOtp);
   264→        setError('');
   265→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   266→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   267→        return;
   268→      }
   269→
   270→      const newOtp = [...otp];
   271→      newOtp[index] = digit;
   272→      setOtp(newOtp);
   273→      setError('');
   274→
   275→      // Auto-advance to next input
   276→      if (digit && index < 3) {
   277→        inputRefs.current[index + 1]?.focus();
   278→      }
   279→    },
   280→    [otp],
   281→  );
   282→
   283→  const handleKeyDown = useCallback(
   284→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   285→      if (e.key === 'Backspace') {
   286→        if (!otp[index] && index > 0) {
   287→          // If current is empty, go back to previous
   288→          const newOtp = [...otp];
   289→          newOtp[index - 1] = '';
   290→          setOtp(newOtp);
   291→          inputRefs.current[index - 1]?.focus();
   292→        } else {
   293→          // Clear current
   294→          const newOtp = [...otp];
   295→          newOtp[index] = '';
   296→          setOtp(newOtp);
   297→        }
   298→      }
   299→    },
   300→    [otp],
   301→  );
   302→
   303→  const handlePaste = useCallback(
   304→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   305→      e.preventDefault();
   306→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   307→      if (!pasted) return;
   308→
   309→      const newOtp = [...otp];
   310→      for (let i = 0; i < pasted.length && i < 4; i++) {
   311→        newOtp[i] = pasted[i];
   312→      }
   313→      setOtp(newOtp);
   314→      setError('');
   315→
   316→      const focusIdx = Math.min(pasted.length, 3);
   317→      inputRefs.current[focusIdx]?.focus();
   318→    },
   319→    [otp],
   320→  );
   321→
   322→  const handleVerify = useCallback(() => {
   323→    const enteredOtp = otp.join('');
   324→    if (enteredOtp.length !== 4) {
   325→      setError('Please enter all 4 digits');
   326→      return;
   327→    }
   328→
   329→    setStep('otp_verifying');
   330→    setError('');
   331→
   332→    // Simulate verification (demo: accept 1234)
   333→    setTimeout(() => {
   334→      if (enteredOtp === DEMO_OTP) {
   335→        setStep('success');
   336→        // Call onStart after a brief delay for the success animation
   337→        setTimeout(() => {
   338→          onStart(shift.id);
   339→        }, 1500);
   340→      } else {
   341→        setError('Invalid OTP. Please try again.');
   342→        setStep('otp_enter');
   343→        setOtp(['', '', '', '']);
   344→        inputRefs.current[0]?.focus();
   345→      }
   346→    }, 1800);
   347→  }, [otp, shift.id, onStart]);
   348→
   349→  const handleResend = useCallback(() => {
   350→    if (resendTimer > 0) return;
   351→    setResendTimer(RESEND_COOLDOWN);
   352→    setOtp(['', '', '', '']);
   353→    setError('');
   354→    inputRefs.current[0]?.focus();
   355→  }, [resendTimer]);
   356→
   357→  const handleCopyOtp = useCallback(() => {
   358→    navigator.clipboard.writeText(DEMO_OTP).then(() => {
   359→      setCopiedOtp(true);
   360→      setTimeout(() => setCopiedOtp(false), 2000);
   361→    });
   362→  }, []);
   363→
   364→  const handleDialogClose = useCallback(
   365→    (open: boolean) => {
   366→      if (!open) onCancel();
   367→    },
   368→    [onCancel],
   369→  );
   370→
   371→  // ── Sub-renderers ──────────────────────────────────────────────────
   372→
   373→  // Shift summary card
   374→  const renderShiftSummary = () => (
   375→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   376→      <div className="flex items-center gap-2 text-muted-foreground">
   377→        <Badge variant="outline" className="text-xs font-medium">
   378→          {shift.serviceType}
   379→        </Badge>
   380→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   381→      </div>
   382→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   383→        <div className="flex items-center gap-2">
   384→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   385→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   386→        </div>
   387→        {shift.address && (
   388→          <div className="flex items-center gap-2">
   389→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   390→            <span className="truncate">{shift.address}</span>
   391→          </div>
   392→        )}
   393→        <div className="flex items-center gap-2">
   394→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   395→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   396→        </div>
   397→      </div>
   398→    </div>
   399→  );
   400→
   401→  // Progress stepper — order: Selfie → Location → OTP
   402→  const renderStepper = (currentStep: number) => (
   403→    <div className="flex items-center justify-center gap-1 mb-2">
   404→      {[
   405→        { n: 1, label: 'Selfie' },
   406→        { n: 2, label: 'Location' },
   407→        { n: 3, label: 'OTP' },
   408→      ].map((s, i) => (
   409→        <div key={s.n} className="flex items-center gap-1">
   410→          <div
   411→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   412→              currentStep >= s.n
   413→                ? 'bg-blue-600 text-white'
   414→                : 'bg-gray-100 text-gray-400'
   415→            }`}
   416→            title={s.label}
   417→          >
   418→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   419→          </div>
   420→          {i < 2 && (
   421→            <div
   422→              className={`w-8 h-0.5 ${
   423→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   424→              }`}
   425→            />
   426→          )}
   427→        </div>
   428→      ))}
   429→    </div>
   430→  );
   431→
   432→  // OTP input fields
   433→  const renderOtpInputs = () => (
   434→    <div className="flex justify-center gap-3">
   435→      {otp.map((digit, i) => (
   436→        <Input
   437→          key={i}
   438→          ref={(el) => { inputRefs.current[i] = el; }}
   439→          type="text"
   440→          inputMode="numeric"
   441→          maxLength={4}
   442→          value={digit}
   443→          onChange={(e) => handleOtpChange(i, e.target.value)}
   444→          onKeyDown={(e) => handleKeyDown(i, e)}
   445→          onPaste={handlePaste}
   446→          disabled={step === 'otp_verifying'}
   447→          className={
   448→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   449→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   450→          }
   451→          aria-label={`OTP digit ${i + 1}`}
   452→        />
   453→      ))}
   454→    </div>
   455→  );
   456→
   457→  // Hidden canvas for selfie capture
   458→  const renderHiddenCanvas = () => (
   459→    <canvas ref={canvasRef} className="hidden" />
   460→  );
   461→
   462→  // ── RENDER: Intro ──────────────────────────────────────────────────
   463→  if (step === 'intro') {
   464→    return (
   465→      <Dialog open onOpenChange={handleDialogClose}>
   466→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   467→          <div className="flex flex-col min-h-full">
   468→            <DialogHeader className="text-center sm:text-center">
   469→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   470→                <ShieldCheck className="size-6 text-blue-600" />
   471→              </div>
   472→              <DialogTitle className="text-lg">Shift Check-In</DialogTitle>
   473→              <DialogDescription className="text-xs">
   474→                Daily attendance is mandatory before starting any shift.
   475→              </DialogDescription>
   476→            </DialogHeader>
   477→
   478→            <div className="flex-1 space-y-3">
   479→              {renderStepper(0)}
   480→
   481→              {/* 3-step overview */}
   482→              <div className="space-y-2">
   483→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   484→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">1</div>
   485→                  <div>
   486→                    <p className="text-xs font-medium text-blue-700">Live Selfie</p>
   487→                    <p className="text-[11px] text-blue-600">Liveness detection for identity</p>
   488→                  </div>
   489→                </div>
   490→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   491→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">2</div>
   492→                  <div>
   493→                    <p className="text-xs font-medium text-blue-700">Location Verification</p>
   494→                    <p className="text-[11px] text-blue-600">High accuracy GPS check</p>
   495→                  </div>
   496→                </div>
   497→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   498→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">3</div>
   499→                  <div>
   500→                    <p className="text-xs font-medium text-blue-700">OTP Confirmation</p>
   501→                    <p className="text-[11px] text-blue-600">4-digit OTP with {familyLabel.toLowerCase()}</p>
   502→                  </div>
   503→                </div>
   504→              </div>
   505→
   506→              {renderShiftSummary()}
   507→
   508→              {/* Demo mode toggle */}
   509→              <div className="flex items-center gap-2 justify-center">
   510→                <button
   511→                  onClick={() => setUseDemoMode(!useDemoMode)}
   512→                  className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-blue-600 transition-colors"
   513→                >
   514→                  <Eye className="size-3.5" />
   515→                  {useDemoMode ? 'Demo Mode ON' : 'Enable Demo Mode'}
   516→                </button>
   517→                {useDemoMode && (
   518→                  <Badge variant="outline" className="text-[10px] bg-amber-50 border-amber-200 text-amber-700">
   519→                    DEMO
   520→                  </Badge>
   521→                )}
   522→              </div>
   523→            </div>
   524→
   525→            <DialogFooter className="mt-auto pt-4 border-t border-gray-100 flex-col gap-2 sm:flex-col sm:space-x-0">
   526→              <Button
   527→                onClick={startSelfieCapture}
   528→                className="w-full bg-blue-600 hover:bg-blue-700 text-white"
   529→              >
   530→                <Camera className="size-4 mr-2" />
   531→                Start Check-In
   532→              </Button>
   533→              <a
   534→                href={NURTUREE_HELPLINE}
   535→                className="inline-flex items-center justify-center gap-2 rounded-md border border-emerald-200 bg-emerald-50 text-emerald-700 px-4 h-9 text-sm font-medium hover:bg-emerald-100 transition-colors w-full"
   536→              >
   537→                <Phone className="size-4" />
   538→                Call Nurturee Helpline
   539→              </a>
   540→              <Button variant="ghost" onClick={onCancel} className="w-full">
   541→                Cancel
   542→              </Button>
   543→            </DialogFooter>
   544→          </div>
   545→        </DialogContent>
   546→      </Dialog>
   547→    );
   548→  }
   549→
   550→  // ── RENDER: Selfie Capture ────────────────────────────────────────
   551→  if (step === 'selfie') {
   552→    return (
   553→      <Dialog open onOpenChange={handleDialogClose}>
   554→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   555→          <div className="flex flex-col min-h-full">
   556→            <DialogHeader className="text-center sm:text-center">
   557→              {renderStepper(1)}
   558→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   559→                <Camera className="size-6 text-blue-600" />
   560→              </div>
   561→              <DialogTitle className="text-lg">Capture Selfie</DialogTitle>
   562→              <DialogDescription className="text-xs">
   563→                Take a live selfie for identity verification.
   564→              </DialogDescription>
   565→            </DialogHeader>
   566→
   567→            <div className="flex-1 flex flex-col justify-center space-y-3">
   568→              {/* Camera preview */}
   569→              <div className="relative flex justify-center">
   570→                <div className="relative w-56 h-72 rounded-2xl overflow-hidden bg-black shadow-lg">
   571→                  <video
   572→                    ref={videoRef}
   573→                    className="w-full h-full object-cover"
   574→                    style={{ transform: 'scaleX(-1)' }}
   575→                    autoPlay
   576→                    playsInline
   577→                    muted
