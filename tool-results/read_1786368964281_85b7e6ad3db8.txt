   181→
   182→function loadDocsFromStorage(): VerificationDoc[] {
   183→  try {
   184→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   185→    if (stored) {
   186→      const parsed = JSON.parse(stored);
   187→      if (Array.isArray(parsed) && parsed.length === 4) return parsed;
   188→    }
   189→  } catch {
   190→    // fall through
   191→  }
   192→  return getInitialDocs();
   193→}
   194→
   195→function saveDocsToStorage(docs: VerificationDoc[]) {
   196→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   197→}
   198→
   199→function getStatusDisplay(status: VerificationStatus) {
   200→  switch (status) {
   201→    case 'approved':
   202→      return { label: 'Verified', color: 'text-emerald-700', bg: 'bg-emerald-100', badgeBg: 'bg-emerald-50 border-emerald-200', icon: CheckCircle };
   203→    case 'submitted':
   204→      return { label: 'Submitted', color: 'text-blue-700', bg: 'bg-blue-100', badgeBg: 'bg-blue-50 border-blue-200', icon: Send };
   205→    case 'under_review':
   206→      return { label: 'Under Review', color: 'text-blue-700', bg: 'bg-blue-100', badgeBg: 'bg-blue-50 border-blue-200', icon: RefreshCw };
   207→    case 'rejected':
   208→      return { label: 'Rejected', color: 'text-red-700', bg: 'bg-red-100', badgeBg: 'bg-red-50 border-red-200', icon: XCircle };
   209→    case 'max_attempts_reached':
   210→      return { label: 'Locked', color: 'text-gray-600', bg: 'bg-gray-100', badgeBg: 'bg-gray-50 border-gray-200', icon: Ban };
   211→    default:

   218→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   219→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   220→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   221→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   222→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   223→  const [submitting, setSubmitting] = useState(false);
   224→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   225→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   226→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   227→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   228→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   229→
   230→  const currentYear = new Date().getFullYear();
   231→
   232→  // Clean up timers on unmount
   233→  useEffect(() => {
   234→    return () => {
   235→      Object.values(reviewTimersRef.current).forEach((timers) => {
   236→        timers.forEach(clearTimeout);
   237→      });
   238→    };
   239→  }, []);
   240→
   241→  // Persist docs to localStorage whenever they change
   242→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   243→    setDocs((prev) => {
   244→      const updated = updater(prev);
   245→      saveDocsToStorage(updated);
   246→      return updated;
   247→    });
   248→  }, []);
   249→
   250→  // ─── Backend Review Simulation ───────────────────────────────────────────
   251→
   252→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   253→    // Clear any existing timers for this type
   254→    if (reviewTimersRef.current[type]) {
   255→      reviewTimersRef.current[type].forEach(clearTimeout);
   256→    }
   257→    reviewTimersRef.current[type] = [];
   258→
   259→    const doc = docs.find((d) => d.type === type);
   260→    if (!doc) return;
   261→
   262→    // Timer 1: After 3s → under_review
   263→    const t1 = setTimeout(() => {
   264→      setReviewingType(type);
   265→      updateDocs((prev) =>
   266→        prev.map((d) =>
   267→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   268→        )
   269→      );
   270→    }, 3000);
   271→
   272→    // Timer 2: After 5-7s total → resolve
   273→    const resolveDelay = 5000 + Math.random() * 2000;
   274→    const t2 = setTimeout(() => {
   275→      const currentDoc = docs.find((d) => d.type === type);
   276→      if (!currentDoc) return;
   277→
   278→      const currentAttempts = currentDoc.attempts.length;
   279→      let willApprove: boolean;
   280→
   281→      if (type === 'aadhaar') {
   282→        // Aadhaar always approve on first attempt (demo-friendly)
   283→        willApprove = true;
   284→      } else {
   285→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   286→        if (attemptNumber === 1) {
   287→          willApprove = Math.random() < 0.4;
   288→        } else if (attemptNumber === 2) {
   289→          willApprove = Math.random() < 0.7;
   290→        } else {
   291→          willApprove = true;
   292→        }
   293→      }
   294→
   295→      if (willApprove) {
   296→        // Approve
   297→        setApprovingType(type);
   298→        updateDocs((prev) =>
   299→          prev.map((d) => {
   300→            if (d.type !== type) return d;
   301→            return {
   302→              ...d,
   303→              status: 'approved' as VerificationStatus,
   304→              attempts: d.attempts.map((a) =>
   305→                a.attemptNumber === attemptNumber
   306→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   307→                  : a
   308→              ),
   309→            };
   310→          })
   311→        );
   312→        setReviewingType(null);
   313→
   314→        // If Aadhaar approved and onComplete exists, call it after delay
   315→        if (type === 'aadhaar' && onComplete) {
   316→          setTimeout(() => {
   317→            setApprovingType(null);
   318→            setSubmittedType(null);
   319→            onComplete();
   320→          }, 1000);
   321→        } else {
   322→          // For optional: show success briefly, then allow going back
   323→          setTimeout(() => {
   324→            setApprovingType(null);
   325→            setSubmittedType(null);
   326→            setActiveType(null);
   327→          }, 2000);
   328→        }
   329→      } else {
   330→        // Reject
   331→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   332→        updateDocs((prev) =>
   333→          prev.map((d) => {
   334→            if (d.type !== type) return d;
   335→            const updatedAttempts = d.attempts.map((a) =>
   336→              a.attemptNumber === attemptNumber
   337→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   338→                : a
   339→            );
   340→            // Check if max attempts reached
   341→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   342→            return {
   343→              ...d,
   344→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   345→              attempts: updatedAttempts,
   346→              currentDraft: { documents: [], formFields: {} },
   347→            };
   348→          })
   349→        );
   350→        setReviewingType(null);
   351→        // Transition from submitted view to allow re-upload
   352→        setTimeout(() => {
   353→          setSubmittedType(null);
   354→        }, 300);
   355→      }
   356→    }, resolveDelay);
   357→
   358→    reviewTimersRef.current[type] = [t1, t2];
   359→  }, [docs, onComplete, updateDocs]);
   360→
   361→  // ─── Handlers ────────────────────────────────────────────────────────────
   362→
   363→  const handleSelectType = (type: VerificationType) => {
   364→    const doc = docs.find((d) => d.type === type);
   365→    if (!doc) return;
   366→    // If approved or max_attempts_reached, don't open form (but allow viewing)
   367→    setActiveType(type);
   368→    // Restore draft if exists, otherwise use empty
   369→    setFormFields(doc.currentDraft?.formFields || {});
   370→    setUploadedFiles(doc.currentDraft?.documents || []);
   371→    setSubmittedType(null);
   372→  };
   373→
   374→  const handleBack = () => {
   375→    // Save draft
   376→    if (activeType) {
   377→      updateDocs((prev) =>
   378→        prev.map((d) =>
   379→          d.type === activeType
   380→            ? { ...d, currentDraft: { documents: uploadedFiles, formFields } }
   381→            : d
   382→        )
   383→      );
   384→    }
   385→    // Clean up timers for this type if we navigate back
   386→    if (activeType && reviewTimersRef.current[activeType]) {
   387→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   388→      if (!submittedType && !reviewingType) {
   389→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   390→        delete reviewTimersRef.current[activeType];
   391→      }
   392→    }
   393→    setActiveType(null);
   394→    setFormFields({});
   395→    setUploadedFiles([]);
   396→    setSubmittedType(null);
   397→    setReviewingType(null);
   398→  };
   399→
   400→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   401→
   402→  const handleFileUploadClick = () => {
   403→    fileInputRef.current?.click();
   404→  };
   405→
   406→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   407→    const files = e.target.files;
   408→    if (!files || files.length === 0) return;
   409→
   410→    const config = activeType ? verificationConfig[activeType] : null;
   411→    const newFiles: Array<{ id: string; name: string; size: number; type: string }> = [];
   412→
   413→    for (let i = 0; i < files.length; i++) {
   414→      const file = files[i];
   415→      // Validate file type
   416→      if (config && !config.allowedTypes.includes(file.type)) {
   417→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   418→        continue;
   419→      }
   420→      // Validate file size
   421→      if (config && file.size > config.maxFileSize) {
   422→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   423→        continue;
   424→      }
   425→      newFiles.push({
   426→        id: `file-${Date.now()}-${i}`,
   427→        name: file.name,
   428→        size: file.size,
   429→        type: file.type,
   430→      });
   431→    }
   432→
   433→    if (newFiles.length > 0) {
   434→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   435→    }
   436→
   437→    // Reset input so same file can be re-selected
   438→    e.target.value = '';
   439→  };
   440→
   441→  const handleRemoveFile = (fileId: string) => {
   442→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   443→  };
   444→
   445→  const handleSubmit = async () => {
   446→    if (!activeType) return;
   447→    setSubmitting(true);
   448→
   449→    // Simulate submission delay
   450→    await new Promise((resolve) => setTimeout(resolve, 1500));
   451→
   452→    const doc = docs.find((d) => d.type === activeType);
   453→    const nextAttemptNumber = (doc?.attempts?.length || 0) + 1;
   454→
   455→    const newAttempt = {
   456→      attemptNumber: nextAttemptNumber,
   457→      submittedAt: new Date().toISOString(),
   458→      documents: [...uploadedFiles],
   459→      formFields: { ...formFields },
   460→      status: 'submitted' as const,
   461→    };
   462→
   463→    updateDocs((prev) =>
   464→      prev.map((d) => {
   465→        if (d.type !== activeType) return d;
   466→        return {
   467→          ...d,
   468→          status: 'submitted' as VerificationStatus,
   469→          attempts: [...d.attempts, newAttempt],
   470→          currentDraft: { documents: [], formFields: {} },
   471→        };
   472→      })
   473→    );
   474→
   475→    setSubmitting(false);
   476→    setSubmittedType(activeType);
   477→
   478→    // Start the backend review simulation
   479→    startReviewSimulation(activeType, nextAttemptNumber);
   480→  };
   481→
   482→  // ─── Derived State ────────────────────────────────────────────────────────
   483→
   484→  const aadhaarDoc = docs.find((d) => d.type === 'aadhaar');
   485→  const aadhaarApproved = aadhaarDoc?.status === 'approved';
   486→
   487→  // ─── View 1: Overview ────────────────────────────────────────────────────
   488→
   489→  if (!activeType) {
   490→    const allTypes = (Object.keys(verificationConfig) as VerificationType[]);
   491→    const mandatoryTypes = allTypes.filter((t) => verificationConfig[t].mandatory);
   492→    const optionalTypes = allTypes.filter((t) => !verificationConfig[t].mandatory);
   493→    const optionalPending = optionalTypes.filter((t) => {
   494→      const doc = docs.find((d) => d.type === t);
   495→      return doc?.status !== 'approved' && doc?.status !== 'max_attempts_reached';
   496→    });
   497→
   498→    return (
   499→      <div className="min-h-screen bg-gray-50 flex flex-col">
   500→        <div className="flex-1 flex flex-col items-center p-4 pt-8">
   501→          <div className="w-full max-w-2xl space-y-6">
   502→            {/* Header */}
   503→            <div className="text-center space-y-3">
   504→              <div className="inline-flex items-center justify-center size-16 rounded-2xl bg-gradient-to-br from-blue-600 to-indigo-600 text-white mb-2">
   505→                <Shield className="size-8" />
   506→              </div>
   507→              <h1 className="text-2xl font-bold text-gray-900">
   508→                {aadhaarApproved ? 'Document Verification' : 'Complete Your Verification'}
   509→              </h1>
   510→              <p className="text-sm text-gray-500 max-w-md mx-auto">
   511→                {aadhaarApproved
   512→                  ? 'Your identity is verified! Submit remaining documents to build trust with families and unlock more assignments.'
   513→                  : 'Welcome to Nurturee! To start accepting assignments, please verify your identity. Aadhaar verification is mandatory.'}
   514→              </p>
   515→            </div>
   516→
   517→            {/* Aadhaar Success Banner (shown after approval) */}
   518→            {aadhaarApproved && (
   519→              <div className="rounded-xl bg-emerald-50 border border-emerald-200 p-4 flex items-center gap-3">
   520→                <div className="size-10 rounded-full bg-emerald-100 flex items-center justify-center shrink-0">
   521→                  <CheckCircle className="size-5 text-emerald-600" />
   522→                </div>
   523→                <div>
   524→                  <p className="text-sm font-semibold text-emerald-800">Identity Verified</p>
   525→                  <p className="text-xs text-emerald-600 mt-0.5">
   526→                    Your Aadhaar has been verified. You can now access the dashboard and submit optional documents at any time.
   527→                  </p>
   528→                </div>
   529→              </div>
   530→            )}
   531→
   532→            {/* ─── Mandatory Section ─── */}
   533→            {!aadhaarApproved && (
   534→              <div className="space-y-3">
   535→                <div className="flex items-center gap-2">
   536→                  <Badge className="bg-red-50 text-red-600 border border-red-200 px-2.5 py-0.5 text-xs font-medium">
   537→                    Step 1 — Mandatory
   538→                  </Badge>
   539→                  <div className="flex-1 h-px bg-gray-200" />
