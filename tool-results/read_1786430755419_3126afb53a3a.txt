    62→
    63→function getDateGroupLabel(dateStr: string): string {
    64→  const date = parseISO(dateStr);
    65→  if (isToday(date)) return 'Today';
    66→  if (isTomorrow(date)) return 'Tomorrow';
    67→  if (isYesterday(date)) return 'Yesterday';
    68→  return format(date, 'EEE, MMM d, yyyy');
    69→}
    70→
    71→function getDateGroupSortKey(dateStr: string): number {
    72→  const date = parseISO(dateStr);
    73→  if (isToday(date)) return 0;
    74→  if (isTomorrow(date)) return 1;
    75→  if (isYesterday(date)) return -1;
    76→  return date.getTime();
    77→}

   103→        if (!isWithinInterval(shiftDate, { start: monthStart, end: monthEnd })) return false;

   105→      return true;

   113→      (s) => s.status === 'confirmed' || s.status === 'in_progress'
   114→    ).length;

   116→    return { total, completed, upcoming, cancelled };

   120→    return mockShifts.filter((shift) => {

   125→      if (endDate && shiftDate > endDate && !isSameDay(shiftDate, endDate)) return false;

   161→    { key: 'Completed', label: 'Completed', icon: CheckCircle, count: stats.completed, color: 'text-emerald-500', activeColor: 'text-emerald-600 border-emerald-300 bg-emerald-50 ring-1 ring-emerald-100' },
   162→    { key: 'Cancelled', label: 'Cancelled', icon: XCircle, count: stats.cancelled, color: 'text-red-400', activeColor: 'text-red-600 border-red-300 bg-red-50 ring-1 ring-red-100' },

   274→                <span className="text-xs text-gray-400">({shifts.length})</span>

   293→                            <Clock className="w-3.5 h-3.5 text-gray-400" />
   294→                            <span className="text-sm font-medium">
   295→                              {shift.startTime} - {shift.endTime}
   296→                            </span>
   297→                          </div>

   299→                            variant="outline"
   300→                            className={`text-[11px] px-2 py-0 ${badgeInfo.className}`}

   303→                              <span className="relative flex h-2 w-2 mr-1.5">
   304→                                <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-blue-400 opacity-75"></span>
   305→                                <span className="relative inline-flex rounded-full h-2 w-2 bg-blue-500"></span>
   306→                              </span>

   308→                            {badgeInfo.label}
   309→                          </Badge>

   312→                          <IndianRupee className="w-3.5 h-3.5" />

   318→                        </div>

   345→                      {/* Row 4: Address (if in_home) */}

   348→                          <MapPin className="w-3.5 h-3.5 text-gray-400 mt-0.5 shrink-0" />
