   184→function loadDocsFromStorage(): VerificationDoc[] {
   185→  try {
   186→    // Version check — if storage format is outdated, clear and start fresh
   187→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   188→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   189→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   190→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   191→      return getInitialDocs();
   192→    }
   193→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   194→    if (stored) {
   195→      const parsed = JSON.parse(stored);
   196→      if (Array.isArray(parsed) && parsed.length === 4) {
   197→        // Validate structure — reject data from old format
   198→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   199→        if (isValid) return parsed;
   200→      }
   201→    }
   202→  } catch {
   203→    // fall through
   204→  }
   205→  return getInitialDocs();
   206→}
   207→
   208→function saveDocsToStorage(docs: VerificationDoc[]) {
   209→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   210→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   211→}

   232→const OPTIONAL_DEADLINE_DAYS = 15;
   233→const REGISTERED_AT_KEY = 'nurturee_caregiver_registered_at';
   234→
   235→function getDaysRemaining(): number | null {
   236→  try {
   237→    const registeredAt = localStorage.getItem(REGISTERED_AT_KEY);
   238→    if (!registeredAt) return null;
   239→    const registered = new Date(registeredAt).getTime();
   240→    const deadline = registered + OPTIONAL_DEADLINE_DAYS * 24 * 60 * 60 * 1000;
   241→    const remaining = Math.ceil((deadline - Date.now()) / (24 * 60 * 60 * 1000));
   242→    return Math.max(0, remaining);
   243→  } catch {
   244→    return null;
   245→  }
   246→}

   250→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   251→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   252→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   253→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   254→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   255→  const [submitting, setSubmitting] = useState(false);
   256→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   257→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   258→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   259→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   260→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   261→
   262→  const currentYear = new Date().getFullYear();
   263→
   264→  // Clean up timers on unmount
   265→  useEffect(() => {
   266→    return () => {
   267→      Object.values(reviewTimersRef.current).forEach((timers) => {
   268→        timers.forEach(clearTimeout);
   269→      });
   270→    };
   271→  }, []);
   272→
   273→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   274→  useEffect(() => {
   275→    if (submittedType) {
   276→      const doc = docs.find((d) => d.type === submittedType);
   277→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   278→        setSubmittedType(null);
   279→        setReviewingType(null);
   280→      }
   281→    }
   282→    if (reviewingType) {
   283→      const doc = docs.find((d) => d.type === reviewingType);
   284→      if (!doc || doc.status !== 'under_review') {
   285→        setReviewingType(null);
   286→      }
   287→    }
   288→  }, [docs, submittedType, reviewingType]);
   289→
   290→  // Persist docs to localStorage whenever they change
   291→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   292→    setDocs((prev) => {
   293→      const updated = updater(prev);
   294→      saveDocsToStorage(updated);
   295→      return updated;
   296→    });
   297→  }, []);
   298→
   299→  // ─── Backend Review Simulation ───────────────────────────────────────────
   300→
   301→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   302→    // Clear any existing timers for this type
   303→    if (reviewTimersRef.current[type]) {
   304→      reviewTimersRef.current[type].forEach(clearTimeout);
   305→    }
   306→    reviewTimersRef.current[type] = [];
   307→
   308→    const doc = docs.find((d) => d.type === type);
   309→    if (!doc) return;
   310→
   311→    // Timer 1: After 3s → under_review
   312→    const t1 = setTimeout(() => {
   313→      setReviewingType(type);
   314→      updateDocs((prev) =>
   315→        prev.map((d) =>
   316→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   317→        )
   318→      );
   319→    }, 3000);
   320→
   321→    // Timer 2: After 5-7s total → resolve
   322→    const resolveDelay = 5000 + Math.random() * 2000;
   323→    const t2 = setTimeout(() => {
   324→      const currentDoc = docs.find((d) => d.type === type);
   325→      if (!currentDoc) return;
   326→
   327→      const currentAttempts = currentDoc.attempts.length;
   328→      let willApprove: boolean;
   329→
   330→      if (type === 'aadhaar') {
   331→        // Aadhaar always approve on first attempt (demo-friendly)
   332→        willApprove = true;
   333→      } else {
   334→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   335→        if (attemptNumber === 1) {
   336→          willApprove = Math.random() < 0.4;
   337→        } else if (attemptNumber === 2) {
   338→          willApprove = Math.random() < 0.7;
   339→        } else {
   340→          willApprove = true;
   341→        }
   342→      }
   343→
   344→      if (willApprove) {
   345→        // Approve
   346→        setApprovingType(type);
   347→        updateDocs((prev) =>
   348→          prev.map((d) => {
   349→            if (d.type !== type) return d;
   350→            return {
   351→              ...d,
   352→              status: 'approved' as VerificationStatus,
   353→              attempts: d.attempts.map((a) =>
   354→                a.attemptNumber === attemptNumber
   355→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   356→                  : a
   357→              ),
   358→            };
   359→          })
   360→        );
   361→        setReviewingType(null);
   362→
   363→        // If Aadhaar approved and onComplete exists, call it after delay
   364→        if (type === 'aadhaar' && onComplete) {
   365→          setTimeout(() => {
   366→            setApprovingType(null);
   367→            setSubmittedType(null);
   368→            onComplete();
   369→          }, 1000);
   370→        } else {
   371→          // For optional: show success briefly, then allow going back
   372→          setTimeout(() => {
   373→            setApprovingType(null);
   374→            setSubmittedType(null);
   375→            setActiveType(null);
   376→          }, 2000);
   377→        }
   378→      } else {
   379→        // Reject
   380→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   381→        updateDocs((prev) =>
   382→          prev.map((d) => {
   383→            if (d.type !== type) return d;
   384→            const updatedAttempts = d.attempts.map((a) =>
   385→              a.attemptNumber === attemptNumber
   386→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   387→                : a
   388→            );
   389→            // Check if max attempts reached
   390→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   391→            return {
   392→              ...d,
   393→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   394→              attempts: updatedAttempts,
   395→              currentDraft: { documents: [], formFields: {} },
   396→            };
   397→          })
   398→        );
   399→        setReviewingType(null);
   400→        // Transition from submitted view to allow re-upload
   401→        setTimeout(() => {
   402→          setSubmittedType(null);
   403→        }, 300);
   404→      }
   405→    }, resolveDelay);
   406→
   407→    reviewTimersRef.current[type] = [t1, t2];
   408→  }, [docs, onComplete, updateDocs]);
   409→
   410→  // ─── Handlers ────────────────────────────────────────────────────────────
   411→
   412→  const handleSelectType = (type: VerificationType) => {
   413→    const doc = docs.find((d) => d.type === type);
   414→    if (!doc) return;
   415→    // If approved or max_attempts_reached, don't open form (but allow viewing)
   416→    setActiveType(type);
   417→    // Restore draft if exists, otherwise use empty
   418→    setFormFields(doc.currentDraft?.formFields || {});
   419→    setUploadedFiles(doc.currentDraft?.documents || []);
   420→    setSubmittedType(null);
   421→  };
   422→
   423→  const handleBack = () => {
   424→    // Save draft
   425→    if (activeType) {
   426→      updateDocs((prev) =>
   427→        prev.map((d) =>
   428→          d.type === activeType
   429→            ? { ...d, currentDraft: { documents: uploadedFiles, formFields } }
   430→            : d
   431→        )
   432→      );
   433→    }
   434→    // Clean up timers for this type if we navigate back
   435→    if (activeType && reviewTimersRef.current[activeType]) {
   436→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   437→      if (!submittedType && !reviewingType) {
   438→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   439→        delete reviewTimersRef.current[activeType];
   440→      }
   441→    }
   442→    setActiveType(null);
   443→    setFormFields({});
   444→    setUploadedFiles([]);
   445→    setSubmittedType(null);
   446→    setReviewingType(null);
   447→  };
   448→
   449→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   450→  const fileInputId = `file-upload-${activeType || 'default'}`;
   451→
   452→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   453→    const files = e.target.files;
   454→    if (!files || files.length === 0) return;
   455→
   456→    const config = activeType ? verificationConfig[activeType] : null;
   457→    const newFiles: Array<{ id: string; name: string; size: number; type: string }> = [];
   458→
   459→    for (let i = 0; i < files.length; i++) {
   460→      const file = files[i];
   461→      // Validate file type
   462→      if (config && !config.allowedTypes.includes(file.type)) {
   463→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   464→        continue;
   465→      }
   466→      // Validate file size
   467→      if (config && file.size > config.maxFileSize) {
   468→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   469→        continue;
   470→      }
   471→      newFiles.push({
   472→        id: `file-${Date.now()}-${i}`,
   473→        name: file.name,
   474→        size: file.size,
   475→        type: file.type,
   476→      });
   477→    }
   478→
   479→    if (newFiles.length > 0) {
   480→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   481→    }
   482→
   483→    // Reset input so same file can be re-selected
   484→    e.target.value = '';
   485→  };
   486→
   487→  const handleRemoveFile = (fileId: string) => {
   488→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   489→  };
   490→
   491→  const handleSubmit = async () => {
   492→    if (!activeType) return;
   493→    setSubmitting(true);
   494→
   495→    // Simulate submission delay
   496→    await new Promise((resolve) => setTimeout(resolve, 1500));
   497→
   498→    const doc = docs.find((d) => d.type === activeType);
   499→    const nextAttemptNumber = (doc?.attempts?.length || 0) + 1;
   500→
   501→    const newAttempt = {
   502→      attemptNumber: nextAttemptNumber,
   503→      submittedAt: new Date().toISOString(),
   504→      documents: [...uploadedFiles],
   505→      formFields: { ...formFields },
   506→      status: 'submitted' as const,
   507→    };
   508→
   509→    updateDocs((prev) =>
   510→      prev.map((d) => {
   511→        if (d.type !== activeType) return d;
   512→        return {
   513→          ...d,
   514→          status: 'submitted' as VerificationStatus,
   515→          attempts: [...d.attempts, newAttempt],
   516→          currentDraft: { documents: [], formFields: {} },
   517→        };
   518→      })
   519→    );
   520→
   521→    setSubmitting(false);
   522→    setSubmittedType(activeType);
   523→
   524→    // Start the backend review simulation
   525→    startReviewSimulation(activeType, nextAttemptNumber);
   526→  };
   527→
   528→  // ─── Derived State ────────────────────────────────────────────────────────
   529→
   530→  const aadhaarDoc = docs.find((d) => d.type === 'aadhaar');
   531→  const aadhaarApproved = aadhaarDoc?.status === 'approved';
   532→
   533→  // ─── View 1: Overview ────────────────────────────────────────────────────
   534→
   535→  if (!activeType) {
   536→    const allTypes = (Object.keys(verificationConfig) as VerificationType[]);
   537→    const mandatoryTypes = allTypes.filter((t) => verificationConfig[t].mandatory);
   538→    const optionalTypes = allTypes.filter((t) => !verificationConfig[t].mandatory);
   539→    const optionalPending = optionalTypes.filter((t) => {
   540→      const doc = docs.find((d) => d.type === t);
   541→      return doc?.status !== 'approved' && doc?.status !== 'max_attempts_reached';
   542→    });
   543→
   544→    return (
