    36→// ── Skin-tone face detection (pure canvas, works on all browsers) ────────
    37→function detectSkinInOval(
    38→  video: HTMLVideoElement,
    39→  canvas: HTMLCanvasElement,
    40→): boolean {
    41→  const vw = video.videoWidth || 480;
    42→  const vh = video.videoHeight || 640;
    43→  canvas.width = vw;
    44→  canvas.height = vh;
    45→  const ctx = canvas.getContext('2d');
    46→  if (!ctx) return false;
    47→
    48→  ctx.drawImage(video, 0, 0, vw, vh);
    49→
    50→  // Sample the center oval region (face area)
    51→  const cx = vw / 2;
    52→  const cy = vh / 2;
    53→  const rx = vw * 0.2;  // oval horizontal radius
    54→  const ry = vh * 0.28; // oval vertical radius
    55→  const step = 8;        // sample every 8px for speed
    56→
    57→  let skinPixels = 0;
    58→  let totalPixels = 0;
    59→
    60→  for (let y = cy - ry; y <= cy + ry; y += step) {
    61→    for (let x = cx - rx; x <= cx + rx; x += step) {
    62→      // Check if inside ellipse: ((x-cx)/rx)^2 + ((y-cy)/ry)^2 <= 1
    63→      const dx = (x - cx) / rx;
    64→      const dy = (y - cy) / ry;
    65→      if (dx * dx + dy * dy > 1) continue;
    66→
    67→      totalPixels++;
    68→      const pixel = ctx.getImageData(Math.floor(x), Math.floor(y), 1, 1).data;
    69→      const r = pixel[0], g = pixel[1], b = pixel[2];
    70→
    71→      // Skin-tone detection in RGB space (covers wide range of skin tones)
    72→      // Rules: R > G > B, R between 60-255, G between 30-220, B between 15-180
    73→      // And logical skin-tone constraints based on chrominance
    74→      if (
    75→        r > 60 && g > 30 && b > 15 &&
    76→        r > g && r > b &&
    77→        (r - g) > 10 &&
    78→        (r - b) > 15 &&
    79→        Math.abs(r - g) < 100 &&
    80→        b < r * 0.75 &&
    81→        g < r * 0.9
    82→      ) {
    83→        skinPixels++;
    84→      }
    85→    }
    86→  }
    87→
    88→  // Need at least 30% skin-tone pixels in the oval to consider face detected
    89→  return totalPixels > 0 && (skinPixels / totalPixels) > 0.30;
    90→}

   116→function formatTime(time24: string): string {
   117→  const [hours, minutes] = time24.split(':').map(Number);
   118→  const period = hours >= 12 ? 'PM' : 'AM';
   119→  const hours12 = hours % 12 || 12;
   120→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
   121→}

   123→function getFamilyLabel(category: string): string {
   124→  return category === 'child_care'
   125→    ? "Child's Parent"
   126→    : "Elder's Family Member";
   127→}

   130→export default function ShiftCheckIn({
   131→  shift,
   132→  onStart,
   133→  onCancel,
   134→}: ShiftCheckInProps) {
   135→  const [step, setStep] = useState<CheckInStep>('intro');
   136→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
   137→  const [otp, setOtp] = useState(['', '', '', '']);
   138→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
   139→  const [error, setError] = useState('');
   140→  const [useDemoMode, setUseDemoMode] = useState(false);
   141→  const [faceDetected, setFaceDetected] = useState(false);
   142→
   143→  const videoRef = useRef<HTMLVideoElement | null>(null);
   144→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   145→  const detectionCanvasRef = useRef<HTMLCanvasElement | null>(null);
   146→  const streamRef = useRef<MediaStream | null>(null);
   147→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   148→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   149→  const faceDetectRef = useRef<ReturnType<typeof setInterval> | null>(null);
   150→  const autoCaptureRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   151→
   152→  const familyLabel = getFamilyLabel(shift.serviceCategory);
   153→
   154→  // ── Cleanup camera stream on unmount ──────────
   155→  useEffect(() => {
   156→    return () => {
   157→      if (streamRef.current) {
   158→        streamRef.current.getTracks().forEach((t) => t.stop());
   159→      }
   160→      if (timerRef.current) clearInterval(timerRef.current);
   161→      if (faceDetectRef.current) clearInterval(faceDetectRef.current);
   162→      if (autoCaptureRef.current) clearTimeout(autoCaptureRef.current);
   163→    };
   164→  }, []);
   165→
   166→  // ── Auto-start location check when step reaches location_check ──
   167→  useEffect(() => {
   168→    if (step === 'location_check') {
   169→      startLocationCheck();
   170→    }
   171→  }, [step]); // intentionally only depend on step to trigger once per entry
   172→
   173→  // ── Resend countdown ───────────────────────────────────────────────
   174→  useEffect(() => {
   175→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   176→
   177→    timerRef.current = setInterval(() => {
   178→      setResendTimer((prev) => {
   179→        if (prev <= 1) {
   180→          if (timerRef.current) clearInterval(timerRef.current);
   181→          return 0;
   182→        }
   183→        return prev - 1;
   184→      });
   185→    }, 1000);
   186→
   187→    return () => {
   188→      if (timerRef.current) clearInterval(timerRef.current);
   189→    };
   190→  }, [step, resendTimer]);
   191→
   192→  // ── Reset OTP inputs ──────────────────────────────────────────────
   193→  useEffect(() => {
   194→    if (step === 'otp_enter') {
   195→      setOtp(['', '', '', '']);
   196→      setError('');
   197→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   198→    }
   199→  }, [step]);
   200→
   201→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   202→  const startSelfieCapture = useCallback(async () => {
   203→    setStep('selfie');
   204→    setFaceDetected(false);
   205→    setError('');
   206→
   207→    try {
   208→      const stream = await navigator.mediaDevices.getUserMedia({
   209→        video: { facingMode: 'user', width: 480, height: 640 },
   210→        audio: false,
   211→      });
   212→      streamRef.current = stream;
   213→
   214→      if (videoRef.current) {
   215→        const video = videoRef.current;
   216→        video.srcObject = stream;
   217→        await video.play();
   218→
   219→        // Start face detection loop (~5 fps) once video is playing
   220→        const detCanvas = detectionCanvasRef.current || document.createElement('canvas');
   221→        detectionCanvasRef.current = detCanvas;
   222→
   223→        if (faceDetectRef.current) clearInterval(faceDetectRef.current);
   224→        faceDetectRef.current = setInterval(() => {
   225→          if (!videoRef.current || videoRef.current.readyState < 2) return;
   226→          const detected = detectSkinInOval(videoRef.current, detCanvas);
   227→          setFaceDetected(detected);
   228→
   229→          // Auto-capture after face is stable in oval for 1.5s
   230→          if (detected && !autoCaptureRef.current) {
   231→            autoCaptureRef.current = setTimeout(() => {
   232→              autoCaptureRef.current = null;
   233→              captureSelfie();
   234→            }, 1500);
   235→          } else if (!detected && autoCaptureRef.current) {
   236→            clearTimeout(autoCaptureRef.current);
   237→            autoCaptureRef.current = null;
   238→          }
   239→        }, 200);
   240→      }
   241→    } catch {
   242→      if (useDemoMode) {
   243→        setStep('location_check');
   244→      } else {
   245→        setError('Camera access denied. Please allow camera access.');
   246→      }
   247→    }
   248→  }, [useDemoMode]);
   249→
   250→  const stopFaceDetection = useCallback(() => {
   251→    if (faceDetectRef.current) {
   252→      clearInterval(faceDetectRef.current);
   253→      faceDetectRef.current = null;
   254→    }
   255→    if (autoCaptureRef.current) {
   256→      clearTimeout(autoCaptureRef.current);
   257→      autoCaptureRef.current = null;
   258→    }
   259→    setFaceDetected(false);
   260→  }, []);

   263→    if (!videoRef.current || !canvasRef.current) return;
   264→
   265→    const video = videoRef.current;
   266→    const canvas = canvasRef.current;
   267→    canvas.width = video.videoWidth;
   268→    canvas.height = video.videoHeight;
   269→
   270→    const ctx = canvas.getContext('2d');
   271→    if (!ctx) return;
   272→
   273→    // Mirror the image (selfie style)
   274→    ctx.translate(canvas.width, 0);
   275→    ctx.scale(-1, 1);
   276→    ctx.drawImage(video, 0, 0);

   282→    // Stop camera stream + face detection
   283→    stopFaceDetection();

   285→      streamRef.current.getTracks().forEach((t) => t.stop());
   286→      streamRef.current = null;
   287→    }
   288→
   289→    // Show preview for user to accept or retake
   290→    setStep('selfie_preview');
   291→  }, []);
   292→
   293→  const skipSelfie = useCallback(() => {
   294→    stopFaceDetection();
   295→    if (streamRef.current) {
   296→      streamRef.current.getTracks().forEach((t) => t.stop());
   297→      streamRef.current = null;
   298→    }
   299→    setStep('location_check');
   300→  }, [stopFaceDetection]);
   301→

   304→    startSelfieCapture();
   305→  }, [startSelfieCapture]);
   306→
   307→  // Accept selfie and proceed to liveness check (backend-ready)
   308→  const acceptSelfie = useCallback(() => {
   309→    setStep('selfie_processing');
   310→    // TODO: Send selfieDataUrl to backend for liveness/face verification
   311→    // For now, simulate processing delay
   312→    setTimeout(() => {
   313→      setStep('location_check');
   314→    }, 1500);
   315→  }, []);
   316→
   317→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   318→  const startLocationCheck = useCallback(async () => {
   319→    setStep('location_check');
   320→    setError('');
   321→
   322→    try {
   323→      if (useDemoMode) {
   324→        // Demo mode — simulate GPS fetch
   325→        await new Promise((r) => setTimeout(r, 2000));
   326→        setStep('location_passed');
   327→        return;
   328→      }
   329→
   330→      // Real high-accuracy GPS check
   331→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   332→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   333→          enableHighAccuracy: true,
   334→          timeout: 15000,
   335→          maximumAge: 0,
   336→        });
   337→      });
   338→
   339→      const accuracy = position.coords.accuracy;
   340→      if (accuracy <= 100) {
   341→        setStep('location_passed');
   342→      } else {
   343→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   344→        setStep('location_failed');
   345→      }
   346→    } catch (err) {
   347→      setError('Location check failed. Please enable location services and try again.');
   348→      setStep('location_failed');
   349→    }
   350→  }, [useDemoMode]);
   351→
   352→  // ── Step 3: OTP Verification ──────────────────────────────────────
   353→  const handleRequestOTP = useCallback(() => {
   354→    setResendTimer(RESEND_COOLDOWN);
   355→    setStep('otp_enter');
   356→  }, []);
   357→
   358→  const handleOtpChange = useCallback(
   359→    (index: number, value: string) => {
   360→      const digit = value.replace(/\D/g, '');
   361→
   362→      if (digit.length > 1) {
   363→        // Handle paste: fill as many slots as possible
   364→        const digits = digit.slice(0, 4);
   365→        const newOtp = [...otp];
   366→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   367→          newOtp[index + i] = digits[i];
   368→        }
   369→        setOtp(newOtp);
   370→        setError('');
   371→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   372→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   373→        return;
   374→      }
   375→
   376→      const newOtp = [...otp];
   377→      newOtp[index] = digit;
   378→      setOtp(newOtp);
   379→      setError('');
   380→
   381→      // Auto-advance to next input
   382→      if (digit && index < 3) {
   383→        inputRefs.current[index + 1]?.focus();
   384→      }
   385→    },
   386→    [otp],
   387→  );
   388→
   389→  const handleKeyDown = useCallback(
   390→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   391→      if (e.key === 'Backspace') {
   392→        if (!otp[index] && index > 0) {
   393→          // If current is empty, go back to previous
   394→          const newOtp = [...otp];
   395→          newOtp[index - 1] = '';
   396→          setOtp(newOtp);
   397→          inputRefs.current[index - 1]?.focus();
   398→        } else {
   399→          // Clear current
   400→          const newOtp = [...otp];
   401→          newOtp[index] = '';
   402→          setOtp(newOtp);
   403→        }
   404→      }
   405→    },
   406→    [otp],
   407→  );
   408→
   409→  const handlePaste = useCallback(
   410→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   411→      e.preventDefault();
   412→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   413→      if (!pasted) return;
   414→
   415→      const newOtp = [...otp];
   416→      for (let i = 0; i < pasted.length && i < 4; i++) {
   417→        newOtp[i] = pasted[i];
   418→      }
   419→      setOtp(newOtp);
   420→      setError('');
   421→
   422→      const focusIdx = Math.min(pasted.length, 3);
   423→      inputRefs.current[focusIdx]?.focus();
   424→    },
   425→    [otp],
   426→  );
   427→
   428→  const handleVerify = useCallback(() => {
   429→    const enteredOtp = otp.join('');
   430→    if (enteredOtp.length !== 4) {
   431→      setError('Please enter all 4 digits');
   432→      return;
   433→    }
   434→
   435→    setStep('otp_verifying');
   436→    setError('');
   437→
   438→    // Simulate verification (demo: accept 1234)
   439→    setTimeout(() => {
   440→      if (enteredOtp === DEMO_OTP) {
   441→        setStep('success');
   442→        // Call onStart after a brief delay for the success animation
   443→        setTimeout(() => {
   444→          onStart(shift.id);
   445→        }, 1500);
   446→      } else {
   447→        setError('Invalid OTP. Please try again.');
   448→        setStep('otp_enter');
   449→        setOtp(['', '', '', '']);
   450→        inputRefs.current[0]?.focus();
   451→      }
   452→    }, 1800);
   453→  }, [otp, shift.id, onStart]);
   454→
   455→  const handleResend = useCallback(() => {
   456→    if (resendTimer > 0) return;
   457→    setResendTimer(RESEND_COOLDOWN);
   458→    setOtp(['', '', '', '']);
   459→    setError('');
   460→    inputRefs.current[0]?.focus();
   461→  }, [resendTimer]);
   462→
   463→  const handleDialogClose = useCallback(
   464→    (open: boolean) => {
   465→      if (!open) onCancel();
   466→    },
   467→    [onCancel],
   468→  );
   469→
   470→  // ── Sub-renderers ──────────────────────────────────────────────────
   471→
   472→  // Shift summary card
   473→  const renderShiftSummary = () => (
   474→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   475→      <div className="flex items-center gap-2 text-muted-foreground">
   476→        <Badge variant="outline" className="text-xs font-medium">
   477→          {shift.serviceType}
   478→        </Badge>
   479→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   480→      </div>
   481→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   482→        <div className="flex items-center gap-2">
   483→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   484→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   485→        </div>
   486→        {shift.address && (
   487→          <div className="flex items-center gap-2">
   488→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   489→            <span className="truncate">{shift.address}</span>
   490→          </div>
   491→        )}
   492→        <div className="flex items-center gap-2">
   493→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   494→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   495→        </div>
   496→      </div>
   497→    </div>
   498→  );
   499→
   500→  // Progress stepper — order: Selfie → Location → OTP
   501→  const renderStepper = (currentStep: number) => (
   502→    <div className="flex items-center justify-center gap-1 mb-2">
   503→      {[
   504→        { n: 1, label: 'Selfie' },
   505→        { n: 2, label: 'Location' },
   506→        { n: 3, label: 'OTP' },
   507→      ].map((s, i) => (
   508→        <div key={s.n} className="flex items-center gap-1">
   509→          <div
   510→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   511→              currentStep >= s.n
   512→                ? 'bg-blue-600 text-white'
   513→                : 'bg-gray-100 text-gray-400'
   514→            }`}
   515→            title={s.label}
   516→          >
   517→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   518→          </div>
   519→          {i < 2 && (
   520→            <div
   521→              className={`w-8 h-0.5 ${
   522→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   523→              }`}
   524→            />
   525→          )}
   526→        </div>
   527→      ))}
   528→    </div>
   529→  );
   530→
   531→  // OTP input fields
   532→  const renderOtpInputs = () => (
   533→    <div className="flex justify-center gap-3">
   534→      {otp.map((digit, i) => (
   535→        <Input
   536→          key={i}
   537→          ref={(el) => { inputRefs.current[i] = el; }}
   538→          type="text"
   539→          inputMode="numeric"
   540→          maxLength={4}
   541→          value={digit}
   542→          onChange={(e) => handleOtpChange(i, e.target.value)}
   543→          onKeyDown={(e) => handleKeyDown(i, e)}
   544→          onPaste={handlePaste}
   545→          disabled={step === 'otp_verifying'}
   546→          className={
   547→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   548→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   549→          }
   550→          aria-label={`OTP digit ${i + 1}`}
   551→        />
   552→      ))}
   553→    </div>
   554→  );
   555→
   556→  // Hidden canvas for selfie capture
   557→  const renderHiddenCanvas = () => (
   558→    <canvas ref={canvasRef} className="hidden" />
   559→  );
   560→
   561→  // ── RENDER: Intro ──────────────────────────────────────────────────
   562→  if (step === 'intro') {

   566→          <div className="flex flex-col min-h-full">
   567→            <DialogHeader className="text-center sm:text-center">
   568→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">

   570→              </div>
   571→              <DialogTitle className="text-lg">Shift Check-In</DialogTitle>
   572→              <DialogDescription className="text-xs">

   574→              </DialogDescription>

   576→
   577→            <div className="flex-1 space-y-3">
   578→              {renderStepper(0)}
   579→
   580→              {/* 3-step overview */}
   581→              <div className="space-y-2">
   582→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   583→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">1</div>
