    68→    68→function formatTime(time24: string): string {
    69→    69→  const [hours, minutes] = time24.split(':').map(Number);
    70→    70→  const period = hours >= 12 ? 'PM' : 'AM';
    71→    71→  const hours12 = hours % 12 || 12;
    72→    72→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    73→    73→}

    75→    75→function getFamilyLabel(category: string): string {
    76→    76→  return category === 'child_care'
    77→    77→    ? "Child's Parent"
    78→    78→    : "Elder's Family Member";
    79→    79→}

    82→    82→export default function ShiftCheckIn({
    83→    83→  shift,
    84→    84→  onStart,
    85→    85→  onCancel,
    86→    86→}: ShiftCheckInProps) {
    87→    87→  const [step, setStep] = useState<CheckInStep>('intro');
    88→    88→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
    89→    89→  const [otp, setOtp] = useState(['', '', '', '']);
    90→    90→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
    91→    91→  const [error, setError] = useState('');
    92→    92→  const [useDemoMode, setUseDemoMode] = useState(false);
    93→    93→  const [modelsReady, setModelsReady] = useState(false);
    94→    94→  const [blinkDetected, setBlinkDetected] = useState(false);
    95→    95→  const [faceVisible, setFaceVisible] = useState(false);
    96→    96→  const [qualityChecks, setQualityChecks] = useState<QualityCheck[]>([]);
    97→    97→
    98→    98→  const videoRef = useRef<HTMLVideoElement | null>(null);
    99→    99→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   100→   100→  const streamRef = useRef<MediaStream | null>(null);
   101→   101→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   102→   102→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   103→   103→  const detectionLoopRef = useRef<DetectionLoopHandle | null>(null);
   104→   104→
   105→   105→  const familyLabel = getFamilyLabel(shift.serviceCategory);
   106→   106→
   107→   107→  // ── Preload face detection models during intro ─────────────────
   108→   108→  useEffect(() => {
   109→   109→    if (step === 'intro' && !modelsReady) {
   110→   110→      loadModels().then((loaded) => setModelsReady(loaded));
   111→   111→    }
   112→   112→  }, [step, modelsReady]);
   113→   113→
   114→   114→  // ── Cleanup camera stream + detection loop on unmount ──────────
   115→   115→  useEffect(() => {
   116→   116→    return () => {
   117→   117→      if (streamRef.current) {
   118→   118→        streamRef.current.getTracks().forEach((t) => t.stop());
   119→   119→      }
   120→   120→      if (timerRef.current) clearInterval(timerRef.current);
   121→   121→      detectionLoopRef.current?.stop();
   122→   122→    };
   123→   123→  }, []);
   124→   124→
   125→   125→  // ── Auto-start location check when step reaches location_check ──
   126→   126→  useEffect(() => {
   127→   127→    if (step === 'location_check') {
   128→   128→      startLocationCheck();
   129→   129→    }
   130→   130→  }, [step]); // intentionally only depend on step to trigger once per entry
   131→   131→
   132→   132→  // ── Resend countdown ───────────────────────────────────────────────
   133→   133→  useEffect(() => {
   134→   134→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   135→   135→
   136→   136→    timerRef.current = setInterval(() => {
   137→   137→      setResendTimer((prev) => {
   138→   138→        if (prev <= 1) {
   139→   139→          if (timerRef.current) clearInterval(timerRef.current);
   140→   140→          return 0;
   141→   141→        }
   142→   142→        return prev - 1;
   143→   143→      });
   144→   144→    }, 1000);
   145→   145→
   146→   146→    return () => {
   147→   147→      if (timerRef.current) clearInterval(timerRef.current);
   148→   148→    };
   149→   149→  }, [step, resendTimer]);
   150→   150→
   151→   151→  // ── Reset OTP inputs ──────────────────────────────────────────────
   152→   152→  useEffect(() => {
   153→   153→    if (step === 'otp_enter') {
   154→   154→      setOtp(['', '', '', '']);
   155→   155→      setError('');
   156→   156→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   157→   157→    }
   158→   158→  }, [step]);
   159→   159→
   160→   160→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   161→   161→  const startSelfieCapture = useCallback(async () => {
   162→   162→    setStep('selfie');
   163→   163→    setError('');
   164→   164→    setBlinkDetected(false);
   165→   165→    setFaceVisible(false);
   166→   166→
   167→   167→    try {
   168→   168→      const stream = await navigator.mediaDevices.getUserMedia({
   169→   169→        video: { facingMode: 'user', width: 480, height: 640 },
   170→   170→        audio: false,
   171→   171→      });
   172→   172→      streamRef.current = stream;
   173→   173→
   174→   174→      if (videoRef.current) {
   175→   175→        videoRef.current.srcObject = stream;
   176→   176→        await videoRef.current.play();
   177→   177→      }
   178→   178→
   179→   179→      // Start blink detection if models are loaded
   180→   180→      if (videoRef.current && modelsReady) {
   181→   181→        // Small delay to let video stabilize
   182→   182→        setTimeout(() => {
   183→   183→          if (videoRef.current) {
   184→   184→            detectionLoopRef.current = startBlinkDetection(
   185→   185→              videoRef.current,
   186→   186→              () => {
   187→   187→                // Blink detected!
   188→   188→                setBlinkDetected(true);
   189→   189→              },
   190→   190→              (state) => {
   191→   191→                setFaceVisible(state.faceDetected);
   192→   192→              },
   193→   193→            );
   194→   194→          }
   195→   195→        }, 500);
   196→   196→      }
   197→   197→    } catch {
   198→   198→      if (useDemoMode) {
   199→   199→        setStep('location_check');
   200→   200→      } else {
   201→   201→        setError('Camera access denied. Please allow camera access.');
   202→   202→      }
   203→   203→    }
   204→   204→  }, [useDemoMode, modelsReady]);
   205→   205→
   206→   206→  const captureSelfie = useCallback(async () => {
   207→   207→    if (!videoRef.current || !canvasRef.current) return;
   208→   208→
   209→   209→    // Stop blink detection loop
   210→   210→    detectionLoopRef.current?.stop();
   211→   211→    detectionLoopRef.current = null;
   212→   212→
   213→   213→    const video = videoRef.current;
   214→   214→    const canvas = canvasRef.current;
   215→   215→    canvas.width = video.videoWidth;
   216→   216→    canvas.height = video.videoHeight;
   217→   217→
   218→   218→    const ctx = canvas.getContext('2d');
   219→   219→    if (!ctx) return;
   220→   220→
   221→   221→    // Mirror the image (selfie style)
   222→   222→    ctx.translate(canvas.width, 0);
   223→   223→    ctx.scale(-1, 1);
   224→   224→    ctx.drawImage(video, 0, 0);
   225→   225→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   226→   226→
   227→   227→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   228→   228→    setSelfieDataUrl(dataUrl);
   229→   229→
   230→   230→    // Stop camera stream
   231→   231→    if (streamRef.current) {
   232→   232→      streamRef.current.getTracks().forEach((t) => t.stop());
   233→   233→      streamRef.current = null;
   234→   234→    }
   235→   235→
   236→   236→    // Run quality checks (non-blocking)
   237→   237→    if (modelsReady) {
   238→   238→      try {
   239→   239→        const faceResult = await detectFace(canvas);
   240→   240→        const checks = checkImageQuality(canvas, faceResult);
   241→   241→        setQualityChecks(checks);
   242→   242→      } catch {
   243→   243→        // Quality check failed — continue without it
   244→   244→        setQualityChecks([]);
   245→   245→      }
   246→   246→    }
   247→   247→
   248→   248→    // Show preview for user to accept or retake
   249→   249→    setStep('selfie_preview');
   250→   250→  }, [modelsReady]);
   251→   251→
   252→   252→  const skipSelfie = useCallback(() => {
   253→   253→    if (streamRef.current) {
   254→   254→      streamRef.current.getTracks().forEach((t) => t.stop());
   255→   255→      streamRef.current = null;
   256→   256→    }
   257→   257→    setStep('location_check');
   258→   258→  }, []);
   259→   259→
   260→   260→  const retakeSelfie = useCallback(() => {
   261→   261→    setSelfieDataUrl(null);
   262→   262→    startSelfieCapture();
   263→   263→  }, [startSelfieCapture]);
   264→   264→
   265→   265→  // Accept selfie and proceed to liveness check (backend-ready)
   266→   266→  const acceptSelfie = useCallback(() => {
   267→   267→    setStep('selfie_processing');
   268→   268→    // TODO: Send selfieDataUrl to backend for liveness/face verification
   269→   269→    // For now, simulate processing delay
   270→   270→    setTimeout(() => {
   271→   271→      setStep('location_check');
   272→   272→    }, 1500);
   273→   273→  }, []);
   274→   274→
   275→   275→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   276→   276→  const startLocationCheck = useCallback(async () => {
   277→   277→    setStep('location_check');
   278→   278→    setError('');
   279→   279→
   280→   280→    try {
   281→   281→      if (useDemoMode) {
   282→   282→        // Demo mode — simulate GPS fetch
   283→   283→        await new Promise((r) => setTimeout(r, 2000));
   284→   284→        setStep('location_passed');
   285→   285→        return;
   286→   286→      }
   287→   287→
   288→   288→      // Real high-accuracy GPS check
   289→   289→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   290→   290→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   291→   291→          enableHighAccuracy: true,
   292→   292→          timeout: 15000,
   293→   293→          maximumAge: 0,
   294→   294→        });
   295→   295→      });
   296→   296→
   297→   297→      const accuracy = position.coords.accuracy;
   298→   298→      if (accuracy <= 100) {
   299→   299→        setStep('location_passed');
   300→   300→      } else {
   301→   301→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   302→   302→        setStep('location_failed');
   303→   303→      }
   304→   304→    } catch (err) {
   305→   305→      setError('Location check failed. Please enable location services and try again.');
   306→   306→      setStep('location_failed');
   307→   307→    }
   308→   308→  }, [useDemoMode]);
   309→   309→
   310→   310→  // ── Step 3: OTP Verification ──────────────────────────────────────
   311→   311→  const handleRequestOTP = useCallback(() => {
   312→   312→    setResendTimer(RESEND_COOLDOWN);
   313→   313→    setStep('otp_enter');
   314→   314→  }, []);
   315→   315→
   316→   316→  const handleOtpChange = useCallback(
   317→   317→    (index: number, value: string) => {
   318→   318→      const digit = value.replace(/\D/g, '');
   319→   319→
   320→   320→      if (digit.length > 1) {
   321→   321→        // Handle paste: fill as many slots as possible
   322→   322→        const digits = digit.slice(0, 4);
   323→   323→        const newOtp = [...otp];
   324→   324→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   325→   325→          newOtp[index + i] = digits[i];
   326→   326→        }
   327→   327→        setOtp(newOtp);
   328→   328→        setError('');
   329→   329→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   330→   330→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   331→   331→        return;
   332→   332→      }
   333→   333→
   334→   334→      const newOtp = [...otp];
   335→   335→      newOtp[index] = digit;
   336→   336→      setOtp(newOtp);
   337→   337→      setError('');
   338→   338→
   339→   339→      // Auto-advance to next input
   340→   340→      if (digit && index < 3) {
   341→   341→        inputRefs.current[index + 1]?.focus();
   342→   342→      }
   343→   343→    },
   344→   344→    [otp],
   345→   345→  );
   346→   346→
   347→   347→  const handleKeyDown = useCallback(
   348→   348→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   349→   349→      if (e.key === 'Backspace') {
   350→   350→        if (!otp[index] && index > 0) {
   351→   351→          // If current is empty, go back to previous
   352→   352→          const newOtp = [...otp];
   353→   353→          newOtp[index - 1] = '';
   354→   354→          setOtp(newOtp);
   355→   355→          inputRefs.current[index - 1]?.focus();
   356→   356→        } else {
   357→   357→          // Clear current
   358→   358→          const newOtp = [...otp];
   359→   359→          newOtp[index] = '';
   360→   360→          setOtp(newOtp);
   361→   361→        }
   362→   362→      }
   363→   363→    },
   364→   364→    [otp],
   365→   365→  );
   366→   366→
   367→   367→  const handlePaste = useCallback(
   368→   368→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   369→   369→      e.preventDefault();
   370→   370→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   371→   371→      if (!pasted) return;
   372→   372→
   373→   373→      const newOtp = [...otp];
   374→   374→      for (let i = 0; i < pasted.length && i < 4; i++) {
   375→   375→        newOtp[i] = pasted[i];
   376→   376→      }
   377→   377→      setOtp(newOtp);
   378→   378→      setError('');
   379→   379→
   380→   380→      const focusIdx = Math.min(pasted.length, 3);
   381→   381→      inputRefs.current[focusIdx]?.focus();
   382→   382→    },
   383→   383→    [otp],
   384→   384→  );
   385→   385→
   386→   386→  const handleVerify = useCallback(() => {
   387→   387→    const enteredOtp = otp.join('');
   388→   388→    if (enteredOtp.length !== 4) {
   389→   389→      setError('Please enter all 4 digits');
   390→   390→      return;
   391→   391→    }
   392→   392→
   393→   393→    setStep('otp_verifying');
   394→   394→    setError('');
   395→   395→
   396→   396→    // Simulate verification (demo: accept 1234)
   397→   397→    setTimeout(() => {
   398→   398→      if (enteredOtp === DEMO_OTP) {
   399→   399→        setStep('success');
   400→   400→        // Call onStart after a brief delay for the success animation
   401→   401→        setTimeout(() => {
   402→   402→          onStart(shift.id);
   403→   403→        }, 1500);
   404→   404→      } else {
   405→   405→        setError('Invalid OTP. Please try again.');
   406→   406→        setStep('otp_enter');
   407→   407→        setOtp(['', '', '', '']);
   408→   408→        inputRefs.current[0]?.focus();
   409→   409→      }
   410→   410→    }, 1800);
   411→   411→  }, [otp, shift.id, onStart]);
   412→   412→
   413→   413→  const handleResend = useCallback(() => {
   414→   414→    if (resendTimer > 0) return;
   415→   415→    setResendTimer(RESEND_COOLDOWN);
   416→   416→    setOtp(['', '', '', '']);
   417→   417→    setError('');
   418→   418→    inputRefs.current[0]?.focus();
   419→   419→  }, [resendTimer]);
   420→   420→
   421→   421→  const handleDialogClose = useCallback(
   422→   422→    (open: boolean) => {
   423→   423→      if (!open) onCancel();
   424→   424→    },
   425→   425→    [onCancel],
   426→   426→  );
   427→   427→
   428→   428→  // ── Sub-renderers ──────────────────────────────────────────────────
   429→   429→
   430→   430→  // Shift summary card
   431→   431→  const renderShiftSummary = () => (
   432→   432→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   433→   433→      <div className="flex items-center gap-2 text-muted-foreground">
   434→   434→        <Badge variant="outline" className="text-xs font-medium">
   435→   435→          {shift.serviceType}
   436→   436→        </Badge>
   437→   437→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   438→   438→      </div>
   439→   439→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   440→   440→        <div className="flex items-center gap-2">
   441→   441→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   442→   442→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   443→   443→        </div>
   444→   444→        {shift.address && (
   445→   445→          <div className="flex items-center gap-2">
   446→   446→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   447→   447→            <span className="truncate">{shift.address}</span>
   448→   448→          </div>
   449→   449→        )}
   450→   450→        <div className="flex items-center gap-2">
   451→   451→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   452→   452→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   453→   453→        </div>
   454→   454→      </div>
   455→   455→    </div>
   456→   456→  );
   457→   457→
   458→   458→  // Progress stepper — order: Selfie → Location → OTP
   459→   459→  const renderStepper = (currentStep: number) => (
   460→   460→    <div className="flex items-center justify-center gap-1 mb-2">
   461→   461→      {[
   462→   462→        { n: 1, label: 'Selfie' },
   463→   463→        { n: 2, label: 'Location' },
   464→   464→        { n: 3, label: 'OTP' },
   465→   465→      ].map((s, i) => (
   466→   466→        <div key={s.n} className="flex items-center gap-1">
   467→   467→          <div
   468→   468→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   469→   469→              currentStep >= s.n
   470→   470→                ? 'bg-blue-600 text-white'
   471→   471→                : 'bg-gray-100 text-gray-400'
   472→   472→            }`}
   473→   473→            title={s.label}
   474→   474→          >
   475→   475→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   476→   476→          </div>
   477→   477→          {i < 2 && (
   478→   478→            <div
   479→   479→              className={`w-8 h-0.5 ${
   480→   480→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   481→   481→              }`}
   482→   482→            />
   483→   483→          )}
   484→   484→        </div>
   485→   485→      ))}
   486→   486→    </div>
   487→   487→  );
   488→   488→
   489→   489→  // OTP input fields
   490→   490→  const renderOtpInputs = () => (
   491→   491→    <div className="flex justify-center gap-3">
   492→   492→      {otp.map((digit, i) => (
   493→   493→        <Input
   494→   494→          key={i}
   495→   495→          ref={(el) => { inputRefs.current[i] = el; }}
   496→   496→          type="text"
   497→   497→          inputMode="numeric"
   498→   498→          maxLength={4}
