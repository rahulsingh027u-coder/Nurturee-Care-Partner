     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react';
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→  DialogDescription,
    10→  DialogFooter,
    11→} from '@/components/ui/dialog';
    12→import { Input } from '@/components/ui/input';
    13→import { Button } from '@/components/ui/button';
    14→import { Badge } from '@/components/ui/badge';
    15→import {
    16→  MapPin,
    17→  ShieldCheck,
    18→  Loader2,
    19→  AlertCircle,
    20→  CheckCircle,
    21→  Camera,
    22→  Clock,
    23→  ArrowLeft,
    24→  IndianRupee,
    25→  Eye,
    26→  XCircle,

    28→  Phone,
    29→  ScanFace,
    30→} from 'lucide-react';
    31→
    32→import type { Shift } from '@/lib/care-partner-data';

    34→  loadModels,
    35→  startBlinkDetection,
    36→  detectFace,
    37→  checkImageQuality,
    38→  getQualityVerdict,
    39→} from '@/lib/face-detection';
    40→import type { QualityCheck, DetectionLoopHandle } from '@/lib/face-detection';
    41→
    42→const NURTUREE_HELPLINE = 'tel:+918045678900';
    43→
    44→// ── Types ────────────────────────────────────────────────────────────
    45→type CheckInStep =
    46→  | 'intro'
    47→  | 'selfie'
    48→  | 'selfie_preview'
    49→  | 'selfie_processing'
    50→  | 'location_check'
    51→  | 'location_passed'
    52→  | 'location_failed'
    53→  | 'otp_enter'
    54→  | 'otp_verifying'
    55→  | 'success'
    56→  | 'failed';
    57→
    58→interface ShiftCheckInProps {
    59→  shift: Shift;
    60→  onStart: (shiftId: string) => void;
    61→  onCancel: () => void;
    62→}
    63→
    64→// ── Helpers ──────────────────────────────────────────────────────────
    65→const DEMO_OTP = '1234';
    66→const RESEND_COOLDOWN = 30;
    67→
    68→function formatTime(time24: string): string {
    69→  const [hours, minutes] = time24.split(':').map(Number);
    70→  const period = hours >= 12 ? 'PM' : 'AM';
    71→  const hours12 = hours % 12 || 12;
    72→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    73→}
    74→
    75→function getFamilyLabel(category: string): string {
    76→  return category === 'child_care'
    77→    ? "Child's Parent"
    78→    : "Elder's Family Member";
    79→}
    80→
    81→// ── Component ────────────────────────────────────────────────────────
    82→export default function ShiftCheckIn({
    83→  shift,
    84→  onStart,
    85→  onCancel,
    86→}: ShiftCheckInProps) {
    87→  const [step, setStep] = useState<CheckInStep>('intro');
    88→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
    89→  const [otp, setOtp] = useState(['', '', '', '']);
    90→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
    91→  const [error, setError] = useState('');
    92→  const [useDemoMode, setUseDemoMode] = useState(false);
    93→  const [modelsReady, setModelsReady] = useState(false);
    94→  const [blinkDetected, setBlinkDetected] = useState(false);
    95→  const [faceVisible, setFaceVisible] = useState(false);
    96→  const [qualityChecks, setQualityChecks] = useState<QualityCheck[]>([]);
    97→
    98→  const videoRef = useRef<HTMLVideoElement | null>(null);
    99→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   100→  const streamRef = useRef<MediaStream | null>(null);
   101→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   102→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   103→  const detectionLoopRef = useRef<DetectionLoopHandle | null>(null);
   104→
   105→  const familyLabel = getFamilyLabel(shift.serviceCategory);
   106→
   107→  // ── Preload face detection models during intro ─────────────────
   108→  useEffect(() => {
   109→    if (step === 'intro' && !modelsReady) {
   110→      loadModels().then((loaded) => setModelsReady(loaded));
   111→    }
   112→  }, [step, modelsReady]);
   113→
   114→  // ── Cleanup camera stream + detection loop on unmount ──────────
   115→  useEffect(() => {
   116→    return () => {
   117→      if (streamRef.current) {
   118→        streamRef.current.getTracks().forEach((t) => t.stop());
   119→      }
   120→      if (timerRef.current) clearInterval(timerRef.current);
   121→      detectionLoopRef.current?.stop();
   122→    };
   123→  }, []);
   124→
   125→  // ── Auto-start location check when step reaches location_check ──
   126→  useEffect(() => {
   127→    if (step === 'location_check') {
   128→      startLocationCheck();
   129→    }
   130→  }, [step]); // intentionally only depend on step to trigger once per entry
   131→
   132→  // ── Resend countdown ───────────────────────────────────────────────
   133→  useEffect(() => {
   134→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   135→
   136→    timerRef.current = setInterval(() => {
   137→      setResendTimer((prev) => {
   138→        if (prev <= 1) {
   139→          if (timerRef.current) clearInterval(timerRef.current);
   140→          return 0;
   141→        }
   142→        return prev - 1;
   143→      });
   144→    }, 1000);
   145→
   146→    return () => {
   147→      if (timerRef.current) clearInterval(timerRef.current);
   148→    };
   149→  }, [step, resendTimer]);
   150→
   151→  // ── Reset OTP inputs ──────────────────────────────────────────────
   152→  useEffect(() => {
   153→    if (step === 'otp_enter') {
   154→      setOtp(['', '', '', '']);
   155→      setError('');
   156→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   157→    }
   158→  }, [step]);
   159→
   160→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   161→  const startSelfieCapture = useCallback(async () => {
   162→    setStep('selfie');
   163→    setError('');
   164→    setBlinkDetected(false);
   165→    setFaceVisible(false);
   166→
   167→    try {
   168→      const stream = await navigator.mediaDevices.getUserMedia({
   169→        video: { facingMode: 'user', width: 480, height: 640 },
   170→        audio: false,
   171→      });
   172→      streamRef.current = stream;
   173→
   174→      if (videoRef.current) {
   175→        videoRef.current.srcObject = stream;
   176→        await videoRef.current.play();
   177→      }
   178→
   179→      // Start blink detection if models are loaded
   180→      if (videoRef.current && modelsReady) {
   181→        // Small delay to let video stabilize
   182→        setTimeout(() => {
   183→          if (videoRef.current) {
   184→            detectionLoopRef.current = startBlinkDetection(
   185→              videoRef.current,
   186→              () => {
   187→                // Blink detected!
   188→                setBlinkDetected(true);
   189→              },
   190→              (state) => {
   191→                setFaceVisible(state.faceDetected);
   192→              },
   193→            );
   194→          }
   195→        }, 500);
   196→      }
   197→    } catch {
   198→      if (useDemoMode) {
   199→        setStep('location_check');
   200→      } else {
   201→        setError('Camera access denied. Please allow camera access.');
   202→      }
   203→    }
   204→  }, [useDemoMode, modelsReady]);
   205→
   206→  const captureSelfie = useCallback(async () => {
   207→    if (!videoRef.current || !canvasRef.current) return;
   208→
   209→    // Stop blink detection loop
   210→    detectionLoopRef.current?.stop();
   211→    detectionLoopRef.current = null;
   212→
   213→    const video = videoRef.current;
   214→    const canvas = canvasRef.current;
   215→    canvas.width = video.videoWidth;
   216→    canvas.height = video.videoHeight;
   217→
   218→    const ctx = canvas.getContext('2d');
   219→    if (!ctx) return;
   220→
   221→    // Mirror the image (selfie style)
   222→    ctx.translate(canvas.width, 0);
   223→    ctx.scale(-1, 1);
   224→    ctx.drawImage(video, 0, 0);
   225→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   226→
   227→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   228→    setSelfieDataUrl(dataUrl);
   229→
   230→    // Stop camera stream
   231→    if (streamRef.current) {
   232→      streamRef.current.getTracks().forEach((t) => t.stop());
   233→      streamRef.current = null;
   234→    }
   235→
   236→    // Run quality checks (non-blocking)
   237→    if (modelsReady) {
   238→      try {
   239→        const faceResult = await detectFace(canvas);
   240→        const checks = checkImageQuality(canvas, faceResult);
   241→        setQualityChecks(checks);
   242→      } catch {
   243→        // Quality check failed — continue without it
   244→        setQualityChecks([]);
   245→      }
   246→    }
   247→
   248→    // Show preview for user to accept or retake
   249→    setStep('selfie_preview');
   250→  }, [modelsReady]);
   251→
   252→  const skipSelfie = useCallback(() => {
   253→    if (streamRef.current) {
   254→      streamRef.current.getTracks().forEach((t) => t.stop());
   255→      streamRef.current = null;
   256→    }
   257→    setStep('location_check');
   258→  }, []);
   259→
   260→  const retakeSelfie = useCallback(() => {
   261→    setSelfieDataUrl(null);
   262→    startSelfieCapture();
   263→  }, [startSelfieCapture]);
   264→
   265→  // Accept selfie and proceed to liveness check (backend-ready)
   266→  const acceptSelfie = useCallback(() => {
   267→    setStep('selfie_processing');
   268→    // TODO: Send selfieDataUrl to backend for liveness/face verification
   269→    // For now, simulate processing delay
   270→    setTimeout(() => {
   271→      setStep('location_check');
   272→    }, 1500);
   273→  }, []);
   274→
   275→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   276→  const startLocationCheck = useCallback(async () => {
   277→    setStep('location_check');
   278→    setError('');
   279→
   280→    try {
   281→      if (useDemoMode) {
   282→        // Demo mode — simulate GPS fetch
   283→        await new Promise((r) => setTimeout(r, 2000));
   284→        setStep('location_passed');
   285→        return;
   286→      }
   287→
   288→      // Real high-accuracy GPS check
   289→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   290→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   291→          enableHighAccuracy: true,
   292→          timeout: 15000,
   293→          maximumAge: 0,
   294→        });
   295→      });
   296→
   297→      const accuracy = position.coords.accuracy;
   298→      if (accuracy <= 100) {
   299→        setStep('location_passed');
   300→      } else {
   301→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   302→        setStep('location_failed');
   303→      }
   304→    } catch (err) {
   305→      setError('Location check failed. Please enable location services and try again.');
   306→      setStep('location_failed');
   307→    }
   308→  }, [useDemoMode]);
   309→
   310→  // ── Step 3: OTP Verification ──────────────────────────────────────
   311→  const handleRequestOTP = useCallback(() => {
   312→    setResendTimer(RESEND_COOLDOWN);
   313→    setStep('otp_enter');
   314→  }, []);
   315→
   316→  const handleOtpChange = useCallback(
   317→    (index: number, value: string) => {
   318→      const digit = value.replace(/\D/g, '');
   319→
   320→      if (digit.length > 1) {
   321→        // Handle paste: fill as many slots as possible
   322→        const digits = digit.slice(0, 4);
   323→        const newOtp = [...otp];
   324→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   325→          newOtp[index + i] = digits[i];
   326→        }
   327→        setOtp(newOtp);
   328→        setError('');
   329→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   330→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   331→        return;
   332→      }
   333→
   334→      const newOtp = [...otp];
   335→      newOtp[index] = digit;
   336→      setOtp(newOtp);
   337→      setError('');
   338→
   339→      // Auto-advance to next input
   340→      if (digit && index < 3) {
   341→        inputRefs.current[index + 1]?.focus();
   342→      }
   343→    },
   344→    [otp],
   345→  );
   346→
   347→  const handleKeyDown = useCallback(
   348→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   349→      if (e.key === 'Backspace') {
   350→        if (!otp[index] && index > 0) {
   351→          // If current is empty, go back to previous
   352→          const newOtp = [...otp];
   353→          newOtp[index - 1] = '';
   354→          setOtp(newOtp);
   355→          inputRefs.current[index - 1]?.focus();
   356→        } else {
   357→          // Clear current
   358→          const newOtp = [...otp];
   359→          newOtp[index] = '';
   360→          setOtp(newOtp);
   361→        }
   362→      }
   363→    },
   364→    [otp],
   365→  );
   366→
   367→  const handlePaste = useCallback(
   368→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   369→      e.preventDefault();
   370→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   371→      if (!pasted) return;
   372→
   373→      const newOtp = [...otp];
   374→      for (let i = 0; i < pasted.length && i < 4; i++) {
   375→        newOtp[i] = pasted[i];
   376→      }
   377→      setOtp(newOtp);
   378→      setError('');
   379→
   380→      const focusIdx = Math.min(pasted.length, 3);
   381→      inputRefs.current[focusIdx]?.focus();
   382→    },
   383→    [otp],
   384→  );
   385→
   386→  const handleVerify = useCallback(() => {
   387→    const enteredOtp = otp.join('');
   388→    if (enteredOtp.length !== 4) {
   389→      setError('Please enter all 4 digits');
   390→      return;
   391→    }
   392→
   393→    setStep('otp_verifying');
   394→    setError('');
   395→
   396→    // Simulate verification (demo: accept 1234)
   397→    setTimeout(() => {
   398→      if (enteredOtp === DEMO_OTP) {
   399→        setStep('success');
   400→        // Call onStart after a brief delay for the success animation
   401→        setTimeout(() => {
   402→          onStart(shift.id);
   403→        }, 1500);
   404→      } else {
   405→        setError('Invalid OTP. Please try again.');
   406→        setStep('otp_enter');
   407→        setOtp(['', '', '', '']);
   408→        inputRefs.current[0]?.focus();
   409→      }
   410→    }, 1800);
   411→  }, [otp, shift.id, onStart]);
   412→
   413→  const handleResend = useCallback(() => {
   414→    if (resendTimer > 0) return;
   415→    setResendTimer(RESEND_COOLDOWN);
   416→    setOtp(['', '', '', '']);
   417→    setError('');
   418→    inputRefs.current[0]?.focus();
   419→  }, [resendTimer]);
   420→
   421→  const handleDialogClose = useCallback(
   422→    (open: boolean) => {
   423→      if (!open) onCancel();
   424→    },
   425→    [onCancel],
   426→  );
   427→
   428→  // ── Sub-renderers ──────────────────────────────────────────────────
   429→
   430→  // Shift summary card
   431→  const renderShiftSummary = () => (
   432→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   433→      <div className="flex items-center gap-2 text-muted-foreground">
   434→        <Badge variant="outline" className="text-xs font-medium">
   435→          {shift.serviceType}
   436→        </Badge>
   437→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   438→      </div>
   439→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   440→        <div className="flex items-center gap-2">
   441→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   442→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   443→        </div>
   444→        {shift.address && (
   445→          <div className="flex items-center gap-2">
   446→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   447→            <span className="truncate">{shift.address}</span>
   448→          </div>
   449→        )}
   450→        <div className="flex items-center gap-2">
   451→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   452→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   453→        </div>
   454→      </div>
   455→    </div>
   456→  );
   457→
   458→  // Progress stepper — order: Selfie → Location → OTP
   459→  const renderStepper = (currentStep: number) => (
   460→    <div className="flex items-center justify-center gap-1 mb-2">
   461→      {[
   462→        { n: 1, label: 'Selfie' },
   463→        { n: 2, label: 'Location' },
   464→        { n: 3, label: 'OTP' },
   465→      ].map((s, i) => (
   466→        <div key={s.n} className="flex items-center gap-1">
   467→          <div
   468→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   469→              currentStep >= s.n
   470→                ? 'bg-blue-600 text-white'
   471→                : 'bg-gray-100 text-gray-400'
   472→            }`}
   473→            title={s.label}
   474→          >
   475→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   476→          </div>
   477→          {i < 2 && (
   478→            <div
   479→              className={`w-8 h-0.5 ${
   480→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   481→              }`}
   482→            />
   483→          )}
   484→        </div>
   485→      ))}
   486→    </div>
   487→  );
   488→
   489→  // OTP input fields
   490→  const renderOtpInputs = () => (
   491→    <div className="flex justify-center gap-3">
   492→      {otp.map((digit, i) => (
   493→        <Input
   494→          key={i}
   495→          ref={(el) => { inputRefs.current[i] = el; }}
   496→          type="text"
   497→          inputMode="numeric"
   498→          maxLength={4}
   499→          value={digit}
   500→          onChange={(e) => handleOtpChange(i, e.target.value)}
   501→          onKeyDown={(e) => handleKeyDown(i, e)}
   502→          onPaste={handlePaste}
   503→          disabled={step === 'otp_verifying'}
   504→          className={
   505→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   506→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   507→          }
   508→          aria-label={`OTP digit ${i + 1}`}
   509→        />
   510→      ))}
   511→    </div>
   512→  );
   513→
   514→  // Hidden canvas for selfie capture
   515→  const renderHiddenCanvas = () => (
   516→    <canvas ref={canvasRef} className="hidden" />
   517→  );
   518→
   519→  // ── RENDER: Intro ──────────────────────────────────────────────────
   520→  if (step === 'intro') {
   521→    return (
   522→      <Dialog open onOpenChange={handleDialogClose}>
   523→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   524→          <div className="flex flex-col min-h-full">
   525→            <DialogHeader className="text-center sm:text-center">
   526→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   527→                <ShieldCheck className="size-6 text-blue-600" />
   528→              </div>
   529→              <DialogTitle className="text-lg">Shift Check-In</DialogTitle>
   530→              <DialogDescription className="text-xs">
   531→                Daily attendance is mandatory before starting any shift.
   532→              </DialogDescription>
   533→            </DialogHeader>
   534→
   535→            <div className="flex-1 space-y-3">
   536→              {renderStepper(0)}
   537→
   538→              {/* 3-step overview */}
   539→              <div className="space-y-2">
   540→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   541→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">1</div>
   542→                  <div>
   543→                    <p className="text-xs font-medium text-blue-700">Live Selfie</p>
   544→                    <p className="text-[11px] text-blue-600">Liveness detection for identity</p>
   545→                  </div>
   546→                </div>
   547→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   548→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">2</div>
   549→                  <div>
   550→                    <p className="text-xs font-medium text-blue-700">Location Verification</p>
   551→                    <p className="text-[11px] text-blue-600">High accuracy GPS check</p>
   552→                  </div>
   553→                </div>
   554→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   555→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">3</div>
   556→                  <div>
   557→                    <p className="text-xs font-medium text-blue-700">OTP Confirmation</p>
   558→                    <p className="text-[11px] text-blue-600">4-digit OTP with {familyLabel.toLowerCase()}</p>
   559→                  </div>
   560→                </div>
   561→              </div>
   562→
   563→              {renderShiftSummary()}
   564→
   565→              {/* Demo mode toggle */}
   566→              <div className="flex items-center gap-2 justify-center">
   567→                <button
   568→                  onClick={() => setUseDemoMode(!useDemoMode)}
   569→                  className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-blue-600 transition-colors"
   570→                >
   571→                  <Eye className="size-3.5" />
   572→                  {useDemoMode ? 'Demo Mode ON' : 'Enable Demo Mode'}
   573→                </button>
   574→                {useDemoMode && (
   575→                  <Badge variant="outline" className="text-[10px] bg-amber-50 border-amber-200 text-amber-700">
   576→                    DEMO
   577→                  </Badge>
   578→                )}
   579→              </div>
   580→            </div>
   581→
   582→            <DialogFooter className="mt-auto pt-4 border-t border-gray-100 flex-col gap-2 sm:flex-col sm:space-x-0">
   583→              <Button
   584→                onClick={startSelfieCapture}
   585→                className="w-full bg-blue-600 hover:bg-blue-700 text-white"
