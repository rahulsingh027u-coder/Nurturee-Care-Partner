   183→
   184→function loadDocsFromStorage(): VerificationDoc[] {
   185→  try {
   186→    // Version check — if storage format is outdated, clear and start fresh
   187→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   188→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   189→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   190→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   191→      return getInitialDocs();
   192→    }
   193→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   194→    if (stored) {
   195→      const parsed = JSON.parse(stored);
   196→      if (Array.isArray(parsed) && parsed.length === 4) {
   197→        // Validate structure — reject data from old format
   198→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   199→        if (isValid) return parsed;
   200→      }
   201→    }
   202→  } catch {
   203→    // fall through
   204→  }
   205→  return getInitialDocs();
   206→}
   207→
   208→function saveDocsToStorage(docs: VerificationDoc[]) {
   209→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   210→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   211→}

   232→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   233→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   234→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   235→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   236→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   237→  const [submitting, setSubmitting] = useState(false);
   238→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   239→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   240→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   241→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   242→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   243→
   244→  const currentYear = new Date().getFullYear();
   245→
   246→  // Clean up timers on unmount
   247→  useEffect(() => {
   248→    return () => {
   249→      Object.values(reviewTimersRef.current).forEach((timers) => {
   250→        timers.forEach(clearTimeout);
   251→      });
   252→    };
   253→  }, []);
   254→
   255→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   256→  useEffect(() => {
   257→    if (submittedType) {
   258→      const doc = docs.find((d) => d.type === submittedType);
   259→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   260→        setSubmittedType(null);
   261→        setReviewingType(null);
   262→      }
   263→    }
   264→    if (reviewingType) {
   265→      const doc = docs.find((d) => d.type === reviewingType);
   266→      if (!doc || doc.status !== 'under_review') {
   267→        setReviewingType(null);
   268→      }
   269→    }
   270→  }, [docs, submittedType, reviewingType]);
   271→
   272→  // Persist docs to localStorage whenever they change
   273→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   274→    setDocs((prev) => {
   275→      const updated = updater(prev);
   276→      saveDocsToStorage(updated);
   277→      return updated;
   278→    });
   279→  }, []);
   280→
   281→  // ─── Backend Review Simulation ───────────────────────────────────────────
   282→
   283→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   284→    // Clear any existing timers for this type
   285→    if (reviewTimersRef.current[type]) {
   286→      reviewTimersRef.current[type].forEach(clearTimeout);
   287→    }
   288→    reviewTimersRef.current[type] = [];
   289→
   290→    const doc = docs.find((d) => d.type === type);
   291→    if (!doc) return;
   292→
   293→    // Timer 1: After 3s → under_review
   294→    const t1 = setTimeout(() => {
   295→      setReviewingType(type);
   296→      updateDocs((prev) =>
   297→        prev.map((d) =>
   298→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   299→        )
   300→      );
   301→    }, 3000);
   302→
   303→    // Timer 2: After 5-7s total → resolve
   304→    const resolveDelay = 5000 + Math.random() * 2000;
   305→    const t2 = setTimeout(() => {
   306→      const currentDoc = docs.find((d) => d.type === type);
   307→      if (!currentDoc) return;
   308→
   309→      const currentAttempts = currentDoc.attempts.length;
   310→      let willApprove: boolean;
   311→
   312→      if (type === 'aadhaar') {
   313→        // Aadhaar always approve on first attempt (demo-friendly)
   314→        willApprove = true;
   315→      } else {
   316→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   317→        if (attemptNumber === 1) {
   318→          willApprove = Math.random() < 0.4;
   319→        } else if (attemptNumber === 2) {
   320→          willApprove = Math.random() < 0.7;
   321→        } else {
   322→          willApprove = true;
   323→        }
   324→      }
   325→
   326→      if (willApprove) {
   327→        // Approve
   328→        setApprovingType(type);
   329→        updateDocs((prev) =>
   330→          prev.map((d) => {
   331→            if (d.type !== type) return d;
   332→            return {
   333→              ...d,
   334→              status: 'approved' as VerificationStatus,
   335→              attempts: d.attempts.map((a) =>
   336→                a.attemptNumber === attemptNumber
   337→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   338→                  : a
   339→              ),
   340→            };
   341→          })
   342→        );
   343→        setReviewingType(null);
   344→
   345→        // If Aadhaar approved and onComplete exists, call it after delay
   346→        if (type === 'aadhaar' && onComplete) {
   347→          setTimeout(() => {
   348→            setApprovingType(null);
   349→            setSubmittedType(null);
   350→            onComplete();
   351→          }, 1000);
   352→        } else {
   353→          // For optional: show success briefly, then allow going back
   354→          setTimeout(() => {
   355→            setApprovingType(null);
   356→            setSubmittedType(null);
   357→            setActiveType(null);
   358→          }, 2000);
   359→        }
   360→      } else {
   361→        // Reject
   362→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   363→        updateDocs((prev) =>
   364→          prev.map((d) => {
   365→            if (d.type !== type) return d;
   366→            const updatedAttempts = d.attempts.map((a) =>
   367→              a.attemptNumber === attemptNumber
   368→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   369→                : a
   370→            );
   371→            // Check if max attempts reached
   372→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   373→            return {
   374→              ...d,
   375→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   376→              attempts: updatedAttempts,
   377→              currentDraft: { documents: [], formFields: {} },
   378→            };
   379→          })
   380→        );
   381→        setReviewingType(null);
   382→        // Transition from submitted view to allow re-upload
   383→        setTimeout(() => {
   384→          setSubmittedType(null);
   385→        }, 300);
   386→      }
   387→    }, resolveDelay);
   388→
   389→    reviewTimersRef.current[type] = [t1, t2];
   390→  }, [docs, onComplete, updateDocs]);
   391→
   392→  // ─── Handlers ────────────────────────────────────────────────────────────
   393→
   394→  const handleSelectType = (type: VerificationType) => {
   395→    const doc = docs.find((d) => d.type === type);
   396→    if (!doc) return;

   398→    setActiveType(type);
   399→    // Restore draft if exists, otherwise use empty

   403→  };
   404→
   405→  const handleBack = () => {

   407→    if (activeType) {
   408→      updateDocs((prev) =>
   409→        prev.map((d) =>
   410→          d.type === activeType

   412→            : d
   413→        )
   414→      );
   415→    }
   416→    // Clean up timers for this type if we navigate back
   417→    if (activeType && reviewTimersRef.current[activeType]) {
   418→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   419→      if (!submittedType && !reviewingType) {
   420→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   421→        delete reviewTimersRef.current[activeType];
   422→      }
   423→    }
   424→    setActiveType(null);
   425→    setFormFields({});
   426→    setUploadedFiles([]);
   427→    setSubmittedType(null);
   428→    setReviewingType(null);
   429→  };
   430→
   431→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   432→  const fileInputId = `file-upload-${activeType || 'default'}`;
   433→
   434→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   435→    const files = e.target.files;
   436→    if (!files || files.length === 0) return;
   437→
   438→    const config = activeType ? verificationConfig[activeType] : null;
   439→    const newFiles: Array<{ id: string; name: string; size: number; type: string }> = [];
   440→
   441→    for (let i = 0; i < files.length; i++) {
   442→      const file = files[i];
   443→      // Validate file type
   444→      if (config && !config.allowedTypes.includes(file.type)) {
   445→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   446→        continue;
   447→      }
   448→      // Validate file size
   449→      if (config && file.size > config.maxFileSize) {
   450→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   451→        continue;
   452→      }
   453→      newFiles.push({
   454→        id: `file-${Date.now()}-${i}`,
   455→        name: file.name,
   456→        size: file.size,
   457→        type: file.type,
   458→      });
   459→    }
   460→
   461→    if (newFiles.length > 0) {
   462→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   463→    }
   464→
   465→    // Reset input so same file can be re-selected
   466→    e.target.value = '';
   467→  };
   468→
   469→  const handleRemoveFile = (fileId: string) => {
   470→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   471→  };
   472→
   473→  const handleSubmit = async () => {
   474→    if (!activeType) return;
   475→    setSubmitting(true);
   476→
   477→    // Simulate submission delay
   478→    await new Promise((resolve) => setTimeout(resolve, 1500));
   479→
   480→    const doc = docs.find((d) => d.type === activeType);
   481→    const nextAttemptNumber = (doc?.attempts?.length || 0) + 1;
   482→
   483→    const newAttempt = {
   484→      attemptNumber: nextAttemptNumber,
   485→      submittedAt: new Date().toISOString(),
   486→      documents: [...uploadedFiles],
   487→      formFields: { ...formFields },
   488→      status: 'submitted' as const,
   489→    };
   490→
   491→    updateDocs((prev) =>
   492→      prev.map((d) => {
   493→        if (d.type !== activeType) return d;
   494→        return {
   495→          ...d,
   496→          status: 'submitted' as VerificationStatus,
   497→          attempts: [...d.attempts, newAttempt],
   498→          currentDraft: { documents: [], formFields: {} },
   499→        };
   500→      })
   501→    );
   502→
   503→    setSubmitting(false);
   504→    setSubmittedType(activeType);
   505→
   506→    // Start the backend review simulation
   507→    startReviewSimulation(activeType, nextAttemptNumber);
   508→  };
   509→
   510→  // ─── Derived State ────────────────────────────────────────────────────────
   511→
   512→  const aadhaarDoc = docs.find((d) => d.type === 'aadhaar');
   513→  const aadhaarApproved = aadhaarDoc?.status === 'approved';
   514→
   515→  // ─── View 1: Overview ────────────────────────────────────────────────────
   516→
   517→  if (!activeType) {
   518→    const allTypes = (Object.keys(verificationConfig) as VerificationType[]);
   519→    const mandatoryTypes = allTypes.filter((t) => verificationConfig[t].mandatory);
   520→    const optionalTypes = allTypes.filter((t) => !verificationConfig[t].mandatory);
   521→    const optionalPending = optionalTypes.filter((t) => {
   522→      const doc = docs.find((d) => d.type === t);
   523→      return doc?.status !== 'approved' && doc?.status !== 'max_attempts_reached';
   524→    });
   525→
   526→    return (
   527→      <div className={`${optionalMode ? '' : 'min-h-screen'} bg-gray-50 flex flex-col`}>
   528→        <div className="flex-1 flex flex-col items-center p-4 pt-8">
   529→          <div className="w-full max-w-2xl space-y-6">
   530→            {/* Header */}
   531→            <div className="text-center space-y-3">
   532→              <div className="inline-flex items-center justify-center size-16 rounded-2xl bg-gradient-to-br from-blue-600 to-indigo-600 text-white mb-2">
   533→                <Shield className="size-8" />
   534→              </div>
   535→              <h1 className="text-2xl font-bold text-gray-900">
   536→                {aadhaarApproved ? 'Document Verification' : 'Complete Your Verification'}
   537→              </h1>
   538→              <p className="text-sm text-gray-500 max-w-md mx-auto">
