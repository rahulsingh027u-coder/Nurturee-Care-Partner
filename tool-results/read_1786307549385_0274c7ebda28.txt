    51→function formatCurrency(amount: number): string {
    52→  return '₹' + amount.toLocaleString('en-IN');
    53→}

    55→function getGreeting(): string {
    56→  const hour = new Date().getHours();
    57→  if (hour < 12) return 'Good Morning';
    58→  if (hour < 17) return 'Good Afternoon';
    59→  return 'Good Evening';
    60→}
    61→
    62→function formatDate(): string {
    63→  const now = new Date();
    64→  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
    65→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
    66→  return `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
    67→}
