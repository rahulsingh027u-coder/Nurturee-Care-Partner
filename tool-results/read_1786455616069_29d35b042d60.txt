    71→function getDateGroupLabel(dateStr: string): string {
    72→  const date = parseISO(dateStr);
    73→  if (isToday(date)) return 'Today';
    74→  if (isTomorrow(date)) return 'Tomorrow';
    75→  if (isYesterday(date)) return 'Yesterday';
    76→  return format(date, 'EEE, MMM d, yyyy');
    77→}
    78→
    79→function getDateGroupSortKey(dateStr: string): number {
    80→  const date = parseISO(dateStr);
    81→  if (isToday(date)) return 0;
    82→  if (isTomorrow(date)) return 1;
    83→  if (isYesterday(date)) return -1;
    84→  return date.getTime();
    85→}

    87→function reportReasonLabel(value: string): string {
    88→  const map: Record<string, string> = {
    89→    client_not_available: 'Client not available',
    90→    safety_concern: 'Safety concern',
    91→    address_mismatch: 'Address mismatch',
    92→    service_dispute: 'Service dispute',
    93→    equipment_issue: 'Equipment issue',
    94→    other: 'Other',
    95→  };
    96→  return map[value] || value;
    97→}
