    35→const avatarColors = ['blue', 'violet', 'emerald', 'amber', 'rose', 'cyan'] as const;
    36→
    37→type AvatarColor = (typeof avatarColors)[number];
    38→
    39→const avatarColorMap: Record<AvatarColor, { bg: string; text: string }> = {
    40→  blue: { bg: 'bg-blue-100', text: 'text-blue-700' },
    41→  violet: { bg: 'bg-violet-100', text: 'text-violet-700' },
    42→  emerald: { bg: 'bg-emerald-100', text: 'text-emerald-700' },
    43→  amber: { bg: 'bg-amber-100', text: 'text-amber-700' },
    44→  rose: { bg: 'bg-rose-100', text: 'text-rose-700' },
    45→  cyan: { bg: 'bg-cyan-100', text: 'text-cyan-700' },
    46→};
    47→
    48→function getAvatarColor(name: string): AvatarColor {
    49→  let hash = 0;
    50→  for (let i = 0; i < name.length; i++) {
    51→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    52→  }
    53→  return avatarColors[Math.abs(hash) % avatarColors.length];
    54→}
    55→
    56→function getInitials(name: string): string {
    57→  return name
    58→    .split(' ')
    59→    .map((part) => part[0])
    60→    .join('')
    61→    .toUpperCase()
    62→    .slice(0, 2);
    63→}
    64→
    65→function formatDate(dateStr: string): string {
    66→  const date = new Date(dateStr);
    67→  return date.toLocaleDateString('en-IN', {
    68→    day: 'numeric',
    69→    month: 'short',
    70→    year: 'numeric',
    71→  });
    72→}
