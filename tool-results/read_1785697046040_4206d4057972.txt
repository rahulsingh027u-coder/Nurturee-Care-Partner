    91→    91→function formatCurrency(amount: number): string {
    92→    92→  return new Intl.NumberFormat("en-IN", {
    93→    93→    style: "currency",
    94→    94→    currency: "INR",
    95→    95→    minimumFractionDigits: 0,
    96→    96→    maximumFractionDigits: 0,
    97→    97→  }).format(amount);
    98→    98→}
    99→    99→
   100→   100→function getGreeting(): string {
   101→   101→  const hour = new Date().getHours();
   102→   102→  if (hour < 12) return "Good Morning";
   103→   103→  if (hour < 17) return "Good Afternoon";
   104→   104→  return "Good Evening";
   105→   105→}
   106→   106→
   107→   107→function formatTodayDate(): string {
   108→   108→  return new Date().toLocaleDateString("en-IN", {
   109→   109→    weekday: "long",
   110→   110→    year: "numeric",
   111→   111→    month: "long",
   112→   112→    day: "numeric",
   113→   113→  });
   114→   114→}
   115→   115→
   116→   116→function getStatusBadgeClasses(status: string): string {
   117→   117→  switch (status) {
   118→   118→    case "confirmed":
   119→   119→      return "bg-blue-100 text-blue-700 border-blue-200";
   120→   120→    case "in-progress":
   121→   121→      return "bg-blue-100 text-blue-800 border-blue-200";
   122→   122→    case "completed":
   123→   123→      return "bg-emerald-100 text-emerald-700 border-emerald-200";
   124→   124→    case "cancelled":
   125→   125→    case "no-show":
   126→   126→      return "bg-red-100 text-red-700 border-red-200";
   127→   127→    case "pending":
   128→   128→    default:
   129→   129→      return "bg-amber-100 text-amber-700 border-amber-200";
   130→   130→  }
   131→   131→}
   132→   132→
   133→   133→function getStatusLabel(status: string): string {
   134→   134→  switch (status) {
   135→   135→    case "in-progress":
   136→   136→      return "In Progress";
   137→   137→    case "no-show":
   138→   138→      return "No Show";
   139→   139→    default:
   140→   140→      return status.charAt(0).toUpperCase() + status.slice(1);
   141→   141→  }
   142→   142→}
