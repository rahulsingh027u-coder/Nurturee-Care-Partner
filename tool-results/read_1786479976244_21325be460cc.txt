    69→    69→function getDateGroupLabel(dateStr: string): string {
    70→    70→  const date = parseISO(dateStr);
    71→    71→  if (isToday(date)) return 'Today';
    72→    72→  if (isTomorrow(date)) return 'Tomorrow';
    73→    73→  if (isYesterday(date)) return 'Yesterday';
    74→    74→  return format(date, 'EEE, MMM d, yyyy');
    75→    75→}
    76→    76→
    77→    77→function getDateGroupSortKey(dateStr: string): number {
    78→    78→  const date = parseISO(dateStr);
    79→    79→  if (isToday(date)) return 0;
    80→    80→  if (isTomorrow(date)) return 1;
    81→    81→  if (isYesterday(date)) return -1;
    82→    82→  return date.getTime();
    83→    83→}
