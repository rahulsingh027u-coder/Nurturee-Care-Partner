    57→function formatCurrency(amount: number): string {
    58→  return '₹' + amount.toLocaleString('en-IN');
    59→}
    60→
    61→function getGreeting(): string {
    62→  const hour = new Date().getHours();
    63→  if (hour < 12) return 'Good Morning';
    64→  if (hour < 17) return 'Good Afternoon';
    65→  return 'Good Evening';
    66→}
    67→
    68→function formatDate(): string {
    69→  const now = new Date();
    70→  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
    71→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
    72→  return `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
    73→}

   561→                    </div>

   572→        className="border-gray-100 shadow-sm hover:border-blue-200 hover:shadow-md transition-all cursor-pointer"
   573→        onClick={() => onNavigate('earnings')}

   626→            <>

   628→                <SheetTitle className="flex items-center gap-2">
   629→                  <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-blue-50">

   636→              <div className="mt-4 space-y-4">
   637→                {/* Status & Date */}

   716→                    variant="outline"
   717→                    size="sm"
   718→                    className="rounded-lg text-xs ml-auto"

   723→                  </Button>
   724→                </div>
   725→              </div>
   726→            </>
