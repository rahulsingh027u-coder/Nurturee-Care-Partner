    65→    65→function maskPhone(phone: string): string {
    66→    66→  const digits = phone.replace(/\D/g, '');
    67→    67→  if (digits.length < 4) return phone;
    68→    68→  const visibleStart = digits.slice(0, 2);
    69→    69→  const visibleEnd = digits.slice(-2);
    70→    70→  const masked = '*'.repeat(digits.length - 4);
    71→    71→  return `${visibleStart} ${masked} ${visibleEnd}`;
    72→    72→}
    73→    73→
    74→    74→function formatTime(time24: string): string {
    75→    75→  const [hours, minutes] = time24.split(':').map(Number);
    76→    76→  const period = hours >= 12 ? 'PM' : 'AM';
    77→    77→  const hours12 = hours % 12 || 12;
    78→    78→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    79→    79→}
    80→    80→
    81→    81→function getFamilyLabel(category: string): string {
    82→    82→  return category === 'child_care'
    83→    83→    ? "Child's Parent"
    84→    84→    : "Elder's Family Member";
    85→    85→}
