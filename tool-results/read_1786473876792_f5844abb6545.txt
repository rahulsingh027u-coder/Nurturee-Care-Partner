     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react';
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→  DialogDescription,
    10→  DialogFooter,
    11→} from '@/components/ui/dialog';
    12→import { Input } from '@/components/ui/input';
    13→import { Button } from '@/components/ui/button';
    14→import { Badge } from '@/components/ui/badge';
    15→import {
    16→  MapPin,
    17→  ShieldCheck,
    18→  Loader2,
    19→  AlertCircle,
    20→  CheckCircle,
    21→  Camera,
    22→  Clock,
    23→  ArrowLeft,
    24→  IndianRupee,
    25→  Eye,
    26→  XCircle,
    27→  RotateCcw,
    28→  Phone,
    29→} from 'lucide-react';
    30→
    31→import type { Shift } from '@/lib/care-partner-data';
    32→import {
    33→  checkImageQuality,
    34→} from '@/lib/face-detection';
    35→import type { QualityCheck } from '@/lib/face-detection';
    36→
    37→const NURTUREE_HELPLINE = 'tel:+918045678900';
    38→
    39→// ── Types ────────────────────────────────────────────────────────────
    40→type CheckInStep =
    41→  | 'intro'
    42→  | 'selfie'
    43→  | 'selfie_preview'
    44→  | 'selfie_processing'
    45→  | 'location_check'
    46→  | 'location_passed'
    47→  | 'location_failed'
    48→  | 'otp_enter'
    49→  | 'otp_verifying'
    50→  | 'success'
    51→  | 'failed';
    52→
    53→interface ShiftCheckInProps {
    54→  shift: Shift;
    55→  onStart: (shiftId: string) => void;
    56→  onCancel: () => void;
    57→}
    58→
    59→// ── Helpers ──────────────────────────────────────────────────────────
    60→const DEMO_OTP = '1234';
    61→const RESEND_COOLDOWN = 30;
    62→
    63→function formatTime(time24: string): string {
    64→  const [hours, minutes] = time24.split(':').map(Number);
    65→  const period = hours >= 12 ? 'PM' : 'AM';
    66→  const hours12 = hours % 12 || 12;
    67→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    68→}
    69→
    70→function getFamilyLabel(category: string): string {
    71→  return category === 'child_care'
    72→    ? "Child's Parent"
    73→    : "Elder's Family Member";
    74→}
    75→
    76→// ── Component ────────────────────────────────────────────────────────
    77→export default function ShiftCheckIn({
    78→  shift,
    79→  onStart,
    80→  onCancel,
    81→}: ShiftCheckInProps) {
    82→  const [step, setStep] = useState<CheckInStep>('intro');
    83→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
    84→  const [otp, setOtp] = useState(['', '', '', '']);
    85→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
    86→  const [error, setError] = useState('');
    87→  const [useDemoMode, setUseDemoMode] = useState(false);
    88→  const [qualityChecks, setQualityChecks] = useState<QualityCheck[]>([]);
    89→
    90→  const videoRef = useRef<HTMLVideoElement | null>(null);
    91→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    92→  const streamRef = useRef<MediaStream | null>(null);
    93→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
    94→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    95→
    96→  const familyLabel = getFamilyLabel(shift.serviceCategory);
    97→
    98→  // ── Cleanup camera stream on unmount ──────────
    99→  useEffect(() => {
   100→    return () => {
   101→      if (streamRef.current) {
   102→        streamRef.current.getTracks().forEach((t) => t.stop());
   103→      }
   104→      if (timerRef.current) clearInterval(timerRef.current);
   105→    };
   106→  }, []);
   107→
   108→  // ── Auto-start location check when step reaches location_check ──
   109→  useEffect(() => {
   110→    if (step === 'location_check') {
   111→      startLocationCheck();
   112→    }
   113→  }, [step]); // intentionally only depend on step to trigger once per entry
   114→
   115→  // ── Resend countdown ───────────────────────────────────────────────
   116→  useEffect(() => {
   117→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   118→
   119→    timerRef.current = setInterval(() => {
   120→      setResendTimer((prev) => {
   121→        if (prev <= 1) {
   122→          if (timerRef.current) clearInterval(timerRef.current);
   123→          return 0;
   124→        }
   125→        return prev - 1;
   126→      });
   127→    }, 1000);
   128→
   129→    return () => {
   130→      if (timerRef.current) clearInterval(timerRef.current);
   131→    };
   132→  }, [step, resendTimer]);
   133→
   134→  // ── Reset OTP inputs ──────────────────────────────────────────────
   135→  useEffect(() => {
   136→    if (step === 'otp_enter') {
   137→      setOtp(['', '', '', '']);
   138→      setError('');
   139→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   140→    }
   141→  }, [step]);
   142→
   143→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   144→  const startSelfieCapture = useCallback(async () => {
   145→    setStep('selfie');
   146→    setError('');
   147→    setQualityChecks([]);
   148→
   149→    try {
   150→      const stream = await navigator.mediaDevices.getUserMedia({
   151→        video: { facingMode: 'user', width: 480, height: 640 },
   152→        audio: false,
   153→      });
   154→      streamRef.current = stream;
   155→
   156→      if (videoRef.current) {
   157→        const video = videoRef.current;
   158→        video.srcObject = stream;
   159→        await video.play();
   160→      }
   161→    } catch {
   162→      if (useDemoMode) {
   163→        setStep('location_check');
   164→      } else {
   165→        setError('Camera access denied. Please allow camera access.');
   166→      }
   167→    }
   168→  }, [useDemoMode]);
   169→
   170→  const captureSelfie = useCallback(async () => {
   171→    if (!videoRef.current || !canvasRef.current) return;
   172→
   173→    const video = videoRef.current;
   174→    const canvas = canvasRef.current;
   175→    canvas.width = video.videoWidth;
   176→    canvas.height = video.videoHeight;
   177→
   178→    const ctx = canvas.getContext('2d');
   179→    if (!ctx) return;
   180→
   181→    // Mirror the image (selfie style)
   182→    ctx.translate(canvas.width, 0);
   183→    ctx.scale(-1, 1);
   184→    ctx.drawImage(video, 0, 0);
   185→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   186→
   187→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   188→    setSelfieDataUrl(dataUrl);
   189→
   190→    // Stop camera stream
   191→    if (streamRef.current) {
   192→      streamRef.current.getTracks().forEach((t) => t.stop());
   193→      streamRef.current = null;
   194→    }
   195→
   196→    // Run quality checks (pure canvas — no external ML)
   197→    try {
   198→      const checks = checkImageQuality(canvas);
   199→      setQualityChecks(checks);
   200→    } catch {
   201→      setQualityChecks([]);
   202→    }
   203→
   204→    // Show preview for user to accept or retake
   205→    setStep('selfie_preview');
   206→  }, []);
   207→
   208→  const skipSelfie = useCallback(() => {
   209→    if (streamRef.current) {
   210→      streamRef.current.getTracks().forEach((t) => t.stop());
   211→      streamRef.current = null;
   212→    }
   213→    setStep('location_check');
   214→  }, []);
   215→
   216→  const retakeSelfie = useCallback(() => {
   217→    setSelfieDataUrl(null);
   218→    startSelfieCapture();
   219→  }, [startSelfieCapture]);
   220→
   221→  // Accept selfie and proceed to liveness check (backend-ready)
   222→  const acceptSelfie = useCallback(() => {
   223→    setStep('selfie_processing');
   224→    // TODO: Send selfieDataUrl to backend for liveness/face verification
   225→    // For now, simulate processing delay
   226→    setTimeout(() => {
   227→      setStep('location_check');
   228→    }, 1500);
   229→  }, []);
   230→
   231→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   232→  const startLocationCheck = useCallback(async () => {
   233→    setStep('location_check');
   234→    setError('');
   235→
   236→    try {
   237→      if (useDemoMode) {
   238→        // Demo mode — simulate GPS fetch
   239→        await new Promise((r) => setTimeout(r, 2000));
   240→        setStep('location_passed');
   241→        return;
   242→      }
   243→
   244→      // Real high-accuracy GPS check
   245→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   246→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   247→          enableHighAccuracy: true,
   248→          timeout: 15000,
   249→          maximumAge: 0,
   250→        });
   251→      });
   252→
   253→      const accuracy = position.coords.accuracy;
   254→      if (accuracy <= 100) {
   255→        setStep('location_passed');
   256→      } else {
   257→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   258→        setStep('location_failed');
   259→      }
   260→    } catch (err) {
   261→      setError('Location check failed. Please enable location services and try again.');
   262→      setStep('location_failed');
   263→    }
   264→  }, [useDemoMode]);
   265→
   266→  // ── Step 3: OTP Verification ──────────────────────────────────────
   267→  const handleRequestOTP = useCallback(() => {
   268→    setResendTimer(RESEND_COOLDOWN);
   269→    setStep('otp_enter');
   270→  }, []);
   271→
   272→  const handleOtpChange = useCallback(
   273→    (index: number, value: string) => {
   274→      const digit = value.replace(/\D/g, '');
   275→
   276→      if (digit.length > 1) {
   277→        // Handle paste: fill as many slots as possible
   278→        const digits = digit.slice(0, 4);
   279→        const newOtp = [...otp];
   280→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   281→          newOtp[index + i] = digits[i];
   282→        }
   283→        setOtp(newOtp);
   284→        setError('');
   285→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   286→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   287→        return;
   288→      }
   289→
   290→      const newOtp = [...otp];
   291→      newOtp[index] = digit;
   292→      setOtp(newOtp);
   293→      setError('');
   294→
   295→      // Auto-advance to next input
   296→      if (digit && index < 3) {
   297→        inputRefs.current[index + 1]?.focus();
   298→      }
   299→    },
   300→    [otp],
   301→  );
   302→
   303→  const handleKeyDown = useCallback(
   304→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   305→      if (e.key === 'Backspace') {
   306→        if (!otp[index] && index > 0) {
   307→          // If current is empty, go back to previous
   308→          const newOtp = [...otp];
   309→          newOtp[index - 1] = '';
   310→          setOtp(newOtp);
   311→          inputRefs.current[index - 1]?.focus();
   312→        } else {
   313→          // Clear current
   314→          const newOtp = [...otp];
   315→          newOtp[index] = '';
   316→          setOtp(newOtp);
   317→        }
   318→      }
   319→    },
   320→    [otp],
   321→  );
   322→
   323→  const handlePaste = useCallback(
   324→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   325→      e.preventDefault();
   326→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   327→      if (!pasted) return;
   328→
   329→      const newOtp = [...otp];
   330→      for (let i = 0; i < pasted.length && i < 4; i++) {
   331→        newOtp[i] = pasted[i];
   332→      }
   333→      setOtp(newOtp);
   334→      setError('');
   335→
   336→      const focusIdx = Math.min(pasted.length, 3);
   337→      inputRefs.current[focusIdx]?.focus();
   338→    },
   339→    [otp],
   340→  );
   341→
   342→  const handleVerify = useCallback(() => {
   343→    const enteredOtp = otp.join('');
   344→    if (enteredOtp.length !== 4) {
   345→      setError('Please enter all 4 digits');
   346→      return;
   347→    }
   348→
   349→    setStep('otp_verifying');
   350→    setError('');
   351→
   352→    // Simulate verification (demo: accept 1234)
   353→    setTimeout(() => {
   354→      if (enteredOtp === DEMO_OTP) {
   355→        setStep('success');
   356→        // Call onStart after a brief delay for the success animation
   357→        setTimeout(() => {
   358→          onStart(shift.id);
   359→        }, 1500);
   360→      } else {
   361→        setError('Invalid OTP. Please try again.');
   362→        setStep('otp_enter');
   363→        setOtp(['', '', '', '']);
   364→        inputRefs.current[0]?.focus();
   365→      }
   366→    }, 1800);
   367→  }, [otp, shift.id, onStart]);
   368→
   369→  const handleResend = useCallback(() => {
   370→    if (resendTimer > 0) return;
   371→    setResendTimer(RESEND_COOLDOWN);
   372→    setOtp(['', '', '', '']);
   373→    setError('');
   374→    inputRefs.current[0]?.focus();
   375→  }, [resendTimer]);
   376→
   377→  const handleDialogClose = useCallback(
   378→    (open: boolean) => {
   379→      if (!open) onCancel();
   380→    },
   381→    [onCancel],
   382→  );
   383→
   384→  // ── Sub-renderers ──────────────────────────────────────────────────
   385→
   386→  // Shift summary card
   387→  const renderShiftSummary = () => (
   388→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   389→      <div className="flex items-center gap-2 text-muted-foreground">
   390→        <Badge variant="outline" className="text-xs font-medium">
   391→          {shift.serviceType}
   392→        </Badge>
   393→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   394→      </div>
   395→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   396→        <div className="flex items-center gap-2">
   397→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   398→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   399→        </div>
   400→        {shift.address && (
   401→          <div className="flex items-center gap-2">
   402→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   403→            <span className="truncate">{shift.address}</span>
   404→          </div>
   405→        )}
   406→        <div className="flex items-center gap-2">
   407→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   408→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   409→        </div>
   410→      </div>
   411→    </div>
   412→  );
   413→
   414→  // Progress stepper — order: Selfie → Location → OTP
   415→  const renderStepper = (currentStep: number) => (
   416→    <div className="flex items-center justify-center gap-1 mb-2">
   417→      {[
   418→        { n: 1, label: 'Selfie' },
   419→        { n: 2, label: 'Location' },
   420→        { n: 3, label: 'OTP' },
   421→      ].map((s, i) => (
   422→        <div key={s.n} className="flex items-center gap-1">
   423→          <div
   424→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   425→              currentStep >= s.n
   426→                ? 'bg-blue-600 text-white'
   427→                : 'bg-gray-100 text-gray-400'
   428→            }`}
   429→            title={s.label}
   430→          >
   431→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   432→          </div>
   433→          {i < 2 && (
   434→            <div
   435→              className={`w-8 h-0.5 ${
   436→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   437→              }`}
   438→            />
   439→          )}
   440→        </div>
   441→      ))}
   442→    </div>
   443→  );
   444→
   445→  // OTP input fields
   446→  const renderOtpInputs = () => (
   447→    <div className="flex justify-center gap-3">
   448→      {otp.map((digit, i) => (
   449→        <Input
   450→          key={i}
   451→          ref={(el) => { inputRefs.current[i] = el; }}
   452→          type="text"
   453→          inputMode="numeric"
   454→          maxLength={4}
   455→          value={digit}
   456→          onChange={(e) => handleOtpChange(i, e.target.value)}
   457→          onKeyDown={(e) => handleKeyDown(i, e)}
   458→          onPaste={handlePaste}
   459→          disabled={step === 'otp_verifying'}
   460→          className={
   461→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   462→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   463→          }
   464→          aria-label={`OTP digit ${i + 1}`}
   465→        />
   466→      ))}
   467→    </div>
   468→  );
   469→
   470→  // Hidden canvas for selfie capture
   471→  const renderHiddenCanvas = () => (
   472→    <canvas ref={canvasRef} className="hidden" />
   473→  );
   474→
   475→  // ── RENDER: Intro ──────────────────────────────────────────────────
   476→  if (step === 'intro') {
   477→    return (
   478→      <Dialog open onOpenChange={handleDialogClose}>
   479→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   480→          <div className="flex flex-col min-h-full">
   481→            <DialogHeader className="text-center sm:text-center">
   482→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   483→                <ShieldCheck className="size-6 text-blue-600" />
   484→              </div>
   485→              <DialogTitle className="text-lg">Shift Check-In</DialogTitle>
   486→              <DialogDescription className="text-xs">
   487→                Daily attendance is mandatory before starting any shift.
   488→              </DialogDescription>
   489→            </DialogHeader>
   490→
   491→            <div className="flex-1 space-y-3">
   492→              {renderStepper(0)}
   493→
   494→              {/* 3-step overview */}
   495→              <div className="space-y-2">
   496→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   497→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">1</div>
   498→                  <div>
   499→                    <p className="text-xs font-medium text-blue-700">Live Selfie</p>
   500→                    <p className="text-[11px] text-blue-600">Liveness detection for identity</p>
   501→                  </div>
   502→                </div>
   503→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   504→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">2</div>
   505→                  <div>
   506→                    <p className="text-xs font-medium text-blue-700">Location Verification</p>
   507→                    <p className="text-[11px] text-blue-600">High accuracy GPS check</p>
   508→                  </div>
   509→                </div>
   510→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   511→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">3</div>
   512→                  <div>
   513→                    <p className="text-xs font-medium text-blue-700">OTP Confirmation</p>
   514→                    <p className="text-[11px] text-blue-600">4-digit OTP with {familyLabel.toLowerCase()}</p>
   515→                  </div>
   516→                </div>
   517→              </div>
   518→
   519→              {renderShiftSummary()}
   520→
   521→              {/* Demo mode toggle */}
   522→              <div className="flex items-center gap-2 justify-center">
   523→                <button
   524→                  onClick={() => setUseDemoMode(!useDemoMode)}
   525→                  className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-blue-600 transition-colors"
   526→                >
   527→                  <Eye className="size-3.5" />
   528→                  {useDemoMode ? 'Demo Mode ON' : 'Enable Demo Mode'}
   529→                </button>
   530→                {useDemoMode && (
   531→                  <Badge variant="outline" className="text-[10px] bg-amber-50 border-amber-200 text-amber-700">
   532→                    DEMO
   533→                  </Badge>
   534→                )}
   535→              </div>
   536→            </div>
   537→
   538→            <DialogFooter className="mt-auto pt-4 border-t border-gray-100 flex-col gap-2 sm:flex-col sm:space-x-0">
   539→              <Button
   540→                onClick={startSelfieCapture}
   541→                className="w-full bg-blue-600 hover:bg-blue-700 text-white"
   542→              >
   543→                <Camera className="size-4 mr-2" />
   544→                Start Check-In
   545→              </Button>
   546→              <a
   547→                href={NURTUREE_HELPLINE}
   548→                className="inline-flex items-center justify-center gap-2 rounded-md border border-emerald-200 bg-emerald-50 text-emerald-700 px-4 h-9 text-sm font-medium hover:bg-emerald-100 transition-colors w-full"
   549→              >
   550→                <Phone className="size-4" />
   551→                Call Nurturee Helpline
   552→              </a>
   553→              <Button variant="ghost" onClick={onCancel} className="w-full">
   554→                Cancel
   555→              </Button>
   556→            </DialogFooter>
   557→          </div>
   558→        </DialogContent>
   559→      </Dialog>
   560→    );
   561→  }
   562→
   563→  // ── RENDER: Selfie Capture ────────────────────────────────────────
   564→  if (step === 'selfie') {
   565→    return (
   566→      <Dialog open onOpenChange={handleDialogClose}>
   567→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   568→          <div className="flex flex-col min-h-full">
   569→            <DialogHeader className="text-center sm:text-center">
   570→              {renderStepper(1)}
   571→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   572→                <Camera className="size-6 text-blue-600" />
   573→              </div>
   574→              <DialogTitle className="text-lg">Capture Selfie</DialogTitle>
   575→              <DialogDescription className="text-xs">
   576→                Take a live selfie for identity verification.
   577→              </DialogDescription>
   578→            </DialogHeader>
