    36→// ── Skin-tone face detection (pure canvas, works on all browsers) ────────
    37→function detectSkinInOval(
    38→  video: HTMLVideoElement,
    39→  canvas: HTMLCanvasElement,
    40→): boolean {
    41→  const vw = video.videoWidth || 480;
    42→  const vh = video.videoHeight || 640;
    43→  canvas.width = vw;
    44→  canvas.height = vh;
    45→  const ctx = canvas.getContext('2d');
    46→  if (!ctx) return false;
    47→
    48→  ctx.drawImage(video, 0, 0, vw, vh);
    49→
    50→  // Sample the center oval region (face area)
    51→  const cx = vw / 2;
    52→  const cy = vh / 2;
    53→  const rx = vw * 0.2;  // oval horizontal radius
    54→  const ry = vh * 0.28; // oval vertical radius
    55→  const step = 8;        // sample every 8px for speed
    56→
    57→  let skinPixels = 0;
    58→  let totalPixels = 0;
    59→
    60→  for (let y = cy - ry; y <= cy + ry; y += step) {
    61→    for (let x = cx - rx; x <= cx + rx; x += step) {
    62→      // Check if inside ellipse: ((x-cx)/rx)^2 + ((y-cy)/ry)^2 <= 1
    63→      const dx = (x - cx) / rx;
    64→      const dy = (y - cy) / ry;
    65→      if (dx * dx + dy * dy > 1) continue;
    66→
    67→      totalPixels++;
    68→      const pixel = ctx.getImageData(Math.floor(x), Math.floor(y), 1, 1).data;
    69→      const r = pixel[0], g = pixel[1], b = pixel[2];
    70→
    71→      // Skin-tone detection in RGB space (covers wide range of skin tones)
    72→      // Rules: R > G > B, R between 60-255, G between 30-220, B between 15-180
    73→      // And logical skin-tone constraints based on chrominance
    74→      if (
    75→        r > 60 && g > 30 && b > 15 &&
    76→        r > g && r > b &&
    77→        (r - g) > 10 &&
    78→        (r - b) > 15 &&
    79→        Math.abs(r - g) < 100 &&
    80→        b < r * 0.75 &&
    81→        g < r * 0.9
    82→      ) {
    83→        skinPixels++;
    84→      }
    85→    }
    86→  }
    87→
    88→  // Need at least 30% skin-tone pixels in the oval to consider face detected
    89→  return totalPixels > 0 && (skinPixels / totalPixels) > 0.30;
    90→}

   116→// ── Component ────────────────────────────────────────────────────────
   117→export default function ShiftCheckIn({
   118→  shift,
   119→  onStart,
   120→  onCancel,
   121→}: ShiftCheckInProps) {
   122→  const [step, setStep] = useState<CheckInStep>('intro');
   123→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
   124→  const [otp, setOtp] = useState(['', '', '', '']);
   125→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
   126→  const [error, setError] = useState('');
   127→  const [useDemoMode, setUseDemoMode] = useState(false);
   128→  const [faceDetected, setFaceDetected] = useState(false);
   129→
   130→  const videoRef = useRef<HTMLVideoElement | null>(null);
   131→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   132→  const detectionCanvasRef = useRef<HTMLCanvasElement | null>(null);
   133→  const streamRef = useRef<MediaStream | null>(null);
   134→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   135→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   136→  const faceDetectRef = useRef<ReturnType<typeof setInterval> | null>(null);
   137→  const autoCaptureRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   138→
   139→  const familyLabel = getFamilyLabel(shift.serviceCategory);
   140→
   141→  // ── Cleanup camera stream on unmount ──────────
   142→  useEffect(() => {
   143→    return () => {
   144→      if (streamRef.current) {
   145→        streamRef.current.getTracks().forEach((t) => t.stop());
   146→      }
   147→      if (timerRef.current) clearInterval(timerRef.current);
   148→      if (faceDetectRef.current) clearInterval(faceDetectRef.current);
   149→      if (autoCaptureRef.current) clearTimeout(autoCaptureRef.current);
   150→    };
   151→  }, []);
   152→
   153→  // ── Auto-start location check when step reaches location_check ──
   154→  useEffect(() => {
   155→    if (step === 'location_check') {
   156→      startLocationCheck();
   157→    }
   158→  }, [step]); // intentionally only depend on step to trigger once per entry
   159→
   160→  // ── Resend countdown ───────────────────────────────────────────────
   161→  useEffect(() => {
   162→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   163→
   164→    timerRef.current = setInterval(() => {
   165→      setResendTimer((prev) => {
   166→        if (prev <= 1) {
   167→          if (timerRef.current) clearInterval(timerRef.current);
   168→          return 0;
   169→        }
   170→        return prev - 1;
   171→      });
   172→    }, 1000);
   173→
   174→    return () => {
   175→      if (timerRef.current) clearInterval(timerRef.current);
   176→    };
   177→  }, [step, resendTimer]);
   178→
   179→  // ── Reset OTP inputs ──────────────────────────────────────────────
   180→  useEffect(() => {
   181→    if (step === 'otp_enter') {
   182→      setOtp(['', '', '', '']);
   183→      setError('');
   184→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   185→    }
   186→  }, [step]);
   187→
   188→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   189→  const startSelfieCapture = useCallback(async () => {
   190→    setStep('selfie');
   191→    setFaceDetected(false);
   192→    setError('');
   193→
   194→    try {
   195→      const stream = await navigator.mediaDevices.getUserMedia({
   196→        video: { facingMode: 'user', width: 480, height: 640 },
   197→        audio: false,
   198→      });
   199→      streamRef.current = stream;
   200→
   201→      if (videoRef.current) {
   202→        const video = videoRef.current;
   203→        video.srcObject = stream;
   204→        await video.play();
   205→
   206→        // Start face detection loop (~5 fps) once video is playing
   207→        const detCanvas = detectionCanvasRef.current || document.createElement('canvas');
   208→        detectionCanvasRef.current = detCanvas;
   209→
   210→        if (faceDetectRef.current) clearInterval(faceDetectRef.current);
   211→        faceDetectRef.current = setInterval(() => {
   212→          if (!videoRef.current || videoRef.current.readyState < 2) return;
   213→          const detected = detectSkinInOval(videoRef.current, detCanvas);
   214→          setFaceDetected(detected);
   215→
   216→          // Auto-capture after face is stable in oval for 1.5s
   217→          if (detected && !autoCaptureRef.current) {
   218→            autoCaptureRef.current = setTimeout(() => {
   219→              autoCaptureRef.current = null;
   220→              captureSelfie();
   221→            }, 1500);
   222→          } else if (!detected && autoCaptureRef.current) {
   223→            clearTimeout(autoCaptureRef.current);
   224→            autoCaptureRef.current = null;
   225→          }
   226→        }, 200);
   227→      }
   228→    } catch {
   229→      if (useDemoMode) {
   230→        setStep('location_check');
   231→      } else {
   232→        setError('Camera access denied. Please allow camera access.');
   233→      }
   234→    }
   235→  }, [useDemoMode]);
   236→
   237→  const stopFaceDetection = useCallback(() => {
   238→    if (faceDetectRef.current) {
   239→      clearInterval(faceDetectRef.current);
   240→      faceDetectRef.current = null;
   241→    }
   242→    if (autoCaptureRef.current) {
   243→      clearTimeout(autoCaptureRef.current);
   244→      autoCaptureRef.current = null;
   245→    }
   246→    setFaceDetected(false);
   247→  }, []);
   248→
   249→  const captureSelfie = useCallback(async () => {
   250→    if (!videoRef.current || !canvasRef.current) return;
   251→
   252→    const video = videoRef.current;
   253→    const canvas = canvasRef.current;
   254→    canvas.width = video.videoWidth;
   255→    canvas.height = video.videoHeight;
   256→
   257→    const ctx = canvas.getContext('2d');
   258→    if (!ctx) return;
   259→
   260→    // Mirror the image (selfie style)
   261→    ctx.translate(canvas.width, 0);
   262→    ctx.scale(-1, 1);
   263→    ctx.drawImage(video, 0, 0);
   264→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   265→
   266→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   267→    setSelfieDataUrl(dataUrl);
   268→
   269→    // Stop camera stream + face detection
   270→    stopFaceDetection();
   271→    if (streamRef.current) {
   272→      streamRef.current.getTracks().forEach((t) => t.stop());
   273→      streamRef.current = null;
   274→    }
   275→
   276→    // Show preview for user to accept or retake
   277→    setStep('selfie_preview');
   278→  }, [stopFaceDetection]);
   279→
   280→  const skipSelfie = useCallback(() => {
   281→    stopFaceDetection();
   282→    if (streamRef.current) {
   283→      streamRef.current.getTracks().forEach((t) => t.stop());
   284→      streamRef.current = null;
   285→    }
   286→    setStep('location_check');
   287→  }, [stopFaceDetection]);
   288→
   289→  const retakeSelfie = useCallback(() => {
   290→    setSelfieDataUrl(null);
   291→    startSelfieCapture();
   292→  }, [startSelfieCapture]);
   293→
   294→  // Accept selfie and proceed to liveness check (backend-ready)
   295→  const acceptSelfie = useCallback(() => {
   296→    setStep('selfie_processing');
   297→    // TODO: Send selfieDataUrl to backend for liveness/face verification
   298→    // For now, simulate processing delay
   299→    setTimeout(() => {
   300→      setStep('location_check');
   301→    }, 1500);
   302→  }, []);
   303→
   304→  // ── Step 2: Location Verification (High Accuracy GPS) ───────────────
   305→  const startLocationCheck = useCallback(async () => {
   306→    setStep('location_check');
   307→    setError('');
   308→
   309→    try {
   310→      if (useDemoMode) {
   311→        // Demo mode — simulate GPS fetch
   312→        await new Promise((r) => setTimeout(r, 2000));
   313→        setStep('location_passed');
   314→        return;
   315→      }
   316→
   317→      // Real high-accuracy GPS check
   318→      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
   319→        navigator.geolocation.getCurrentPosition(resolve, reject, {
   320→          enableHighAccuracy: true,
   321→          timeout: 15000,
   322→          maximumAge: 0,
   323→        });
   324→      });
   325→
   326→      const accuracy = position.coords.accuracy;
   327→      if (accuracy <= 100) {
   328→        setStep('location_passed');
   329→      } else {
   330→        setError(`GPS accuracy too low (${Math.round(accuracy)}m). Need within 100m.`);
   331→        setStep('location_failed');
   332→      }
   333→    } catch (err) {
   334→      setError('Location check failed. Please enable location services and try again.');
   335→      setStep('location_failed');
   336→    }
   337→  }, [useDemoMode]);
   338→
   339→  // ── Step 3: OTP Verification ──────────────────────────────────────
   340→  const handleRequestOTP = useCallback(() => {
   341→    setResendTimer(RESEND_COOLDOWN);
   342→    setStep('otp_enter');
   343→  }, []);
   344→
   345→  const handleOtpChange = useCallback(
   346→    (index: number, value: string) => {
   347→      const digit = value.replace(/\D/g, '');
   348→
   349→      if (digit.length > 1) {
   350→        // Handle paste: fill as many slots as possible
   351→        const digits = digit.slice(0, 4);
   352→        const newOtp = [...otp];
   353→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   354→          newOtp[index + i] = digits[i];
   355→        }
   356→        setOtp(newOtp);
   357→        setError('');
   358→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   359→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   360→        return;
   361→      }
   362→
   363→      const newOtp = [...otp];
   364→      newOtp[index] = digit;
   365→      setOtp(newOtp);
   366→      setError('');
   367→
   368→      // Auto-advance to next input
   369→      if (digit && index < 3) {
   370→        inputRefs.current[index + 1]?.focus();
   371→      }
   372→    },
   373→    [otp],
   374→  );
   375→
   376→  const handleKeyDown = useCallback(
   377→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   378→      if (e.key === 'Backspace') {
   379→        if (!otp[index] && index > 0) {
   380→          // If current is empty, go back to previous
   381→          const newOtp = [...otp];
   382→          newOtp[index - 1] = '';
   383→          setOtp(newOtp);
   384→          inputRefs.current[index - 1]?.focus();
   385→        } else {
   386→          // Clear current
   387→          const newOtp = [...otp];
   388→          newOtp[index] = '';
   389→          setOtp(newOtp);
   390→        }
   391→      }
   392→    },
   393→    [otp],
   394→  );
   395→
   396→  const handlePaste = useCallback(
   397→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   398→      e.preventDefault();
   399→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   400→      if (!pasted) return;
   401→
   402→      const newOtp = [...otp];
   403→      for (let i = 0; i < pasted.length && i < 4; i++) {
   404→        newOtp[i] = pasted[i];
   405→      }
   406→      setOtp(newOtp);
   407→      setError('');
   408→
   409→      const focusIdx = Math.min(pasted.length, 3);
   410→      inputRefs.current[focusIdx]?.focus();
   411→    },
   412→    [otp],
   413→  );
   414→
   415→  const handleVerify = useCallback(() => {
   416→    const enteredOtp = otp.join('');
   417→    if (enteredOtp.length !== 4) {
   418→      setError('Please enter all 4 digits');
   419→      return;
   420→    }
   421→
   422→    setStep('otp_verifying');
   423→    setError('');
   424→
   425→    // Simulate verification (demo: accept 1234)
   426→    setTimeout(() => {
   427→      if (enteredOtp === DEMO_OTP) {
   428→        setStep('success');
   429→        // Call onStart after a brief delay for the success animation
   430→        setTimeout(() => {
   431→          onStart(shift.id);
   432→        }, 1500);
   433→      } else {
   434→        setError('Invalid OTP. Please try again.');
   435→        setStep('otp_enter');
   436→        setOtp(['', '', '', '']);
   437→        inputRefs.current[0]?.focus();
   438→      }
   439→    }, 1800);
   440→  }, [otp, shift.id, onStart]);
   441→
   442→  const handleResend = useCallback(() => {
   443→    if (resendTimer > 0) return;
   444→    setResendTimer(RESEND_COOLDOWN);
   445→    setOtp(['', '', '', '']);
   446→    setError('');
   447→    inputRefs.current[0]?.focus();
   448→  }, [resendTimer]);
   449→
   450→  const handleDialogClose = useCallback(
   451→    (open: boolean) => {
   452→      if (!open) onCancel();
   453→    },
   454→    [onCancel],
   455→  );
   456→
   457→  // ── Sub-renderers ──────────────────────────────────────────────────
   458→
   459→  // Shift summary card
   460→  const renderShiftSummary = () => (
   461→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   462→      <div className="flex items-center gap-2 text-muted-foreground">
   463→        <Badge variant="outline" className="text-xs font-medium">
   464→          {shift.serviceType}
   465→        </Badge>
   466→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   467→      </div>
   468→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   469→        <div className="flex items-center gap-2">
   470→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   471→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   472→        </div>
   473→        {shift.address && (
   474→          <div className="flex items-center gap-2">
   475→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   476→            <span className="truncate">{shift.address}</span>
   477→          </div>
   478→        )}
   479→        <div className="flex items-center gap-2">
   480→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   481→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   482→        </div>
   483→      </div>
   484→    </div>
   485→  );
   486→
   487→  // Progress stepper — order: Selfie → Location → OTP
   488→  const renderStepper = (currentStep: number) => (
   489→    <div className="flex items-center justify-center gap-1 mb-2">
   490→      {[
   491→        { n: 1, label: 'Selfie' },
   492→        { n: 2, label: 'Location' },
   493→        { n: 3, label: 'OTP' },
   494→      ].map((s, i) => (
   495→        <div key={s.n} className="flex items-center gap-1">
   496→          <div
   497→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   498→              currentStep >= s.n
   499→                ? 'bg-blue-600 text-white'
   500→                : 'bg-gray-100 text-gray-400'
   501→            }`}
   502→            title={s.label}
   503→          >
   504→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   505→          </div>
   506→          {i < 2 && (
   507→            <div
   508→              className={`w-8 h-0.5 ${
   509→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   510→              }`}
   511→            />
   512→          )}
   513→        </div>
   514→      ))}
   515→    </div>
   516→  );
   517→
   518→  // OTP input fields
   519→  const renderOtpInputs = () => (
   520→    <div className="flex justify-center gap-3">
   521→      {otp.map((digit, i) => (
   522→        <Input
   523→          key={i}
   524→          ref={(el) => { inputRefs.current[i] = el; }}
   525→          type="text"
   526→          inputMode="numeric"
   527→          maxLength={1}
   528→          value={digit}
   529→          onChange={(e) => handleOtpChange(i, e.target.value)}
   530→          onKeyDown={(e) => handleKeyDown(i, e)}
   531→          onPaste={handlePaste}
   532→          disabled={step === 'otp_verifying'}
   533→          className={
   534→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   535→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   536→          }
   537→          aria-label={`OTP digit ${i + 1}`}
   538→        />
   539→      ))}
   540→    </div>
   541→  );
   542→
   543→  // Hidden canvas for selfie capture
   544→  const renderHiddenCanvas = () => (
   545→    <canvas ref={canvasRef} className="hidden" />
   546→  );
   547→
   548→  // ── RENDER: Intro ──────────────────────────────────────────────────
   549→  if (step === 'intro') {
   550→    return (
   551→      <Dialog open onOpenChange={handleDialogClose}>
   552→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   553→          <div className="flex flex-col min-h-full">
   554→            <DialogHeader className="text-center sm:text-center">
   555→              <div className="mx-auto mb-1 flex size-12 items-center justify-center rounded-full bg-blue-100">
   556→                <ShieldCheck className="size-6 text-blue-600" />
   557→              </div>
   558→              <DialogTitle className="text-lg">Shift Check-In</DialogTitle>
   559→              <DialogDescription className="text-xs">
   560→                Daily attendance is mandatory before starting any shift.
   561→              </DialogDescription>
   562→            </DialogHeader>
   563→
   564→            <div className="flex-1 space-y-3">
   565→              {renderStepper(0)}
   566→
   567→              {/* 3-step overview */}
   568→              <div className="space-y-2">
   569→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   570→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">1</div>
   571→                  <div>
   572→                    <p className="text-xs font-medium text-blue-700">Live Selfie</p>
   573→                    <p className="text-[11px] text-blue-600">Liveness detection for identity</p>
   574→                  </div>
   575→                </div>
   576→                <div className="flex items-center gap-2.5 px-3 py-2.5 rounded-lg border bg-blue-50">
   577→                  <div className="flex size-7 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0">2</div>
   578→                  <div>
   579→                    <p className="text-xs font-medium text-blue-700">Location Verification</p>
