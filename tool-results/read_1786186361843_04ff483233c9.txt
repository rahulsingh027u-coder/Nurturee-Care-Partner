     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react';
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→  DialogDescription,
    10→  DialogFooter,
    11→} from '@/components/ui/dialog';
    12→import { Input } from '@/components/ui/input';
    13→import { Button } from '@/components/ui/button';
    14→import { Badge } from '@/components/ui/badge';
    15→import {
    16→  MapPin,
    17→  ShieldCheck,
    18→  Loader2,
    19→  AlertCircle,
    20→  CheckCircle,
    21→  Camera,

    23→  ArrowLeft,
    24→  Copy,
    25→  IndianRupee,
    26→  Fingerprint,
    27→  Eye,
    28→  XCircle,
    29→  RotateCcw,
    30→} from 'lucide-react';
    31→
    32→import type { Shift } from '@/lib/caregiver-data';
    33→import {
    34→  runAntiSpoofCheck,
    35→  runDemoAntiSpoofCheck,
    36→  generateDeviceId,
    37→} from '@/lib/attendance';
    38→import type { AntiSpoofResult } from '@/lib/attendance';
    39→
    40→// ── Types ────────────────────────────────────────────────────────────
    41→type CheckInStep =
    42→  | 'intro'
    43→  | 'selfie'
    44→  | 'selfie_processing'
    45→  | 'location_check'
    46→  | 'location_passed'
    47→  | 'location_failed'
    48→  | 'otp_request'
    49→  | 'otp_enter'
    50→  | 'otp_verifying'
    51→  | 'success'
    52→  | 'failed';
    53→
    54→interface ShiftCheckInProps {
    55→  shift: Shift;
    56→  onStart: (shiftId: string) => void;
    57→  onCancel: () => void;
    58→}
    59→
    60→// ── Helpers ──────────────────────────────────────────────────────────
    61→const DEMO_OTP = '1234';
    62→const RESEND_COOLDOWN = 30;
    63→
    64→function formatTime(time24: string): string {
    65→  const [hours, minutes] = time24.split(':').map(Number);
    66→  const period = hours >= 12 ? 'PM' : 'AM';
    67→  const hours12 = hours % 12 || 12;
    68→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
    69→}
    70→
    71→function getFamilyLabel(category: string): string {
    72→  return category === 'child_care'
    73→    ? "Child's Parent"
    74→    : "Elder's Family Member";
    75→}
    76→
    77→// ── Component ────────────────────────────────────────────────────────
    78→export default function ShiftCheckIn({
    79→  shift,
    80→  onStart,
    81→  onCancel,
    82→}: ShiftCheckInProps) {
    83→  const [step, setStep] = useState<CheckInStep>('intro');
    84→  const [antiSpoofResult, setAntiSpoofResult] = useState<AntiSpoofResult | null>(null);
    85→  const [selfieDataUrl, setSelfieDataUrl] = useState<string | null>(null);
    86→  const [otp, setOtp] = useState(['', '', '', '']);
    87→  const [resendTimer, setResendTimer] = useState(RESEND_COOLDOWN);
    88→  const [error, setError] = useState('');
    89→  const [copiedOtp, setCopiedOtp] = useState(false);
    90→  const [deviceId, setDeviceId] = useState('');
    91→  const [useDemoMode, setUseDemoMode] = useState(false);
    92→
    93→  const videoRef = useRef<HTMLVideoElement | null>(null);
    94→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    95→  const streamRef = useRef<MediaStream | null>(null);
    96→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
    97→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    98→
    99→  const familyLabel = getFamilyLabel(shift.serviceCategory);
   100→
   101→  // ── Generate device ID on mount ──────────────────────────────────
   102→  useEffect(() => {
   103→    setDeviceId(generateDeviceId());
   104→  }, []);
   105→
   106→  // ── Cleanup camera stream on unmount ─────────────────────────────
   107→  useEffect(() => {
   108→    return () => {
   109→      if (streamRef.current) {
   110→        streamRef.current.getTracks().forEach((t) => t.stop());
   111→      }
   112→      if (timerRef.current) clearInterval(timerRef.current);
   113→    };
   114→  }, []);
   115→
   116→  // ── Resend countdown ───────────────────────────────────────────────
   117→  useEffect(() => {
   118→    if (step !== 'otp_enter' || resendTimer <= 0) return;
   119→
   120→    timerRef.current = setInterval(() => {
   121→      setResendTimer((prev) => {
   122→        if (prev <= 1) {
   123→          if (timerRef.current) clearInterval(timerRef.current);
   124→          return 0;
   125→        }
   126→        return prev - 1;
   127→      });
   128→    }, 1000);
   129→
   130→    return () => {
   131→      if (timerRef.current) clearInterval(timerRef.current);
   132→    };
   133→  }, [step, resendTimer]);
   134→
   135→  // ── Reset OTP inputs ──────────────────────────────────────────────
   136→  useEffect(() => {
   137→    if (step === 'otp_enter') {
   138→      setOtp(['', '', '', '']);
   139→      setError('');
   140→      setTimeout(() => inputRefs.current[0]?.focus(), 100);
   141→    }
   142→  }, [step]);
   143→
   144→  // ── Step 1: Selfie Capture ─────────────────────────────────────────
   145→  const startSelfieCapture = useCallback(async () => {
   146→    setStep('selfie');
   147→    setError('');
   148→
   149→    try {
   150→      const stream = await navigator.mediaDevices.getUserMedia({
   151→        video: { facingMode: 'user', width: 480, height: 640 },
   152→        audio: false,
   153→      });
   154→      streamRef.current = stream;
   155→
   156→      if (videoRef.current) {
   157→        videoRef.current.srcObject = stream;
   158→        await videoRef.current.play();
   159→      }
   160→    } catch {
   161→      // Camera not available — allow skip in demo
   162→      if (useDemoMode) {
   163→        setStep('location_check');
   164→      } else {
   165→        setError('Camera access denied. Please allow camera access.');
   166→      }
   167→    }
   168→  }, [useDemoMode]);
   169→
   170→  const captureSelfie = useCallback(() => {
   171→    if (!videoRef.current || !canvasRef.current) return;
   172→
   173→    const video = videoRef.current;
   174→    const canvas = canvasRef.current;
   175→    canvas.width = video.videoWidth;
   176→    canvas.height = video.videoHeight;
   177→
   178→    const ctx = canvas.getContext('2d');
   179→    if (!ctx) return;
   180→
   181→    // Mirror the image (selfie style)
   182→    ctx.translate(canvas.width, 0);
   183→    ctx.scale(-1, 1);
   184→    ctx.drawImage(video, 0, 0);
   185→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   186→
   187→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   188→    setSelfieDataUrl(dataUrl);
   189→
   190→    // Stop camera stream
   191→    if (streamRef.current) {
   192→      streamRef.current.getTracks().forEach((t) => t.stop());
   193→      streamRef.current = null;
   194→    }
   195→
   196→    setStep('selfie_processing');
   197→    // Simulate liveness check processing
   198→    setTimeout(() => {
   199→      setStep('location_check');
   200→    }, 1500);
   201→  }, []);
   202→
   203→  const skipSelfie = useCallback(() => {
   204→    if (streamRef.current) {
   205→      streamRef.current.getTracks().forEach((t) => t.stop());
   206→      streamRef.current = null;
   207→    }
   208→    setStep('location_check');
   209→  }, []);
   210→
   211→  const retakeSelfie = useCallback(() => {
   212→    setSelfieDataUrl(null);
   213→    startSelfieCapture();
   214→  }, [startSelfieCapture]);
   215→
   216→  // ── Step 2: Location Verification ──────────────────────────────────
   217→  const startLocationCheck = useCallback(async () => {
   218→    setStep('location_check');
   219→    setError('');
   220→
   221→    try {
   222→      if (useDemoMode) {
   223→        // Demo mode — bypass real checks
   224→        await new Promise((r) => setTimeout(r, 2500));
   225→        const demoResult = runDemoAntiSpoofCheck();
   226→        setAntiSpoofResult(demoResult);
   227→        setStep('location_passed');
   228→        return;
   229→      }
   230→
   231→      const { result } = await runAntiSpoofCheck();
   232→      setAntiSpoofResult(result);
   233→
   234→      if (result.passed) {
   235→        setStep('location_passed');
   236→      } else {
   237→        setStep('location_failed');
   238→      }
   239→    } catch (err) {
   240→      setError('Location check failed. ' + (err instanceof Error ? err.message : 'Unknown error'));
   241→      setStep('location_failed');
   242→    }
   243→  }, [useDemoMode]);
   244→
   245→  // ── Step 3: OTP Verification ──────────────────────────────────────
   246→  const handleRequestOTP = useCallback(() => {
   247→    setResendTimer(RESEND_COOLDOWN);
   248→    setStep('otp_enter');
   249→  }, []);
   250→
   251→  const handleOtpChange = useCallback(
   252→    (index: number, value: string) => {
   253→      const digit = value.replace(/\D/g, '');
   254→
   255→      if (digit.length > 1) {
   256→        // Handle paste: fill as many slots as possible
   257→        const digits = digit.slice(0, 4);
   258→        const newOtp = [...otp];
   259→        for (let i = 0; i < digits.length && index + i < 4; i++) {
   260→          newOtp[index + i] = digits[i];
   261→        }
   262→        setOtp(newOtp);
   263→        setError('');
   264→        const nextEmpty = newOtp.findIndex((v, i) => i > index && v === '');
   265→        inputRefs.current[nextEmpty !== -1 ? nextEmpty : 3]?.focus();
   266→        return;
   267→      }
   268→
   269→      const newOtp = [...otp];
   270→      newOtp[index] = digit;
   271→      setOtp(newOtp);
   272→      setError('');
   273→
   274→      // Auto-advance to next input
   275→      if (digit && index < 3) {
   276→        inputRefs.current[index + 1]?.focus();
   277→      }
   278→    },
   279→    [otp],
   280→  );
   281→
   282→  const handleKeyDown = useCallback(
   283→    (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   284→      if (e.key === 'Backspace') {
   285→        if (!otp[index] && index > 0) {
   286→          // If current is empty, go back to previous
   287→          const newOtp = [...otp];
   288→          newOtp[index - 1] = '';
   289→          setOtp(newOtp);
   290→          inputRefs.current[index - 1]?.focus();
   291→        } else {
   292→          // Clear current
   293→          const newOtp = [...otp];
   294→          newOtp[index] = '';
   295→          setOtp(newOtp);
   296→        }
   297→      }
   298→    },
   299→    [otp],
   300→  );
   301→
   302→  const handlePaste = useCallback(
   303→    (e: React.ClipboardEvent<HTMLInputElement>) => {
   304→      e.preventDefault();
   305→      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 4);
   306→      if (!pasted) return;
   307→
   308→      const newOtp = [...otp];
   309→      for (let i = 0; i < pasted.length && i < 4; i++) {
   310→        newOtp[i] = pasted[i];
   311→      }
   312→      setOtp(newOtp);
   313→      setError('');
   314→
   315→      const focusIdx = Math.min(pasted.length, 3);
   316→      inputRefs.current[focusIdx]?.focus();
   317→    },
   318→    [otp],
   319→  );
   320→
   321→  const handleVerify = useCallback(() => {
   322→    const enteredOtp = otp.join('');
   323→    if (enteredOtp.length !== 4) {
   324→      setError('Please enter all 4 digits');
   325→      return;
   326→    }
   327→
   328→    setStep('otp_verifying');
   329→    setError('');
   330→
   331→    // Simulate verification (demo: accept 1234)
   332→    setTimeout(() => {
   333→      if (enteredOtp === DEMO_OTP) {
   334→        setStep('success');
   335→        // Call onStart after a brief delay for the success animation
   336→        setTimeout(() => {
   337→          onStart(shift.id);
   338→        }, 1500);
   339→      } else {
   340→        setError('Invalid OTP. Please try again.');
   341→        setStep('otp_enter');
   342→        setOtp(['', '', '', '']);
   343→        inputRefs.current[0]?.focus();
   344→      }
   345→    }, 1800);
   346→  }, [otp, shift.id, onStart]);
   347→
   348→  const handleResend = useCallback(() => {
   349→    if (resendTimer > 0) return;
   350→    setResendTimer(RESEND_COOLDOWN);
   351→    setOtp(['', '', '', '']);
   352→    setError('');
   353→    inputRefs.current[0]?.focus();
   354→  }, [resendTimer]);
   355→
   356→  const handleCopyOtp = useCallback(() => {
   357→    navigator.clipboard.writeText(DEMO_OTP).then(() => {
   358→      setCopiedOtp(true);
   359→      setTimeout(() => setCopiedOtp(false), 2000);
   360→    });
   361→  }, []);
   362→
   363→  const handleDialogClose = useCallback(
   364→    (open: boolean) => {
   365→      if (!open) onCancel();
   366→    },
   367→    [onCancel],
   368→  );
   369→
   370→  // ── Sub-renderers ──────────────────────────────────────────────────
   371→
   372→  // Shift summary card
   373→  const renderShiftSummary = () => (
   374→    <div className="rounded-lg border bg-muted/40 p-3 space-y-2 text-sm">
   375→      <div className="flex items-center gap-2 text-muted-foreground">
   376→        <Badge variant="outline" className="text-xs font-medium">
   377→          {shift.serviceType}
   378→        </Badge>
   379→        <span className="font-medium text-foreground">{shift.bookingId}</span>
   380→      </div>
   381→      <div className="grid grid-cols-1 gap-1.5 mt-2">
   382→        <div className="flex items-center gap-2">
   383→          <Clock className="size-3.5 text-muted-foreground shrink-0" />
   384→          <span>{formatTime(shift.startTime)} – {formatTime(shift.endTime)}</span>
   385→        </div>
   386→        {shift.address && (
   387→          <div className="flex items-center gap-2">
   388→            <MapPin className="size-3.5 text-muted-foreground shrink-0" />
   389→            <span className="truncate">{shift.address}</span>
   390→          </div>
   391→        )}
   392→        <div className="flex items-center gap-2">
   393→          <IndianRupee className="size-3.5 text-muted-foreground shrink-0" />
   394→          <span className="font-semibold text-emerald-600">₹{shift.earnings.toLocaleString('en-IN')}</span>
   395→        </div>
   396→      </div>
   397→    </div>
   398→  );
   399→
   400→  // Progress stepper — order: Selfie → Location → OTP
   401→  const renderStepper = (currentStep: number) => (
   402→    <div className="flex items-center justify-center gap-1 mb-2">
   403→      {[
   404→        { n: 1, label: 'Selfie' },
   405→        { n: 2, label: 'Location' },
   406→        { n: 3, label: 'OTP' },
   407→      ].map((s, i) => (
   408→        <div key={s.n} className="flex items-center gap-1">
   409→          <div
   410→            className={`flex items-center justify-center rounded-full size-7 text-xs font-bold ${
   411→              currentStep >= s.n
   412→                ? 'bg-blue-600 text-white'
   413→                : 'bg-gray-100 text-gray-400'
   414→            }`}
   415→            title={s.label}
   416→          >
   417→            {currentStep > s.n ? <CheckCircle className="size-3.5" /> : s.n}
   418→          </div>
   419→          {i < 2 && (
   420→            <div
   421→              className={`w-8 h-0.5 ${
   422→                currentStep > s.n ? 'bg-blue-600' : 'bg-gray-200'
   423→              }`}
   424→            />
   425→          )}
   426→        </div>
   427→      ))}
   428→    </div>
   429→  );
   430→
   431→  // Anti-spoof results display
   432→  const renderAntiSpoofResults = (result: AntiSpoofResult) => (
   433→    <div className="space-y-2">
   434→      {/* Overall score bar */}
   435→      <div className="flex items-center justify-between">
   436→        <span className="text-sm font-medium">Verification Score</span>
   437→        <div className="flex items-center gap-2">
   438→          <div className="w-24 h-2 bg-gray-200 rounded-full overflow-hidden">
   439→            <div
   440→              className={`h-full rounded-full transition-all ${
   441→                result.score >= 80 ? 'bg-emerald-500' :
   442→                result.score >= 50 ? 'bg-amber-500' : 'bg-red-500'
   443→              }`}
   444→              style={{ width: `${result.score}%` }}
   445→            />
   446→          </div>
   447→          <span className={`text-sm font-bold ${
   448→            result.score >= 80 ? 'text-emerald-600' :
   449→            result.score >= 50 ? 'text-amber-600' : 'text-red-600'
   450→          }`}>
   451→            {result.score}/100
   452→          </span>
   453→        </div>
   454→      </div>
   455→
   456→      {/* Layer results */}
   457→      <div className="grid grid-cols-1 gap-1.5">
   458→        {result.layerResults.map((layer) => (
   459→          <div
   460→            key={layer.layer}
   461→            className={`flex items-center gap-2 px-2.5 py-1.5 rounded-lg text-xs ${
   462→              layer.passed ? 'bg-emerald-50' : 'bg-red-50'
   463→            }`}
   464→          >
   465→            {layer.passed ? (
   466→              <CheckCircle className="size-3.5 text-emerald-600 shrink-0" />
   467→            ) : (
   468→              <XCircle className="size-3.5 text-red-500 shrink-0" />
   469→            )}
   470→            <div className="flex-1 min-w-0">
   471→              <span className={`font-medium ${
   472→                layer.passed ? 'text-emerald-700' : 'text-red-700'
   473→              }`}>
   474→                {layer.layer}
   475→              </span>
   476→              <span className="ml-1.5 text-muted-foreground">{layer.message}</span>
   477→            </div>
   478→          </div>
   479→        ))}
   480→      </div>
   481→
   482→      {/* Risk badge */}
   483→      <div className="flex items-center justify-center">
   484→        <Badge
   485→          className={`text-xs ${
   486→            result.riskLevel === 'low'
   487→              ? 'bg-emerald-100 text-emerald-700 border-emerald-200'
   488→              : result.riskLevel === 'medium'
   489→              ? 'bg-amber-100 text-amber-700 border-amber-200'
   490→              : 'bg-red-100 text-red-700 border-red-200'
   491→          }`}
   492→        >
   493→          <ShieldCheck className="size-3 mr-1" />
   494→          Risk Level: {result.riskLevel.toUpperCase()}
   495→        </Badge>
   496→      </div>
   497→    </div>
   498→  );
   499→
   500→  // OTP input fields
   501→  const renderOtpInputs = () => (
   502→    <div className="flex justify-center gap-3">
   503→      {otp.map((digit, i) => (
   504→        <Input
   505→          key={i}
   506→          ref={(el) => { inputRefs.current[i] = el; }}
   507→          type="text"
   508→          inputMode="numeric"
   509→          maxLength={4}
   510→          value={digit}
   511→          onChange={(e) => handleOtpChange(i, e.target.value)}
   512→          onKeyDown={(e) => handleKeyDown(i, e)}
   513→          onPaste={handlePaste}
   514→          disabled={step === 'otp_verifying'}
   515→          className={
   516→            'h-14 w-14 text-center text-2xl font-bold tracking-widest tabular-nums focus-visible:ring-blue-500 ' +
   517→            (error ? 'border-destructive focus-visible:ring-destructive' : '')
   518→          }
   519→          aria-label={`OTP digit ${i + 1}`}
   520→        />
   521→      ))}
   522→    </div>
   523→  );
   524→
   525→  // Hidden canvas for selfie capture
   526→  const renderHiddenCanvas = () => (
   527→    <canvas ref={canvasRef} className="hidden" />
   528→  );
   529→
   530→  // ── RENDER: Intro ──────────────────────────────────────────────────
   531→  if (step === 'intro') {
   532→    return (
   533→      <Dialog open onOpenChange={handleDialogClose}>
   534→        <DialogContent className="sm:max-w-md" showCloseButton={false}>
   535→          <DialogHeader className="text-center sm:text-center">
   536→            <div className="mx-auto mb-2 flex size-14 items-center justify-center rounded-full bg-blue-100">
   537→              <ShieldCheck className="size-7 text-blue-600" />
   538→            </div>
   539→            <DialogTitle className="text-xl">Shift Check-In</DialogTitle>
   540→            <DialogDescription>
   541→              Daily attendance is mandatory before starting any shift. This involves:
   542→            </DialogDescription>
   543→          </DialogHeader>
   544→
   545→          <div className="space-y-4">
   546→            {renderStepper(0)}
   547→
   548→            {/* 3-step overview — Selfie → Location → OTP */}
   549→            <div className="space-y-3">
   550→              <div className="flex items-start gap-3 p-3 rounded-lg border bg-blue-50">
   551→                <div className="flex size-8 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0 mt-0.5">1</div>
   552→                <div>
   553→                  <p className="text-sm font-medium text-blue-700">Live Selfie</p>
   554→                  <p className="text-xs text-blue-600">Liveness detection ensures real person present</p>
   555→                </div>
   556→              </div>
   557→              <div className="flex items-start gap-3 p-3 rounded-lg border bg-blue-50">
   558→                <div className="flex size-8 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0 mt-0.5">2</div>
   559→                <div>
   560→                  <p className="text-sm font-medium text-blue-700">Location Verification</p>
   561→                  <p className="text-xs text-blue-600">8-layer anti-spoofing check — must be at care location</p>
   562→                </div>
   563→              </div>
   564→              <div className="flex items-start gap-3 p-3 rounded-lg border bg-blue-50">
   565→                <div className="flex size-8 items-center justify-center rounded-full bg-blue-600 text-white text-xs font-bold shrink-0 mt-0.5">3</div>
   566→                <div>
   567→                  <p className="text-sm font-medium text-blue-700">OTP Confirmation</p>
   568→                  <p className="text-xs text-blue-600">4-digit OTP verified with {familyLabel.toLowerCase()}</p>
   569→                </div>
   570→              </div>
   571→            </div>
   572→
   573→            {renderShiftSummary()}
   574→
   575→            {/* Demo mode toggle */}
   576→            <div className="flex items-center gap-2 justify-center">
