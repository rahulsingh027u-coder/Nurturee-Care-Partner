    91→function formatCurrency(amount: number): string {
    92→  return new Intl.NumberFormat("en-IN", {
    93→    style: "currency",
    94→    currency: "INR",
    95→    minimumFractionDigits: 0,
    96→    maximumFractionDigits: 0,
    97→  }).format(amount);
    98→}
    99→
   100→function getGreeting(): string {
   101→  const hour = new Date().getHours();
   102→  if (hour < 12) return "Good Morning";
   103→  if (hour < 17) return "Good Afternoon";
   104→  return "Good Evening";
   105→}
   106→
   107→function formatTodayDate(): string {
   108→  return new Date().toLocaleDateString("en-IN", {
   109→    weekday: "long",
   110→    year: "numeric",
   111→    month: "long",
   112→    day: "numeric",
   113→  });
   114→}
   115→
   116→function getStatusBadgeClasses(status: string): string {
   117→  switch (status) {
   118→    case "confirmed":
   119→      return "bg-blue-100 text-blue-700 border-blue-200";
   120→    case "in-progress":
   121→      return "bg-blue-100 text-blue-800 border-blue-200";
   122→    case "completed":
   123→      return "bg-emerald-100 text-emerald-700 border-emerald-200";
   124→    case "cancelled":
   125→    case "no-show":
   126→      return "bg-red-100 text-red-700 border-red-200";
   127→    case "pending":
   128→    default:
   129→      return "bg-amber-100 text-amber-700 border-amber-200";
   130→  }
   131→}
   132→
   133→function getStatusLabel(status: string): string {
   134→  switch (status) {
   135→    case "in-progress":
   136→      return "In Progress";
   137→    case "no-show":
   138→      return "No Show";
   139→    default:
   140→      return status.charAt(0).toUpperCase() + status.slice(1);
   141→  }
   142→}
