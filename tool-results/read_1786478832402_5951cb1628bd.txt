   182→function loadDocsFromStorage(): VerificationDoc[] {
   183→  try {
   184→    // Version check — if storage format is outdated, clear and start fresh
   185→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   186→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   187→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   188→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   189→      return getInitialDocs();
   190→    }
   191→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   192→    if (stored) {
   193→      const parsed = JSON.parse(stored);
   194→      if (Array.isArray(parsed) && parsed.length === 4) {
   195→        // Validate structure — reject data from old format
   196→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   197→        if (isValid) return parsed;
   198→      }
   199→    }
   200→  } catch {
   201→    // fall through
   202→  }
   203→  return getInitialDocs();
   204→}
   205→
   206→function saveDocsToStorage(docs: VerificationDoc[]) {
   207→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   208→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   209→}

   228→// ─── Optional Deadline Helpers ─────────────────────────────────────────────────
   229→
   230→const OPTIONAL_DEADLINE_DAYS = 15;
   231→const REGISTERED_AT_KEY = 'nurturee_care_partner_registered_at';
   232→
   233→function getDaysRemaining(): number | null {
   234→  try {
   235→    const registeredAt = localStorage.getItem(REGISTERED_AT_KEY);
   236→    if (!registeredAt) return null;
   237→    const registered = new Date(registeredAt).getTime();
   238→    const deadline = registered + OPTIONAL_DEADLINE_DAYS * 24 * 60 * 60 * 1000;
   239→    const remaining = Math.ceil((deadline - Date.now()) / (24 * 60 * 60 * 1000));
   240→    return Math.max(0, remaining);
   241→  } catch {
   242→    return null;
   243→  }
   244→}

   248→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   249→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   250→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   251→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   252→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   253→  const [uploadedFront, setUploadedFront] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   254→  const [uploadedBack, setUploadedBack] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   255→  const [submitting, setSubmitting] = useState(false);
   256→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   257→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   258→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   259→  const tempCalendarDateRef = useRef<Record<string, Date | undefined>>({});
   260→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   261→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   262→
   263→  const currentYear = new Date().getFullYear();
   264→
   265→  // Clean up timers on unmount
   266→  useEffect(() => {
   267→    return () => {
   268→      Object.values(reviewTimersRef.current).forEach((timers) => {
   269→        timers.forEach(clearTimeout);
   270→      });
   271→    };
   272→  }, []);
   273→
   274→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   275→  useEffect(() => {
   276→    if (submittedType) {
   277→      const doc = docs.find((d) => d.type === submittedType);
   278→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   279→        setSubmittedType(null);
   280→        setReviewingType(null);
   281→      }
   282→    }
   283→    if (reviewingType) {
   284→      const doc = docs.find((d) => d.type === reviewingType);
   285→      if (!doc || doc.status !== 'under_review') {
   286→        setReviewingType(null);
   287→      }
   288→    }
   289→  }, [docs, submittedType, reviewingType]);
   290→
   291→  // Persist docs to localStorage whenever they change
   292→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   293→    setDocs((prev) => {
   294→      const updated = updater(prev);
   295→      saveDocsToStorage(updated);
   296→      return updated;
   297→    });
   298→  }, []);
   299→
   300→  // ─── Backend Review Simulation ───────────────────────────────────────────
   301→
   302→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   303→    // Clear any existing timers for this type
   304→    if (reviewTimersRef.current[type]) {
   305→      reviewTimersRef.current[type].forEach(clearTimeout);
   306→    }
   307→    reviewTimersRef.current[type] = [];
   308→
   309→    const doc = docs.find((d) => d.type === type);
   310→    if (!doc) return;
   311→
   312→    // Timer 1: After 3s → under_review
   313→    const t1 = setTimeout(() => {
   314→      setReviewingType(type);
   315→      updateDocs((prev) =>
   316→        prev.map((d) =>
   317→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   318→        )
   319→      );
   320→    }, 3000);
   321→
   322→    // Timer 2: After 5-7s total → resolve
   323→    const resolveDelay = 5000 + Math.random() * 2000;
   324→    const t2 = setTimeout(() => {
   325→      const currentDoc = docs.find((d) => d.type === type);
   326→      if (!currentDoc) return;
   327→
   328→      let willApprove: boolean;
   329→
   330→      if (type === 'aadhaar') {
   331→        // Aadhaar always approve on first attempt (demo-friendly)
   332→        willApprove = true;
   333→      } else {
   334→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   335→        if (attemptNumber === 1) {
   336→          willApprove = Math.random() < 0.4;
   337→        } else if (attemptNumber === 2) {
   338→          willApprove = Math.random() < 0.7;
   339→        } else {
   340→          willApprove = true;
   341→        }
   342→      }
   343→
   344→      if (willApprove) {
   345→        // Approve
   346→        setApprovingType(type);
   347→        updateDocs((prev) =>
   348→          prev.map((d) => {
   349→            if (d.type !== type) return d;
   350→            return {
   351→              ...d,
   352→              status: 'approved' as VerificationStatus,
   353→              attempts: d.attempts.map((a) =>
   354→                a.attemptNumber === attemptNumber
   355→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   356→                  : a
   357→              ),
   358→            };
   359→          })
   360→        );
   361→        setReviewingType(null);
   362→
   363→        // If Aadhaar approved and onComplete exists, call it after delay
   364→        if (type === 'aadhaar' && onComplete) {
   365→          setTimeout(() => {
   366→            setApprovingType(null);
   367→            setSubmittedType(null);
   368→            onComplete();
   369→          }, 1000);
   370→        } else {
   371→          // For optional: show success briefly, then allow going back
   372→          setTimeout(() => {
   373→            setApprovingType(null);
   374→            setSubmittedType(null);
   375→            setActiveType(null);
   376→          }, 2000);
   377→        }
   378→      } else {
   379→        // Reject
   380→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   381→        updateDocs((prev) =>
   382→          prev.map((d) => {
   383→            if (d.type !== type) return d;
   384→            const updatedAttempts = d.attempts.map((a) =>
   385→              a.attemptNumber === attemptNumber
   386→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   387→                : a
   388→            );
   389→            // Check if max attempts reached
   390→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   391→            return {
   392→              ...d,
   393→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   394→              attempts: updatedAttempts,
   395→              currentDraft: { documents: [], formFields: {} },
   396→            };
   397→          })
   398→        );
   399→        setReviewingType(null);
   400→        // Transition from submitted view to allow re-upload
   401→        setTimeout(() => {
   402→          setSubmittedType(null);
   403→        }, 300);
   404→      }
   405→    }, resolveDelay);
   406→
   407→    reviewTimersRef.current[type] = [t1, t2];
   408→  }, [docs, onComplete, updateDocs]);
   409→
   410→  // ─── Handlers ────────────────────────────────────────────────────────────
   411→
   412→  const handleSelectType = (type: VerificationType) => {
   413→    const doc = docs.find((d) => d.type === type);
   414→    if (!doc) return;
   415→    setActiveType(type);
   416→    // Restore draft for editable docs; approved/locked docs show read-only
   417→    const isFrozenDoc = doc.status === 'max_attempts_reached' || doc.status === 'approved';
   418→    if (!isFrozenDoc) {
   419→      setFormFields(doc.currentDraft?.formFields || {});
   420→      if (type === 'aadhaar') {
   421→        const draftDocs = doc.currentDraft?.documents || [];
   422→        setUploadedFront(draftDocs.filter(f => f.id.startsWith('front-')));
   423→        setUploadedBack(draftDocs.filter(f => f.id.startsWith('back-')));
   424→        setUploadedFiles([]);
   425→      } else {
   426→        setUploadedFiles(doc.currentDraft?.documents || []);
   427→        setUploadedFront([]);
   428→        setUploadedBack([]);
   429→      }
   430→    } else {
   431→      setFormFields({});
   432→      setUploadedFiles([]);
   433→      setUploadedFront([]);
   434→      setUploadedBack([]);
   435→    }
   436→    setSubmittedType(null);
   437→  };
   438→
   439→  const handleBack = () => {
   440→    // Save draft
   441→    if (activeType) {
   442→      // For Aadhaar, combine front + back into documents array
   443→      const isAadhaar = activeType === 'aadhaar';
   444→      const combinedDocs = isAadhaar
   445→        ? [...uploadedFront.map(f => ({ ...f, id: `front-${f.id}` })), ...uploadedBack.map(f => ({ ...f, id: `back-${f.id}` }))]
   446→        : uploadedFiles;
   447→      updateDocs((prev) =>
   448→        prev.map((d) =>
   449→          d.type === activeType
   450→            ? { ...d, currentDraft: { documents: combinedDocs, formFields } }
   451→            : d
   452→        )
   453→      );
   454→    }
   455→    // Clean up timers for this type if we navigate back
   456→    if (activeType && reviewTimersRef.current[activeType]) {
   457→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   458→      if (!submittedType && !reviewingType) {
   459→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   460→        delete reviewTimersRef.current[activeType];
   461→      }
   462→    }
   463→    setActiveType(null);
   464→    setFormFields({});
   465→    setUploadedFiles([]);
   466→    setUploadedFront([]);
   467→    setUploadedBack([]);
   468→    setSubmittedType(null);
   469→    setReviewingType(null);
   470→  };
   471→
   472→  const validateAndCreateFileEntries = (files: FileList, config: typeof verificationConfig[VerificationType] | null) => {
   473→    const result: Array<{ id: string; name: string; size: number; type: string }> = [];
   474→    for (let i = 0; i < files.length; i++) {
   475→      const file = files[i];
   476→      if (config && !config.allowedTypes.includes(file.type)) {
   477→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   478→        continue;
   479→      }
   480→      if (config && file.size > config.maxFileSize) {
   481→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   482→        continue;
   483→      }
   484→      result.push({ id: `file-${Date.now()}-${i}`, name: file.name, size: file.size, type: file.type });
   485→    }
   486→    return result;
   487→  };
   488→
   489→  // Generic file change (for non-Aadhaar docs)
   490→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   491→    const files = e.target.files;
   492→    if (!files || files.length === 0) return;
   493→    const config = activeType ? verificationConfig[activeType] : null;
   494→    const newFiles = validateAndCreateFileEntries(files, config);
   495→    if (newFiles.length > 0) {
   496→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   497→    }
   498→    e.target.value = '';
   499→  };
   500→
   501→  // Aadhaar-specific: handle front/back file uploads
   502→  const handleAadhaarSideChange = (side: 'front' | 'back') => (e: React.ChangeEvent<HTMLInputElement>) => {
   503→    const files = e.target.files;
   504→    if (!files || files.length === 0) return;
   505→    const config = verificationConfig['aadhaar'];
   506→    const newFiles = validateAndCreateFileEntries(files, config);
   507→    const setter = side === 'front' ? setUploadedFront : setUploadedBack;
   508→    if (newFiles.length > 0) {
   509→      setter(newFiles); // Replace — only one file per side
   510→    }
   511→    e.target.value = '';
   512→  };
   513→
   514→  const handleRemoveFile = (fileId: string) => {
   515→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   516→  };
   517→  const handleRemoveAadhaarSide = (side: 'front' | 'back') => () => {
   518→    if (side === 'front') setUploadedFront([]);
   519→    else setUploadedBack([]);
   520→  };
   521→
   522→  const handleSubmit = async () => {
   523→    if (!activeType) return;
   524→    setSubmitting(true);
   525→
   526→    // Simulate submission delay
   527→    await new Promise((resolve) => setTimeout(resolve, 1500));
