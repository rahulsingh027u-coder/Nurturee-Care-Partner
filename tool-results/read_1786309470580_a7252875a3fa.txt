    94→  active,
    95→  payload,
    96→  label,
    97→}: {
    98→  active?: boolean;
    99→  payload?: { value: number }[];
   100→  label?: string;
   101→}) => {
   102→  if (active && payload && payload.length) {
   103→    return (
   104→      <div className="rounded-lg border border-gray-100 bg-white px-3 py-2 shadow-md">
   105→        <p className="text-[11px] font-medium text-gray-500">{label}</p>

   107→          {formatCurrency(payload[0].value)}
   108→        </p>
   109→      </div>
   110→    );
   111→  }
   112→  return null;
   113→};

   115→// ─── Component ────────────────────────────────────────────────────────────────
   116→
   117→export default function EarningsView() {
   118→  const [activeCard, setActiveCard] = useState<Period>('thisMonth');
   119→  const [chartPeriod, setChartPeriod] = useState<ChartPeriod>('monthly');
   120→  const [selectedTransaction, setSelectedTransaction] = useState<string | null>(null);
   121→  const [payoutRequested, setPayoutRequested] = useState(false);
   122→  const [downloading, setDownloading] = useState(false);
   123→  const [liveOffset, setLiveOffset] = useState(0);
   124→  const [animatingCard, setAnimatingCard] = useState<string | null>(null);
   125→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all');
   126→  const [activeTab, setActiveTab] = useState<string>('overview');
   127→
   128→  // Live earnings simulation — small random increments every 8s
   129→  useEffect(() => {
   130→    const interval = setInterval(() => {
   131→      setLiveOffset((prev) => prev + Math.floor(Math.random() * 200) + 50);
   132→    }, 8000);
   133→    return () => clearInterval(interval);
   134→  }, []);
   135→
   136→  // Computed summary cards
   137→  const summaryCards: SummaryCard[] = [
   138→    {
   139→      key: 'today',
   140→      label: 'Today',
   141→      value: mockCaregiver.earnings.today + (activeCard === 'today' ? liveOffset : 0),
   142→      sublabel: 'Earned today',
   143→      icon: Wallet,
   144→      color: 'bg-blue-50 text-blue-600',
   145→      activeColor: 'bg-blue-100 text-blue-700',
   146→      activeBorder: 'border-blue-300 ring-1 ring-blue-100',
   147→    },
   148→    {
   149→      key: 'thisWeek',
   150→      label: 'This Week',
   151→      value: mockCaregiver.earnings.thisWeek + (activeCard === 'thisWeek' ? liveOffset : 0),
   152→      sublabel: 'Mon – Sun',
   153→      icon: TrendingUp,
   154→      color: 'bg-emerald-50 text-emerald-600',
   155→      activeColor: 'bg-emerald-100 text-emerald-700',
   156→      activeBorder: 'border-emerald-300 ring-1 ring-emerald-100',
   157→    },
   158→    {
   159→      key: 'thisMonth',
   160→      label: 'This Month',
   161→      value: mockCaregiver.earnings.thisMonth + (activeCard === 'thisMonth' ? liveOffset : 0),
   162→      sublabel: new Date().toLocaleDateString('en-IN', { month: 'long', year: 'numeric' }),
   163→      icon: IndianRupee,
   164→      color: 'bg-violet-50 text-violet-600',
   165→      activeColor: 'bg-violet-100 text-violet-700',
   166→      activeBorder: 'border-violet-300 ring-1 ring-violet-100',
   167→    },
   168→    {
   169→      key: 'total',
   170→      label: 'Total',
   171→      value: mockCaregiver.earnings.total + (activeCard === 'total' ? liveOffset : 0),
   172→      sublabel: 'All time earnings',
   173→      icon: Clock,
   174→      color: 'bg-amber-50 text-amber-600',
   175→      activeColor: 'bg-amber-100 text-amber-700',
   176→      activeBorder: 'border-amber-300 ring-1 ring-amber-100',
   177→    },
   178→  ];
   179→
   180→  const activeSummaryCard = summaryCards.find((c) => c.key === activeCard)!;
   181→
   182→  // Chart data with live updates blended in
   183→  const chartData = chartPeriod === 'monthly'
   184→    ? mockMonthlyEarnings.map((d) => ({ ...d, earnings: d.earnings + Math.floor(Math.random() * 500) }))
   185→    : mockWeeklyEarnings.map((d) => ({ ...d, earnings: d.earnings + Math.floor(Math.random() * 100) }));
   186→
   187→  const chartTotal = chartData.reduce((sum, d) => sum + d.earnings, 0);
   188→  const chartAvg = Math.round(chartTotal / chartData.length);
   189→  const chartMax = Math.max(...chartData.map((d) => d.earnings));
   190→
   191→  // Status breakdown
   192→  const paidCount = mockEarnings.filter((e) => e.status === 'paid').length;
   193→  const pendingCount = mockEarnings.filter((e) => e.status === 'pending').length;
   194→  const processingCount = mockEarnings.filter((e) => e.status === 'processing').length;
   195→  const pieData = [
   196→    { name: 'Paid', value: paidCount },
   197→    { name: 'Pending', value: pendingCount },
   198→    { name: 'Processing', value: processingCount },
   199→  ].filter((d) => d.value > 0);
   200→
   201→  // Filtered earnings for selected period AND status
   202→  const today = new Date();
   203→  const filteredEarnings = mockEarnings.filter((e) => {
   204→    const eDate = new Date(e.date);
   205→    // Period filter
   206→    switch (activeCard) {
   207→      case 'today':
   208→        if (eDate.toDateString() !== today.toDateString()) return false;
   209→        break;
   210→      case 'thisWeek': {
   211→        const startOfWeek = new Date(today);
   212→        startOfWeek.setDate(today.getDate() - today.getDay());
   213→        if (eDate < startOfWeek) return false;
   214→        break;
   215→      }
   216→      case 'thisMonth':
   217→        if (eDate.getMonth() !== today.getMonth() || eDate.getFullYear() !== today.getFullYear()) return false;
   218→        break;
   219→      case 'total':
   220→      default:
   221→        break;
   222→    }
   223→    // Status filter
   224→    if (statusFilter !== 'all' && e.status !== statusFilter) return false;
   225→    return true;
   226→  });
   227→
   228→  const handleCardClick = (key: Period) => {
   229→    setAnimatingCard(key);
   230→    setActiveCard(key);
   231→    setLiveOffset(0);
   232→    setTimeout(() => setAnimatingCard(null), 400);
   233→  };
   234→
   235→  const handleRequestPayout = () => {
   236→    setPayoutRequested(true);
   237→    setTimeout(() => setPayoutRequested(false), 3000);
   238→  };
   239→
   240→  const handleDownload = () => {
   241→    setDownloading(true);
   242→    setTimeout(() => setDownloading(false), 2000);
   243→  };
   244→
   245→  const handleRowClick = (id: string) => {
   246→    setSelectedTransaction(selectedTransaction === id ? null : id);
   247→  };
   248→
   249→  // ─── Custom Pie Label ────────────────────────────────────────────────────
   250→
   251→  const renderCustomLabel = ({
   252→    cx,
   253→    cy,
   254→    midAngle,
   255→    innerRadius,
   256→    outerRadius,
   257→    percent,
   258→    name,
   259→  }: {
   260→    cx: number;
   261→    cy: number;
   262→    midAngle: number;
   263→    innerRadius: number;
   264→    outerRadius: number;
   265→    percent: number;
   266→    name: string;
   267→  }) => {
   268→    const RADIAN = Math.PI / 180;
   269→    const radius = innerRadius + (outerRadius - innerRadius) * 1.4;
   270→    const x = cx + radius * Math.cos(-midAngle * RADIAN);
   271→    const y = cy + radius * Math.sin(-midAngle * RADIAN);
   272→
   273→    return (
   274→      <text x={x} y={y} textAnchor={x > cx ? 'start' : 'end'} className="text-[11px] fill-gray-500 font-medium">
   275→        {name} ({(percent * 100).toFixed(0)}%)
   276→      </text>
   277→    );
   278→  };
   279→
   280→  return (
   281→    <div className="space-y-6">
   282→      {/* ===== Section 1: Summary Cards (Clickable) ===== */}
   283→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   284→        {summaryCards.map((card) => {
   285→          const Icon = card.icon;
   286→          const isActive = activeCard === card.key;
   287→          const isAnimating = animatingCard === card.key;
   288→          return (
   289→            <button
   290→              key={card.key}
   291→              onClick={() => handleCardClick(card.key)}
   292→              className={`bg-card rounded-xl border shadow-sm py-5 text-left transition-all duration-300 hover:shadow-md cursor-pointer w-full ${
   293→                isActive
   294→                  ? `${card.activeBorder} ${isAnimating ? 'scale-[1.03]' : 'scale-[1.01]'}`
   295→                  : 'border-gray-100 hover:border-gray-200'
   296→              }`}
   297→            >
   298→              <div className="flex items-center gap-3 mb-3 px-4">
   299→                <div
   300→                  className={`size-10 rounded-xl flex items-center justify-center transition-colors ${
   301→                    isActive ? card.activeColor : card.color
   302→                  }`}
   303→                >
   304→                  <Icon className="size-5" />
   305→                </div>
   306→                {isActive && (
   307→                  <div className="ml-auto">
   308→                    <div className="size-2 rounded-full bg-blue-500 animate-pulse" />
   309→                  </div>
   310→                )}
   311→              </div>
   312→              <div className="px-4">
   313→                <p className={`text-xs font-medium transition-colors ${isActive ? 'text-gray-700' : 'text-gray-500'}`}>
   314→                  {card.label}
   315→                </p>
   316→                <p className={`text-xl font-bold sm:text-2xl transition-colors ${isActive ? 'text-gray-900' : 'text-gray-900'}`}>
   317→                  {formatCurrency(card.value)}
   318→                </p>
   319→                <p className="text-[11px] text-gray-400 mt-0.5">{card.sublabel}</p>
   320→              </div>
   321→            </button>
   322→          );
   323→        })}
   324→      </div>
   325→
   326→      {/* Active card info bar */}
   327→      <div className="flex items-center gap-2 px-1">
   328→        <div className="size-2 rounded-full bg-blue-500" />
   329→        <span className="text-xs text-gray-500">
   330→          Viewing: <span className="font-semibold text-gray-700">{activeSummaryCard.label}</span> earnings
   331→          {filteredEarnings.length > 0 && (
   332→            <span className="text-gray-400"> — {filteredEarnings.length} transaction{filteredEarnings.length > 1 ? 's' : ''}</span>
   333→          )}
   334→        </span>
   335→      </div>
   336→
   337→      {/* ===== Section 2: Tabs ===== */}
   338→      <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   339→        <TabsList className="mb-4">
   340→          <TabsTrigger value="overview">Overview</TabsTrigger>
   341→          <TabsTrigger value="breakdown">Breakdown</TabsTrigger>
   342→          <TabsTrigger value="history">Payment History</TabsTrigger>
   343→        </TabsList>
   344→
   345→        {/* ——— Tab 1: Overview ——— */}
   346→        <TabsContent value="overview" className="space-y-4">
   347→          <div className="grid lg:grid-cols-3 gap-4">
   348→            {/* Chart Card with Monthly/Weekly toggle */}
   349→            <Card className="lg:col-span-2">
   350→              <CardHeader className="pb-2">
   351→                <div className="flex items-center justify-between">
   352→                  <CardTitle className="text-sm font-semibold text-gray-900">
   353→                    Earnings Overview
   354→                  </CardTitle>
   355→                  <div className="flex items-center gap-1 bg-gray-100 rounded-lg p-0.5">
   356→                    <button
   357→                      onClick={() => setChartPeriod('monthly')}
   358→                      className={`px-3 py-1 text-xs font-medium rounded-md transition-all ${
   359→                        chartPeriod === 'monthly'
   360→                          ? 'bg-white text-gray-900 shadow-sm'
   361→                          : 'text-gray-500 hover:text-gray-700'
   362→                      }`}
   363→                    >
   364→                      Monthly
   365→                    </button>
   366→                    <button
   367→                      onClick={() => setChartPeriod('weekly')}
   368→                      className={`px-3 py-1 text-xs font-medium rounded-md transition-all ${
   369→                        chartPeriod === 'weekly'
   370→                          ? 'bg-white text-gray-900 shadow-sm'
   371→                          : 'text-gray-500 hover:text-gray-700'
   372→                      }`}
   373→                    >
   374→                      Weekly
   375→                    </button>
   376→                  </div>
   377→                </div>
   378→              </CardHeader>
   379→              <CardContent>
   380→                <div className="h-[240px]">
   381→                  <ResponsiveContainer width="100%" height="100%">
   382→                    {chartPeriod === 'monthly' ? (
   383→                      <AreaChart data={chartData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   384→                        <defs>
   385→                          <linearGradient id="monthlyGradient" x1="0" y1="0" x2="0" y2="1">
   386→                            <stop offset="0%" stopColor="#3b82f6" stopOpacity={0.3} />
   387→                            <stop offset="100%" stopColor="#3b82f6" stopOpacity={0} />
   388→                          </linearGradient>
   389→                        </defs>
   390→                        <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
   391→                        <XAxis dataKey={chartPeriod === 'monthly' ? 'month' : 'day'} axisLine={false} tickLine={false} fontSize={11} tick={{ fill: '#94a3b8', fontSize: 11 }} />
   392→                        <YAxis hide />
   393→                        <Tooltip content={<CustomTooltip />} cursor={{ stroke: '#3b82f6', strokeWidth: 1, strokeDasharray: '4 4' }} />
   394→                        <Area type="monotone" dataKey="earnings" stroke="#3b82f6" strokeWidth={2} fill="url(#monthlyGradient)" />
   395→                      </AreaChart>
   396→                    ) : (
   397→                      <BarChart data={chartData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   398→                        <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
   399→                        <XAxis dataKey="day" axisLine={false} tickLine={false} fontSize={11} tick={{ fill: '#94a3b8', fontSize: 11 }} />
   400→                        <YAxis hide />
   401→                        <Tooltip content={<CustomTooltip />} cursor={{ fill: 'rgba(59,130,246,0.06)' }} />
   402→                        <Bar dataKey="earnings" radius={[6, 6, 0, 0]} maxBarSize={36}>
   403→                          {chartData.map((entry, index) => (
   404→                            <Cell
   405→                              key={index}
   406→                              fill={entry.earnings === 0 ? '#e2e8f0' : '#3b82f6'}
   407→                            />
   408→                          ))}
   409→                        </Bar>
   410→                      </BarChart>
   411→                    )}
   412→                  </ResponsiveContainer>
   413→                </div>
   414→                {/* Chart Stats Row */}
   415→                <div className="grid grid-cols-3 gap-3 mt-4 pt-4 border-t border-gray-100">
   416→                  <div className="text-center">
   417→                    <p className="text-[11px] text-gray-400 font-medium">Total</p>
   418→                    <p className="text-sm font-bold text-gray-900">{formatCurrency(chartTotal)}</p>
   419→                  </div>
   420→                  <div className="text-center">
   421→                    <p className="text-[11px] text-gray-400 font-medium">Average</p>
   422→                    <p className="text-sm font-bold text-gray-900">{formatCurrency(chartAvg)}</p>
   423→                  </div>
   424→                  <div className="text-center">
   425→                    <p className="text-[11px] text-gray-400 font-medium">Peak</p>
   426→                    <p className="text-sm font-bold text-emerald-600">{formatCurrency(chartMax)}</p>
   427→                  </div>
   428→                </div>
   429→              </CardContent>
   430→            </Card>
   431→
   432→            {/* Right Column — Status Pie + Quick Stats */}
   433→            <div className="space-y-4">
   434→              {/* Status Breakdown Pie */}
   435→              <Card>
   436→                <CardHeader className="pb-2">
   437→                  <CardTitle className="text-sm font-semibold text-gray-900">
   438→                    Status Breakdown
   439→                  </CardTitle>
   440→                </CardHeader>
   441→                <CardContent>
   442→                  <div className="h-[160px]">
   443→                    <ResponsiveContainer width="100%" height="100%">
   444→                      <PieChart>
   445→                        <Pie
   446→                          data={pieData}
   447→                          cx="50%"
   448→                          cy="50%"
   449→                          innerRadius={35}
   450→                          outerRadius={55}
   451→                          dataKey="value"
   452→                          labelLine={false}
   453→                          label={renderCustomLabel}
   454→                        >
   455→                          {pieData.map((_, index) => (
   456→                            <Cell key={index} fill={PIE_COLORS[index % PIE_COLORS.length]} />
   457→                          ))}
   458→                        </Pie>
   459→                        <Tooltip
   460→                          formatter={(value: number) => [`${value} transactions`, 'Count']}
   461→                          contentStyle={{
   462→                            borderRadius: '8px',
   463→                            border: '1px solid #f1f5f9',
   464→                            fontSize: '12px',
   465→                          }}
   466→                        />
   467→                      </PieChart>
   468→                    </ResponsiveContainer>
   469→                  </div>
   470→                  {/* Legend */}
   471→                  <div className="flex items-center justify-center gap-4 mt-2">
   472→                    {pieData.map((item, index) => (
   473→                      <button
   474→                        key={item.name}
   475→                        className="flex items-center gap-1.5 text-xs text-gray-600 hover:text-gray-900 transition-colors cursor-pointer"
   476→                      >
   477→                        <div className="size-2.5 rounded-full" style={{ backgroundColor: PIE_COLORS[index] }} />
   478→                        {item.name} ({item.value})
   479→                      </button>
   480→                    ))}
   481→                  </div>
   482→                </CardContent>
   483→              </Card>
   484→
   485→              {/* Quick Stats */}
   486→              <Card>
   487→                <CardContent className="p-4 space-y-3">
   488→                  <div className="flex items-center justify-between">
   489→                    <div className="flex items-center gap-2">
   490→                      <Receipt className="size-4 text-gray-400" />
   491→                      <span className="text-xs text-gray-500">Total Transactions</span>
   492→                    </div>
   493→                    <span className="text-sm font-bold text-gray-900">{mockEarnings.length}</span>
   494→                  </div>
   495→                  <div className="h-px bg-gray-100" />
   496→                  <div className="flex items-center justify-between">
   497→                    <div className="flex items-center gap-2">
   498→                      <TrendingUp className="size-4 text-emerald-500" />
   499→                      <span className="text-xs text-gray-500">Avg. per Transaction</span>
   500→                    </div>
   501→                    <span className="text-sm font-bold text-gray-900">
   502→                      {formatCurrency(Math.round(mockEarnings.reduce((s, e) => s + e.amount, 0) / mockEarnings.length))}
   503→                    </span>
   504→                  </div>
   505→                  <div className="h-px bg-gray-100" />
   506→                  <div className="flex items-center justify-between">
   507→                    <div className="flex items-center gap-2">
   508→                      <Sparkles className="size-4 text-amber-500" />
   509→                      <span className="text-xs text-gray-500">Best Day</span>
   510→                    </div>
   511→                    <span className="text-sm font-bold text-gray-900">{formatCurrency(chartMax)}</span>
   512→                  </div>
   513→                </CardContent>
   514→              </Card>
   515→            </div>
   516→          </div>
   517→
   518→          {/* Payout Summary — Clickable Status Segments */}
   519→          <Card>
   520→            <CardHeader className="pb-3">
   521→              <div className="flex items-center justify-between">
   522→                <CardTitle className="text-sm font-semibold text-gray-900">
   523→                  Payout Summary
   524→                </CardTitle>
   525→                <Badge variant="outline" className="text-[10px] text-gray-500 border-gray-200 bg-gray-50">
   526→                  <Info className="size-3 mr-1" />
   527→                  Click segments for details
   528→                </Badge>
   529→              </div>
   530→            </CardHeader>
   531→            <CardContent className="space-y-3">
   532→              {/* Paid Segment */}
   533→              <button
   534→                onClick={() => { setStatusFilter('paid'); setActiveTab('history'); }}
   535→                className={`w-full flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 rounded-lg border p-4 transition-all cursor-pointer text-left group ${
