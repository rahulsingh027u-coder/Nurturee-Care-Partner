    59→function formatCurrency(amount: number): string {
    60→  return '₹' + amount.toLocaleString('en-IN');
    61→}
    62→
    63→function getGreeting(): string {
    64→  const hour = new Date().getHours();
    65→  if (hour < 12) return 'Good Morning';
    66→  if (hour < 17) return 'Good Afternoon';
    67→  return 'Good Evening';
    68→}
    69→
    70→function formatDate(): string {
    71→  const now = new Date();
    72→  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
    73→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
    74→  return `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
    75→}
