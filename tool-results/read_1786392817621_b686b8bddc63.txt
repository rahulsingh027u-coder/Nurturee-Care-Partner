    93→  active,
    94→  payload,
    95→  label,
    96→}: {
    97→  active?: boolean;
    98→  payload?: { value: number }[];
    99→  label?: string;
   100→}) => {
   101→  if (active && payload && payload.length) {
   102→    return (
   103→      <div className="rounded-lg border border-gray-100 bg-white px-3 py-2 shadow-md">
   104→        <p className="text-[11px] font-medium text-gray-500">{label}</p>
   105→        <p className="text-sm font-bold text-gray-900">
   106→          {formatCurrency(payload[0].value)}
   107→        </p>
   108→      </div>
   109→    );
   110→  }
   111→  return null;
   112→};

   114→// ─── Component ────────────────────────────────────────────────────────────────
   115→
   116→export default function EarningsView() {
   117→  const [activeCard, setActiveCard] = useState<Period>('thisMonth');
   118→  const [chartPeriod, setChartPeriod] = useState<ChartPeriod>('monthly');
   119→  const [selectedTransaction, setSelectedTransaction] = useState<string | null>(null);
   120→  const [payoutRequested, setPayoutRequested] = useState(false);
   121→  const [downloading, setDownloading] = useState(false);
   122→  const [liveOffset, setLiveOffset] = useState(0);
   123→  const [animatingCard, setAnimatingCard] = useState<string | null>(null);
   124→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all');
   125→  const [activeTab, setActiveTab] = useState<string>('overview');
   126→
   127→  // Live earnings simulation — small random increments every 8s
   128→  useEffect(() => {
   129→    const interval = setInterval(() => {
   130→      setLiveOffset((prev) => prev + Math.floor(Math.random() * 200) + 50);
   131→    }, 8000);
   132→    return () => clearInterval(interval);
   133→  }, []);
   134→
   135→  // Computed summary cards
   136→  const summaryCards: SummaryCard[] = [
   137→    {
   138→      key: 'today',
   139→      label: 'Today',
   140→      value: mockCaregiver.earnings.today + (activeCard === 'today' ? liveOffset : 0),
   141→      sublabel: 'Earned today',
   142→      icon: Wallet,
   143→      color: 'bg-blue-50 text-blue-600',
   144→      activeColor: 'bg-blue-100 text-blue-700',
   145→      activeBorder: 'border-blue-300 ring-1 ring-blue-100',
   146→    },
   147→    {
   148→      key: 'thisWeek',
   149→      label: 'This Week',
   150→      value: mockCaregiver.earnings.thisWeek + (activeCard === 'thisWeek' ? liveOffset : 0),
   151→      sublabel: 'Mon – Sun',
   152→      icon: TrendingUp,
   153→      color: 'bg-emerald-50 text-emerald-600',
   154→      activeColor: 'bg-emerald-100 text-emerald-700',
   155→      activeBorder: 'border-emerald-300 ring-1 ring-emerald-100',
   156→    },
   157→    {
   158→      key: 'thisMonth',
   159→      label: 'This Month',
   160→      value: mockCaregiver.earnings.thisMonth + (activeCard === 'thisMonth' ? liveOffset : 0),
   161→      sublabel: new Date().toLocaleDateString('en-IN', { month: 'long', year: 'numeric' }),
   162→      icon: IndianRupee,
   163→      color: 'bg-violet-50 text-violet-600',
   164→      activeColor: 'bg-violet-100 text-violet-700',
   165→      activeBorder: 'border-violet-300 ring-1 ring-violet-100',
   166→    },
   167→    {
   168→      key: 'total',
   169→      label: 'Total',
   170→      value: mockCaregiver.earnings.total + (activeCard === 'total' ? liveOffset : 0),
   171→      sublabel: 'All time earnings',
   172→      icon: Clock,
   173→      color: 'bg-amber-50 text-amber-600',
   174→      activeColor: 'bg-amber-100 text-amber-700',
   175→      activeBorder: 'border-amber-300 ring-1 ring-amber-100',
   176→    },
   177→  ];
   178→
   179→  const activeSummaryCard = summaryCards.find((c) => c.key === activeCard)!;
   180→
   181→  // Chart data with live updates blended in
   182→  const chartData = chartPeriod === 'monthly'
   183→    ? mockMonthlyEarnings.map((d) => ({ ...d, earnings: d.earnings + Math.floor(Math.random() * 500) }))
   184→    : mockWeeklyEarnings.map((d) => ({ ...d, earnings: d.earnings + Math.floor(Math.random() * 100) }));
   185→
   186→  const chartTotal = chartData.reduce((sum, d) => sum + d.earnings, 0);
   187→  const chartAvg = Math.round(chartTotal / chartData.length);
   188→  const chartMax = Math.max(...chartData.map((d) => d.earnings));
   189→
   190→  // Status breakdown
   191→  const paidCount = mockEarnings.filter((e) => e.status === 'paid').length;
   192→  const pendingCount = mockEarnings.filter((e) => e.status === 'pending').length;
   193→  const processingCount = mockEarnings.filter((e) => e.status === 'processing').length;
   194→  const pieData = [
   195→    { name: 'Paid', value: paidCount },
   196→    { name: 'Pending', value: pendingCount },
   197→    { name: 'Processing', value: processingCount },
   198→  ].filter((d) => d.value > 0);
   199→
   200→  // Filtered earnings for selected period AND status
   201→  const today = new Date();
   202→  const filteredEarnings = mockEarnings.filter((e) => {
   203→    const eDate = new Date(e.date);
   204→    // Period filter
   205→    switch (activeCard) {
   206→      case 'today':
   207→        if (eDate.toDateString() !== today.toDateString()) return false;
   208→        break;
   209→      case 'thisWeek': {
   210→        const startOfWeek = new Date(today);
   211→        startOfWeek.setDate(today.getDate() - today.getDay());
   212→        if (eDate < startOfWeek) return false;
   213→        break;
   214→      }
   215→      case 'thisMonth':
   216→        if (eDate.getMonth() !== today.getMonth() || eDate.getFullYear() !== today.getFullYear()) return false;
   217→        break;
   218→      case 'total':
   219→      default:
   220→        break;
   221→    }
   222→    // Status filter
   223→    if (statusFilter !== 'all' && e.status !== statusFilter) return false;
   224→    return true;
   225→  });
   226→
   227→  const handleCardClick = (key: Period) => {
   228→    setAnimatingCard(key);
   229→    setActiveCard(key);
   230→    setLiveOffset(0);
   231→    setTimeout(() => setAnimatingCard(null), 400);
   232→  };
   233→
   234→  const handleRequestPayout = () => {
   235→    setPayoutRequested(true);
   236→    setTimeout(() => setPayoutRequested(false), 3000);
   237→  };
   238→
   239→  const handleDownload = () => {
   240→    setDownloading(true);
   241→    setTimeout(() => setDownloading(false), 2000);
   242→  };
   243→
   244→  const handleRowClick = (id: string) => {
   245→    setSelectedTransaction(selectedTransaction === id ? null : id);
   246→  };
   247→
   248→  // ─── Custom Pie Label ────────────────────────────────────────────────────
   249→
   250→  const renderCustomLabel = ({
   251→    cx,
   252→    cy,
   253→    midAngle,
   254→    innerRadius,
   255→    outerRadius,
   256→    percent,
   257→    name,
   258→  }: {
   259→    cx: number;
   260→    cy: number;
   261→    midAngle: number;
   262→    innerRadius: number;
   263→    outerRadius: number;
   264→    percent: number;
   265→    name: string;
   266→  }) => {
   267→    const RADIAN = Math.PI / 180;
   268→    const radius = innerRadius + (outerRadius - innerRadius) * 1.4;
   269→    const x = cx + radius * Math.cos(-midAngle * RADIAN);
   270→    const y = cy + radius * Math.sin(-midAngle * RADIAN);
   271→
   272→    return (
   273→      <text x={x} y={y} textAnchor={x > cx ? 'start' : 'end'} className="text-[11px] fill-gray-500 font-medium">
   274→        {name} ({(percent * 100).toFixed(0)}%)
   275→      </text>
   276→    );
   277→  };
   278→
   279→  return (
   280→    <div className="space-y-6">
   281→      {/* ===== Section 1: Summary Cards (Clickable) ===== */}
   282→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   283→        {summaryCards.map((card) => {
   284→          const Icon = card.icon;
   285→          const isActive = activeCard === card.key;
   286→          const isAnimating = animatingCard === card.key;
   287→          return (
   288→            <button
   289→              key={card.key}
   290→              onClick={() => handleCardClick(card.key)}
   291→              className={`bg-card rounded-xl border shadow-sm py-5 text-left transition-all duration-300 hover:shadow-md cursor-pointer w-full ${
   292→                isActive
   293→                  ? `${card.activeBorder} ${isAnimating ? 'scale-[1.03]' : 'scale-[1.01]'}`
   294→                  : 'border-gray-100 hover:border-gray-200'
   295→              }`}
   296→            >
   297→              <div className="flex items-center gap-3 mb-3 px-4">
   298→                <div
   299→                  className={`size-10 rounded-xl flex items-center justify-center transition-colors ${
   300→                    isActive ? card.activeColor : card.color
   301→                  }`}
   302→                >
   303→                  <Icon className="size-5" />
   304→                </div>
   305→                {isActive && (
   306→                  <div className="ml-auto">
   307→                    <div className="size-2 rounded-full bg-blue-500 animate-pulse" />
   308→                  </div>
   309→                )}
   310→              </div>
   311→              <div className="px-4">
   312→                <p className={`text-xs font-medium transition-colors ${isActive ? 'text-gray-700' : 'text-gray-500'}`}>
   313→                  {card.label}
   314→                </p>
   315→                <p className={`text-xl font-bold sm:text-2xl transition-colors ${isActive ? 'text-gray-900' : 'text-gray-900'}`}>
   316→                  {formatCurrency(card.value)}
   317→                </p>
   318→                <p className="text-[11px] text-gray-400 mt-0.5">{card.sublabel}</p>
   319→              </div>
   320→            </button>
   321→          );
   322→        })}
   323→      </div>
   324→
   325→      {/* Active card info bar */}
   326→      <div className="flex items-center gap-2 px-1">
   327→        <div className="size-2 rounded-full bg-blue-500" />
   328→        <span className="text-xs text-gray-500">
   329→          Viewing: <span className="font-semibold text-gray-700">{activeSummaryCard.label}</span> earnings
   330→          {filteredEarnings.length > 0 && (
   331→            <span className="text-gray-400"> — {filteredEarnings.length} transaction{filteredEarnings.length > 1 ? 's' : ''}</span>
   332→          )}
   333→        </span>
   334→      </div>
   335→
   336→      {/* ===== Section 2: Tabs ===== */}
   337→      <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   338→        <TabsList className="mb-4">
   339→          <TabsTrigger value="overview">Overview</TabsTrigger>
   340→          <TabsTrigger value="breakdown">Breakdown</TabsTrigger>
   341→          <TabsTrigger value="history">Payment History</TabsTrigger>
   342→        </TabsList>
   343→
   344→        {/* ——— Tab 1: Overview ——— */}
   345→        <TabsContent value="overview" className="space-y-4">
   346→          <div className="grid lg:grid-cols-3 gap-4">
   347→            {/* Chart Card with Monthly/Weekly toggle */}
   348→            <Card className="lg:col-span-2">
   349→              <CardHeader className="pb-2">
   350→                <div className="flex items-center justify-between">
   351→                  <CardTitle className="text-sm font-semibold text-gray-900">
   352→                    Earnings Overview
   353→                  </CardTitle>
   354→                  <div className="flex items-center gap-1 bg-gray-100 rounded-lg p-0.5">
   355→                    <button
   356→                      onClick={() => setChartPeriod('monthly')}
   357→                      className={`px-3 py-1 text-xs font-medium rounded-md transition-all ${
   358→                        chartPeriod === 'monthly'
   359→                          ? 'bg-white text-gray-900 shadow-sm'
   360→                          : 'text-gray-500 hover:text-gray-700'
   361→                      }`}
   362→                    >
   363→                      Monthly
   364→                    </button>
   365→                    <button
   366→                      onClick={() => setChartPeriod('weekly')}
   367→                      className={`px-3 py-1 text-xs font-medium rounded-md transition-all ${
   368→                        chartPeriod === 'weekly'
   369→                          ? 'bg-white text-gray-900 shadow-sm'
   370→                          : 'text-gray-500 hover:text-gray-700'
   371→                      }`}
   372→                    >
   373→                      Weekly
   374→                    </button>
   375→                  </div>
   376→                </div>
   377→              </CardHeader>
   378→              <CardContent>
   379→                <div className="h-[240px]">
   380→                  <ResponsiveContainer width="100%" height="100%">
   381→                    {chartPeriod === 'monthly' ? (
   382→                      <AreaChart data={chartData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   383→                        <defs>
   384→                          <linearGradient id="monthlyGradient" x1="0" y1="0" x2="0" y2="1">
   385→                            <stop offset="0%" stopColor="#3b82f6" stopOpacity={0.3} />
   386→                            <stop offset="100%" stopColor="#3b82f6" stopOpacity={0} />
   387→                          </linearGradient>
   388→                        </defs>
   389→                        <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
   390→                        <XAxis dataKey={chartPeriod === 'monthly' ? 'month' : 'day'} axisLine={false} tickLine={false} fontSize={11} tick={{ fill: '#94a3b8', fontSize: 11 }} />
   391→                        <YAxis hide />
   392→                        <Tooltip content={<CustomTooltip />} cursor={{ stroke: '#3b82f6', strokeWidth: 1, strokeDasharray: '4 4' }} />
   393→                        <Area type="monotone" dataKey="earnings" stroke="#3b82f6" strokeWidth={2} fill="url(#monthlyGradient)" />
   394→                      </AreaChart>
   395→                    ) : (
   396→                      <BarChart data={chartData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   397→                        <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
   398→                        <XAxis dataKey="day" axisLine={false} tickLine={false} fontSize={11} tick={{ fill: '#94a3b8', fontSize: 11 }} />
   399→                        <YAxis hide />
   400→                        <Tooltip content={<CustomTooltip />} cursor={{ fill: 'rgba(59,130,246,0.06)' }} />
   401→                        <Bar dataKey="earnings" radius={[6, 6, 0, 0]} maxBarSize={36}>
   402→                          {chartData.map((entry, index) => (
   403→                            <Cell
   404→                              key={index}
   405→                              fill={entry.earnings === 0 ? '#e2e8f0' : '#3b82f6'}
   406→                            />
   407→                          ))}
   408→                        </Bar>
   409→                      </BarChart>
   410→                    )}
   411→                  </ResponsiveContainer>
   412→                </div>
   413→                {/* Chart Stats Row */}
   414→                <div className="grid grid-cols-3 gap-3 mt-4 pt-4 border-t border-gray-100">
   415→                  <div className="text-center">
   416→                    <p className="text-[11px] text-gray-400 font-medium">Total</p>
   417→                    <p className="text-sm font-bold text-gray-900">{formatCurrency(chartTotal)}</p>
   418→                  </div>
   419→                  <div className="text-center">
   420→                    <p className="text-[11px] text-gray-400 font-medium">Average</p>
   421→                    <p className="text-sm font-bold text-gray-900">{formatCurrency(chartAvg)}</p>
   422→                  </div>
   423→                  <div className="text-center">
   424→                    <p className="text-[11px] text-gray-400 font-medium">Peak</p>
   425→                    <p className="text-sm font-bold text-emerald-600">{formatCurrency(chartMax)}</p>
   426→                  </div>
   427→                </div>
   428→              </CardContent>
   429→            </Card>
   430→
   431→            {/* Right Column — Status Pie + Quick Stats */}
   432→            <div className="space-y-4">
   433→              {/* Status Breakdown Pie */}
   434→              <Card>
   435→                <CardHeader className="pb-2">
   436→                  <CardTitle className="text-sm font-semibold text-gray-900">
   437→                    Status Breakdown
   438→                  </CardTitle>
   439→                </CardHeader>
   440→                <CardContent>
   441→                  <div className="h-[160px]">
   442→                    <ResponsiveContainer width="100%" height="100%">
   443→                      <PieChart>
   444→                        <Pie
   445→                          data={pieData}
   446→                          cx="50%"
   447→                          cy="50%"
   448→                          innerRadius={35}
   449→                          outerRadius={55}
   450→                          dataKey="value"
   451→                          labelLine={false}
   452→                          label={renderCustomLabel}
   453→                        >
   454→                          {pieData.map((_, index) => (
   455→                            <Cell key={index} fill={PIE_COLORS[index % PIE_COLORS.length]} />
   456→                          ))}
   457→                        </Pie>
   458→                        <Tooltip
   459→                          formatter={(value: number) => [`${value} transactions`, 'Count']}
   460→                          contentStyle={{
   461→                            borderRadius: '8px',
   462→                            border: '1px solid #f1f5f9',
   463→                            fontSize: '12px',
   464→                          }}
   465→                        />
   466→                      </PieChart>
   467→                    </ResponsiveContainer>
   468→                  </div>
   469→                  {/* Legend */}
   470→                  <div className="flex items-center justify-center gap-4 mt-2">
   471→                    {pieData.map((item, index) => (
   472→                      <button
   473→                        key={item.name}
   474→                        className="flex items-center gap-1.5 text-xs text-gray-600 hover:text-gray-900 transition-colors cursor-pointer"
   475→                      >
   476→                        <div className="size-2.5 rounded-full" style={{ backgroundColor: PIE_COLORS[index] }} />
   477→                        {item.name} ({item.value})
   478→                      </button>
   479→                    ))}
   480→                  </div>
   481→                </CardContent>
   482→              </Card>
   483→
   484→              {/* Quick Stats */}
   485→              <Card>
   486→                <CardContent className="p-4 space-y-3">
   487→                  <div className="flex items-center justify-between">
   488→                    <div className="flex items-center gap-2">
   489→                      <Receipt className="size-4 text-gray-400" />
   490→                      <span className="text-xs text-gray-500">Total Transactions</span>
   491→                    </div>
   492→                    <span className="text-sm font-bold text-gray-900">{mockEarnings.length}</span>
   493→                  </div>
   494→                  <div className="h-px bg-gray-100" />
   495→                  <div className="flex items-center justify-between">
   496→                    <div className="flex items-center gap-2">
   497→                      <TrendingUp className="size-4 text-emerald-500" />
   498→                      <span className="text-xs text-gray-500">Avg. per Transaction</span>
   499→                    </div>
   500→                    <span className="text-sm font-bold text-gray-900">
   501→                      {formatCurrency(Math.round(mockEarnings.reduce((s, e) => s + e.amount, 0) / mockEarnings.length))}
   502→                    </span>
   503→                  </div>
   504→                  <div className="h-px bg-gray-100" />
   505→                  <div className="flex items-center justify-between">
   506→                    <div className="flex items-center gap-2">
   507→                      <Sparkles className="size-4 text-amber-500" />
   508→                      <span className="text-xs text-gray-500">Best Day</span>
   509→                    </div>
   510→                    <span className="text-sm font-bold text-gray-900">{formatCurrency(chartMax)}</span>
   511→                  </div>
   512→                </CardContent>
   513→              </Card>
   514→            </div>
   515→          </div>
   516→
   517→          {/* Payout Summary — Clickable Status Segments */}
   518→          <Card>
   519→            <CardHeader className="pb-3">
   520→              <div className="flex items-center justify-between">
   521→                <CardTitle className="text-sm font-semibold text-gray-900">
   522→                  Payout Summary
   523→                </CardTitle>
   524→                <Badge variant="outline" className="text-[10px] text-gray-500 border-gray-200 bg-gray-50">
   525→                  <Info className="size-3 mr-1" />
   526→                  Click segments for details
   527→                </Badge>
   528→              </div>
   529→            </CardHeader>
   530→            <CardContent className="space-y-3">
   531→              {/* Paid Segment */}
   532→              <button
   533→                onClick={() => { setStatusFilter('paid'); setActiveTab('history'); }}
   534→                className={`w-full flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 rounded-lg border p-4 transition-all cursor-pointer text-left group ${
