    71→    71→function getDateGroupLabel(dateStr: string): string {
    72→    72→  const date = parseISO(dateStr);
    73→    73→  if (isToday(date)) return 'Today';
    74→    74→  if (isTomorrow(date)) return 'Tomorrow';
    75→    75→  if (isYesterday(date)) return 'Yesterday';
    76→    76→  return format(date, 'EEE, MMM d, yyyy');
    77→    77→}
    78→    78→
    79→    79→function getDateGroupSortKey(dateStr: string): number {
    80→    80→  const date = parseISO(dateStr);
    81→    81→  if (isToday(date)) return 0;
    82→    82→  if (isTomorrow(date)) return 1;
    83→    83→  if (isYesterday(date)) return -1;
    84→    84→  return date.getTime();
    85→    85→}

    97→    97→}
