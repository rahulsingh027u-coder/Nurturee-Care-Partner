   184→function loadDocsFromStorage(): VerificationDoc[] {
   185→  try {
   186→    // Version check — if storage format is outdated, clear and start fresh
   187→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   188→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   189→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   190→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   191→      return getInitialDocs();
   192→    }
   193→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   194→    if (stored) {
   195→      const parsed = JSON.parse(stored);
   196→      if (Array.isArray(parsed) && parsed.length === 4) {
   197→        // Validate structure — reject data from old format
   198→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   199→        if (isValid) return parsed;
   200→      }
   201→    }
   202→  } catch {
   203→    // fall through
   204→  }
   205→  return getInitialDocs();
   206→}
   207→
   208→function saveDocsToStorage(docs: VerificationDoc[]) {
   209→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   210→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   211→}

   230→// ─── Optional Deadline Helpers ─────────────────────────────────────────────────
   231→
   232→const OPTIONAL_DEADLINE_DAYS = 15;
   233→const REGISTERED_AT_KEY = 'nurturee_care_partner_registered_at';
   234→
   235→function getDaysRemaining(): number | null {
   236→  try {
   237→    const registeredAt = localStorage.getItem(REGISTERED_AT_KEY);
   238→    if (!registeredAt) return null;
   239→    const registered = new Date(registeredAt).getTime();
   240→    const deadline = registered + OPTIONAL_DEADLINE_DAYS * 24 * 60 * 60 * 1000;
   241→    const remaining = Math.ceil((deadline - Date.now()) / (24 * 60 * 60 * 1000));
   242→    return Math.max(0, remaining);
   243→  } catch {
   244→    return null;
   245→  }
   246→}

   250→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   251→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   252→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   253→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   254→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   255→  const [uploadedFront, setUploadedFront] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   256→  const [uploadedBack, setUploadedBack] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   257→  const [submitting, setSubmitting] = useState(false);
   258→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   259→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   260→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   261→  const tempCalendarDateRef = useRef<Record<string, Date | undefined>>({});
   262→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   263→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   264→
   265→  const currentYear = new Date().getFullYear();
   266→
   267→  // Clean up timers on unmount
   268→  useEffect(() => {
   269→    return () => {
   270→      Object.values(reviewTimersRef.current).forEach((timers) => {
   271→        timers.forEach(clearTimeout);
   272→      });
   273→    };
   274→  }, []);
   275→
   276→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   277→  useEffect(() => {
   278→    if (submittedType) {
   279→      const doc = docs.find((d) => d.type === submittedType);
   280→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   281→        setSubmittedType(null);
   282→        setReviewingType(null);
   283→      }
   284→    }
   285→    if (reviewingType) {
   286→      const doc = docs.find((d) => d.type === reviewingType);
   287→      if (!doc || doc.status !== 'under_review') {
   288→        setReviewingType(null);
   289→      }
   290→    }
   291→  }, [docs, submittedType, reviewingType]);
   292→
   293→  // Persist docs to localStorage whenever they change
   294→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   295→    setDocs((prev) => {
   296→      const updated = updater(prev);
   297→      saveDocsToStorage(updated);
   298→      return updated;
   299→    });
   300→  }, []);
   301→
   302→  // ─── Backend Review Simulation ───────────────────────────────────────────
   303→
   304→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   305→    // Clear any existing timers for this type
   306→    if (reviewTimersRef.current[type]) {
   307→      reviewTimersRef.current[type].forEach(clearTimeout);
   308→    }
   309→    reviewTimersRef.current[type] = [];
   310→
   311→    const doc = docs.find((d) => d.type === type);
   312→    if (!doc) return;
   313→
   314→    // Timer 1: After 3s → under_review
   315→    const t1 = setTimeout(() => {
   316→      setReviewingType(type);
   317→      updateDocs((prev) =>
   318→        prev.map((d) =>
   319→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   320→        )
   321→      );
   322→    }, 3000);
   323→
   324→    // Timer 2: After 5-7s total → resolve
   325→    const resolveDelay = 5000 + Math.random() * 2000;
   326→    const t2 = setTimeout(() => {
   327→      const currentDoc = docs.find((d) => d.type === type);
   328→      if (!currentDoc) return;
   329→
   330→      let willApprove: boolean;
   331→
   332→      if (type === 'aadhaar') {
   333→        // Aadhaar always approve on first attempt (demo-friendly)
   334→        willApprove = true;
   335→      } else {
   336→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   337→        if (attemptNumber === 1) {
   338→          willApprove = Math.random() < 0.4;
   339→        } else if (attemptNumber === 2) {
   340→          willApprove = Math.random() < 0.7;
   341→        } else {
   342→          willApprove = true;
   343→        }
   344→      }
   345→
   346→      if (willApprove) {
   347→        // Approve
   348→        setApprovingType(type);
   349→        updateDocs((prev) =>
   350→          prev.map((d) => {
   351→            if (d.type !== type) return d;
   352→            return {
   353→              ...d,
   354→              status: 'approved' as VerificationStatus,
   355→              attempts: d.attempts.map((a) =>
   356→                a.attemptNumber === attemptNumber
   357→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   358→                  : a
   359→              ),
   360→            };
   361→          })
   362→        );
   363→        setReviewingType(null);
   364→
   365→        // If Aadhaar approved and onComplete exists, call it after delay
   366→        if (type === 'aadhaar' && onComplete) {
   367→          setTimeout(() => {
   368→            setApprovingType(null);
   369→            setSubmittedType(null);
   370→            onComplete();
   371→          }, 1000);
   372→        } else {
   373→          // For optional: show success briefly, then allow going back
   374→          setTimeout(() => {
   375→            setApprovingType(null);
   376→            setSubmittedType(null);
   377→            setActiveType(null);
   378→          }, 2000);
   379→        }
   380→      } else {
   381→        // Reject
   382→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   383→        updateDocs((prev) =>
   384→          prev.map((d) => {
   385→            if (d.type !== type) return d;
   386→            const updatedAttempts = d.attempts.map((a) =>
   387→              a.attemptNumber === attemptNumber
   388→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   389→                : a
   390→            );
   391→            // Check if max attempts reached
   392→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   393→            return {
   394→              ...d,
   395→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   396→              attempts: updatedAttempts,
   397→              currentDraft: { documents: [], formFields: {} },
   398→            };
   399→          })
   400→        );
   401→        setReviewingType(null);
   402→        // Transition from submitted view to allow re-upload
   403→        setTimeout(() => {
   404→          setSubmittedType(null);
   405→        }, 300);
   406→      }
   407→    }, resolveDelay);
   408→
   409→    reviewTimersRef.current[type] = [t1, t2];
   410→  }, [docs, onComplete, updateDocs]);
   411→
   412→  // ─── Handlers ────────────────────────────────────────────────────────────
   413→
   414→  const handleSelectType = (type: VerificationType) => {

   416→    if (!doc) return;
   417→    setActiveType(type);

   420→    if (!isFrozenDoc) {

   427→      } else {
   428→        setUploadedFiles(doc.currentDraft?.documents || []);
   429→        setUploadedFront([]);
   430→        setUploadedBack([]);
   431→      }
   432→    } else {
   433→      setFormFields({});
   434→      setUploadedFiles([]);
   435→      setUploadedFront([]);
   436→      setUploadedBack([]);
   437→    }
   438→    setSubmittedType(null);
   439→  };
   440→
   441→  const handleBack = () => {
   442→    // Save draft
   443→    if (activeType) {
   444→      // For Aadhaar, combine front + back into documents array
   445→      const isAadhaar = activeType === 'aadhaar';
   446→      const combinedDocs = isAadhaar
   447→        ? [...uploadedFront.map(f => ({ ...f, id: `front-${f.id}` })), ...uploadedBack.map(f => ({ ...f, id: `back-${f.id}` }))]
   448→        : uploadedFiles;
   449→      updateDocs((prev) =>
   450→        prev.map((d) =>
   451→          d.type === activeType
   452→            ? { ...d, currentDraft: { documents: combinedDocs, formFields } }
   453→            : d
   454→        )
   455→      );
   456→    }
   457→    // Clean up timers for this type if we navigate back
   458→    if (activeType && reviewTimersRef.current[activeType]) {
   459→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   460→      if (!submittedType && !reviewingType) {
   461→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   462→        delete reviewTimersRef.current[activeType];
   463→      }
   464→    }
   465→    setActiveType(null);
   466→    setFormFields({});
   467→    setUploadedFiles([]);
   468→    setUploadedFront([]);
   469→    setUploadedBack([]);
   470→    setSubmittedType(null);
   471→    setReviewingType(null);
   472→  };
   473→
   474→  const validateAndCreateFileEntries = (files: FileList, config: typeof verificationConfig[VerificationType] | null) => {
   475→    const result: Array<{ id: string; name: string; size: number; type: string }> = [];
   476→    for (let i = 0; i < files.length; i++) {
   477→      const file = files[i];
   478→      if (config && !config.allowedTypes.includes(file.type)) {
   479→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   480→        continue;
   481→      }
   482→      if (config && file.size > config.maxFileSize) {
   483→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   484→        continue;
   485→      }
   486→      result.push({ id: `file-${Date.now()}-${i}`, name: file.name, size: file.size, type: file.type });
   487→    }
   488→    return result;
   489→  };
   490→
   491→  // Generic file change (for non-Aadhaar docs)
   492→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {

   494→    if (!files || files.length === 0) return;

   496→    const newFiles = validateAndCreateFileEntries(files, config);
   497→    if (newFiles.length > 0) {
   498→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   499→    }
   500→    e.target.value = '';
   501→  };
   502→
   503→  // Aadhaar-specific: handle front/back file uploads
   504→  const handleAadhaarSideChange = (side: 'front' | 'back') => (e: React.ChangeEvent<HTMLInputElement>) => {
   505→    const files = e.target.files;
   506→    if (!files || files.length === 0) return;
   507→    const config = verificationConfig['aadhaar'];
   508→    const newFiles = validateAndCreateFileEntries(files, config);
   509→    const setter = side === 'front' ? setUploadedFront : setUploadedBack;
   510→    if (newFiles.length > 0) {
   511→      setter(newFiles); // Replace — only one file per side
   512→    }
   513→    e.target.value = '';
   514→  };
   515→
   516→  const handleRemoveFile = (fileId: string) => {
   517→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   518→  };
   519→  const handleRemoveAadhaarSide = (side: 'front' | 'back') => () => {
   520→    if (side === 'front') setUploadedFront([]);
   521→    else setUploadedBack([]);
   522→  };
   523→
   524→  const handleSubmit = async () => {
   525→    if (!activeType) return;
   526→    setSubmitting(true);
