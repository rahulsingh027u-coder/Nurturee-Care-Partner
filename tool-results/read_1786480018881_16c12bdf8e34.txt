   594→                  <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-blue-50">
   595→                    <Briefcase className="w-4 h-4 text-blue-600" />
   596→                  </div>
   597→                  Shift Details
   598→                </SheetTitle>
   599→              </SheetHeader>
   600→
   601→              <div className="mt-4 space-y-4">
   602→                {/* Status & Date */}
   603→                <div className="flex items-center justify-between">
   604→                  <div className="flex items-center gap-2">
   605→                    {getStatusBadge(detailShift.status)}
   606→                    {getModeBadge(detailShift.consultationMode)}
   607→                  </div>
   608→                  <span className="text-xs text-gray-500">{format(parseISO(detailShift.date), 'EEEE, MMMM d, yyyy')}</span>
   609→                </div>
   610→
   611→                {/* Time */}
   612→                <div className="flex items-center gap-2 text-gray-700">
   613→                  <Clock className="w-4 h-4 text-gray-400" />
   614→                  <span className="text-sm font-medium">{detailShift.startTime} — {detailShift.endTime}</span>
   615→                </div>
   616→
   617→                {/* Care Member */}
   618→                <div
   619→                  className="rounded-xl border border-gray-100 bg-gray-50/50 p-3 space-y-2 cursor-pointer hover:bg-blue-50/30 transition-colors"
   620→                  onClick={() => { setDetailShift(null); onNavigate('patients', detailShift.patientId); }}
   621→                >
   622→                  <p className="text-xs font-medium text-gray-400 uppercase tracking-wide">Care Member</p>
   623→                  <div className="flex items-center justify-between">
   624→                    <p className="text-sm font-semibold text-gray-900">{detailShift.patientName}</p>

   626→                  </div>
   627→                  <p className="text-xs text-gray-500">{detailShift.serviceType}</p>
   628→                  {detailShift.patientPhone && (
   629→                    <p className="text-xs text-gray-400 mt-1">{detailShift.patientPhone}</p>
   630→                  )}
   631→                </div>
   632→
   633→                {/* Booking ID */}
   634→                <div className="flex items-center gap-2">
   635→                  <span className="text-xs font-medium text-gray-400">Booking ID:</span>
   636→                  <span className="text-xs text-gray-600 font-mono">{detailShift.bookingId}</span>
   637→                </div>
   638→
   639→                {/* Address */}
   640→                {detailShift.consultationMode === 'in_home' && detailShift.address && (
   641→                  <div className="flex items-start gap-2">
   642→                    <MapPin className="w-4 h-4 text-gray-400 mt-0.5 shrink-0" />
   643→                    <span className="text-xs text-gray-600">{detailShift.address}</span>
   644→                  </div>
   645→                )}
   646→
   647→                {/* Special Instructions */}
   648→                {detailShift.specialInstructions && (
   649→                  <div className="rounded-lg bg-amber-50 border border-amber-200 p-3">
   650→                    <div className="flex items-start gap-1.5">
   651→                      <AlertCircle className="w-4 h-4 text-amber-600 mt-0.5 shrink-0" />
   652→                      <div>
   653→                        <p className="text-xs font-medium text-amber-700 mb-0.5">Special Instructions</p>
   654→                        <p className="text-xs text-amber-700">{detailShift.specialInstructions}</p>
   655→                      </div>
   656→                    </div>
   657→                  </div>
   658→                )}
   659→
   660→                {/* Earnings */}
   661→                {detailShift.earnings > 0 && (
   662→                  <div className="rounded-xl border border-emerald-100 bg-emerald-50 p-3 flex items-center justify-between">
   663→                    <span className="text-xs font-medium text-emerald-700">Earnings</span>
   664→                    <span className="text-lg font-bold text-emerald-700">{formatCurrency(detailShift.earnings)}</span>
   665→                  </div>
   666→                )}
   667→
   668→                {/* Actions */}
   669→                <div className="flex items-center gap-2 pt-2 flex-wrap">
   670→                  {detailShift.status === 'in_progress' && (
   671→                    <Button
   672→                      size="sm"
   673→                      className="bg-gradient-to-r from-emerald-600 to-emerald-500 hover:from-emerald-700 hover:to-emerald-600 text-white rounded-lg text-xs"
   674→                      onClick={() => { setDetailShift(null); setCompleteShift(detailShift); }}
   675→                    >
   676→                      <CheckCircle2 className="size-3.5 mr-1" />
   677→                      Complete Shift
   678→                    </Button>
   679→                  )}
   680→                  {detailShift.status === 'in_progress' && (
   681→                    <Button
   682→                      size="sm"
   683→                      variant="outline"
   684→                      className="rounded-lg text-xs text-red-600 border-red-200 hover:bg-red-50"
   685→                      onClick={() => { setDetailShift(null); setReportIssueOpen(true); }}
   686→                    >
   687→                      <AlertCircle className="size-3.5 mr-1" />
   688→                      Report Issue
   689→                    </Button>
   690→                  )}
   691→                  <Button
   692→                    variant="outline"
   693→                    size="sm"
   694→                    className="rounded-lg text-xs ml-auto"
   695→                    onClick={() => { setDetailShift(null); onNavigate('shifts'); }}
   696→                  >
   697→                    View All Shifts
   698→                    <ArrowRight className="size-3.5 ml-1" />
   699→                  </Button>
   700→                </div>
   701→              </div>
   702→            </>
   703→          )}
   704→        </SheetContent>
   705→      </Sheet>
   706→
   707→      {/* Shift Complete OTP Modal */}
   708→      {completeShift && (
   709→        <ShiftCompleteOTP
   710→          shiftId={completeShift.id}
   711→          bookingId={completeShift.bookingId}
   712→          careMemberName={completeShift.patientName}
   713→          careMemberPhone={completeShift.patientPhone}
   714→          serviceType={completeShift.serviceType}
   715→          serviceCategory={completeShift.serviceCategory}
   716→          address={completeShift.address || ''}
   717→          startTime={completeShift.startTime}
   718→          endTime={completeShift.endTime}
   719→          earnings={completeShift.earnings}
   720→          onComplete={(shiftId) => {
   721→            setCompleteShift(null);
   722→            setShiftCompleted(true);
   723→            setTimeout(() => setShiftCompleted(false), 3000);
   724→          }}
   725→          onCancel={() => setCompleteShift(null)}
   726→        />
   727→      )}
   728→
   729→      {shiftCompleted && (
   730→        <div className="fixed bottom-20 md:bottom-4 left-1/2 -translate-x-1/2 z-50 bg-emerald-600 text-white px-4 py-2 rounded-lg shadow-lg text-sm font-medium flex items-center gap-2 animate-in slide-in-from-bottom-5">
   731→          <CheckCircle className="size-4" />
   732→          Shift completed successfully!
   733→        </div>
   734→      )}
   735→
   736→      {/* Report Issue Dialog (for active shifts) */}
   737→      {reportIssueOpen && (
   738→        <ReportIssueDialog
   739→          open={reportIssueOpen}
   740→          onClose={() => setReportIssueOpen(false)}
   741→        />
   742→      )}
   743→
   744→      {/* SOS Emergency Dialog */}
   745→      <SOSEmergencyDialog
   746→        open={sosOpen}
   747→        onClose={() => setSosOpen(false)}
   748→      />
   749→    </div>
   750→  );
   751→}
