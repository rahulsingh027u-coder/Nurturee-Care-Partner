    63→    63→function getDateGroupLabel(dateStr: string): string {
    64→    64→  const date = parseISO(dateStr);
    65→    65→  if (isToday(date)) return 'Today';
    66→    66→  if (isTomorrow(date)) return 'Tomorrow';
    67→    67→  if (isYesterday(date)) return 'Yesterday';
    68→    68→  return format(date, 'EEE, MMM d, yyyy');
    69→    69→}
    70→    70→
    71→    71→function getDateGroupSortKey(dateStr: string): number {
    72→    72→  const date = parseISO(dateStr);
    73→    73→  if (isToday(date)) return 0;
    74→    74→  if (isTomorrow(date)) return 1;
    75→    75→  if (isYesterday(date)) return -1;
    76→    76→  return date.getTime();
    77→    77→}
