     1	'use client';
     2	
     3	import { useState, useMemo } from 'react';
     4	import Image from 'next/image';
     5	import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card';
     6	import { Badge } from '@/components/ui/badge';
     7	import { Button } from '@/components/ui/button';
     8	import { Input } from '@/components/ui/input';
     9	import { Label } from '@/components/ui/label';
    10	import { Textarea } from '@/components/ui/textarea';
    11	import { Switch } from '@/components/ui/switch';
    12	import {
    13	  Sparkles,
    14	  ArrowLeft,
    15	  ArrowRight,
    16	  CheckCircle,
    17	  Heart,
    18	  Baby,
    19	  Shield,
    20	  Award,
    21	  BadgeCheck,
    22	  Plus,
    23	  Minus,
    24	  AlertCircle,
    25	} from 'lucide-react';
    26	import {
    27	  SKILLS_CATALOG,
    28	  SERVICES_CATALOG,
    29	  DAYS_OF_WEEK,
    30	  type SkillLevel,
    31	  type OnboardingData,
    32	  storeOnboarding,
    33	} from '@/lib/onboarding-data';
    34	
    35	const TOTAL_STEPS = 5;
    36	
    37	const LEVEL_CONFIG: Record<SkillLevel, { color: string; bg: string; border: string; label: string }> = {
    38	  expert: { color: 'text-emerald-700', bg: 'bg-emerald-50', border: 'border-emerald-200', label: 'Expert' },
    39	  advanced: { color: 'text-blue-700', bg: 'bg-blue-50', border: 'border-blue-200', label: 'Advanced' },
    40	  intermediate: { color: 'text-violet-700', bg: 'bg-violet-50', border: 'border-violet-200', label: 'Intermediate' },
    41	  beginner: { color: 'text-gray-600', bg: 'bg-gray-50', border: 'border-gray-200', label: 'Beginner' },
    42	};
    43	
    44	interface OnboardingWizardProps {
    45	  phone: string;
    46	  onComplete: () => void;
    47	}
    48	
    49	export default function OnboardingWizard({ phone, onComplete }: OnboardingWizardProps) {
    50	  const [step, setStep] = useState(1);
    51	
    52	  // ─── Step 1: Personal Details ──────────────────────────────────────────
    53	  const [name, setName] = useState('');
    54	  const [email, setEmail] = useState('');
    55	  const [qualifications, setQualifications] = useState('');
    56	  const [languages, setLanguages] = useState('');
    57	  const [bio, setBio] = useState('');
    58	  const [experience, setExperience] = useState('');
    59	
    60	  // ─── Step 2: Specialist Type ────────────────────────────────────────────
    61	  const [specialty, setSpecialty] = useState<'elder_care' | 'child_care' | null>(null);
    62	
    63	  // ─── Step 3: Skills & Services ─────────────────────────────────────────
    64	  const [selectedSkills, setSelectedSkills] = useState<Array<{ name: string; level: SkillLevel; years: number }>>([]);
    65	  const [selectedServices, setSelectedServices] = useState<Array<{ name: string; description: string }>>([]);
    66	  const [customSkillName, setCustomSkillName] = useState('');
    67	  const [customServiceName, setCustomServiceName] = useState('');
    68	  const [customServiceDescription, setCustomServiceDescription] = useState('');
    69	
    70	  // ─── Step 4: Certifications ───────────────────────────────────────────
    71	  const [certifications, setCertifications] = useState<Array<{ name: string; issuer: string; year: number; featured: boolean }>>([]);
    72	  const [newCertName, setNewCertName] = useState('');
    73	  const [newCertIssuer, setNewCertIssuer] = useState('');
    74	  const [newCertYear, setNewCertYear] = useState('');
    75	
    76	  // ─── Step 5: Availability ─────────────────────────────────────────────
    77	  const [availability, setAvailability] = useState(
    78	    DAYS_OF_WEEK.map((day) => ({ day, isAvailable: false, startTime: '09:00', endTime: '18:00' }))
    79	  );
    80	
    81	  // ─── Helpers ────────────────────────────────────────────────────────────
    82	
    83	  const isSkillSelected = (skillName: string) => selectedSkills.some((s) => s.name === skillName);
    84	
    85	  const toggleSkill = (skillName: string) => {
    86	    setSelectedSkills((prev) => {
    87	      if (prev.some((s) => s.name === skillName)) {
    88	        return prev.filter((s) => s.name !== skillName);
    89	      }
    90	      return [...prev, { name: skillName, level: 'intermediate' as SkillLevel, years: 1 }];
    91	    });
    92	  };
    93	
    94	  const updateSkillLevel = (skillName: string, level: SkillLevel) => {
    95	    setSelectedSkills((prev) =>
    96	      prev.map((s) => (s.name === skillName ? { ...s, level } : s))
    97	    );
    98	  };
    99	
   100	  const updateSkillYears = (skillName: string, years: number) => {
   101	    setSelectedSkills((prev) =>
   102	      prev.map((s) => (s.name === skillName ? { ...s, years: Math.max(0, Math.min(50, years)) } : s))
   103	    );
   104	  };
   105	
   106	  const addCustomSkill = () => {
   107	    const trimmed = customSkillName.trim();
   108	    if (!trimmed || isSkillSelected(trimmed)) return;
   109	    setSelectedSkills((prev) => [...prev, { name: trimmed, level: 'intermediate' as SkillLevel, years: 1 }]);
   110	    setCustomSkillName('');
   111	  };
   112	
   113	  const isServiceSelected = (serviceName: string) => selectedServices.some((s) => s.name === serviceName);
   114	
   115	  const toggleService = (serviceName: string, description: string) => {
   116	    setSelectedServices((prev) => {
   117	      if (prev.some((s) => s.name === serviceName)) {
   118	        return prev.filter((s) => s.name !== serviceName);
   119	      }
   120	      return [...prev, { name: serviceName, description }];
   121	    });
   122	  };
   123	
   124	  const addCustomService = () => {
   125	    const n = customServiceName.trim();
   126	    if (!n || isServiceSelected(n)) return;
   127	    setSelectedServices((prev) => [...prev, { name: n, description: customServiceDescription.trim() || 'Custom service' }]);
   128	    setCustomServiceName('');
   129	    setCustomServiceDescription('');
   130	  };
   131	
   132	  const addCertification = () => {
   133	    const n = newCertName.trim();
   134	    const issuer = newCertIssuer.trim();
   135	    const year = parseInt(newCertYear.trim(), 10);
   136	    if (!n || !issuer || !year || year < 1950 || year > new Date().getFullYear() + 1) return;
   137	    if (certifications.some((c) => c.name.toLowerCase() === n.toLowerCase())) return;
   138	    setCertifications((prev) => [...prev, { name: n, issuer, year, featured: false }]);
   139	    setNewCertName('');
   140	    setNewCertIssuer('');
   141	    setNewCertYear('');
   142	  };
   143	
   144	  const removeCertification = (certName: string) => {
   145	    setCertifications((prev) => prev.filter((c) => c.name !== certName));
   146	  };
   147	
   148	  const toggleDayAvailability = (index: number) => {
   149	    setAvailability((prev) =>
   150	      prev.map((day, i) => (i === index ? { ...day, isAvailable: !day.isAvailable } : day))
   151	    );
   152	  };
   153	
   154	  const updateTime = (index: number, field: 'startTime' | 'endTime', value: string) => {
   155	    setAvailability((prev) =>
   156	      prev.map((day, i) => (i === index ? { ...day, [field]: value } : day))
   157	    );
   158	  };
   159	
   160	  const skillsByCategory = useMemo(() => {
   161	    const grouped: Record<string, typeof SKILLS_CATALOG[number][]> = {};
   162	    for (const skill of SKILLS_CATALOG) {
   163	      const cat = skill.category;
   164	      if (!grouped[cat]) grouped[cat] = [];
   165	      grouped[cat].push(skill);
   166	    }
   167	    return grouped;
   168	  }, []);
   169	
   170	  const servicesByCategory = useMemo(() => {
   171	    const grouped: Record<string, typeof SERVICES_CATALOG[number][]> = {};
   172	    for (const service of SERVICES_CATALOG) {
   173	      const cat = service.category;
   174	      if (!grouped[cat]) grouped[cat] = [];
   175	      grouped[cat].push(service);
   176	    }
   177	    return grouped;
   178	  }, []);
   179	
   180	  // ─── Validation per step ───────────────────────────────────────────────
   181	
   182	  const isStep1Valid = name.trim().length >= 2 && email.trim().includes('@') && qualifications.trim().length > 0 && parseInt(experience, 10) > 0;
   183	  const isStep2Valid = specialty !== null;
   184	  const isStep3Valid = selectedSkills.length > 0;
   185	  const isStep4Valid = true; // certs are optional
   186	  const isStep5Valid = true; // availability can be all off
   187	
   188	  const canProceed = (): boolean => {
   189	    switch (step) {
   190	      case 1: return isStep1Valid;
   191	      case 2: return isStep2Valid;
   192	      case 3: return isStep3Valid;
   193	      case 4: return isStep4Valid;
   194	      case 5: return isStep5Valid;
   195	      default: return false;
   196	    }
   197	  };
   198	
   199	  // ─── Navigation ────────────────────────────────────────────────────────
   200	
   201	  const handleNext = () => {
   202	    if (step < TOTAL_STEPS) {
   203	      setStep(step + 1);
   204	      window.scrollTo({ top: 0, behavior: 'smooth' });
   205	    } else {
   206	      // Final step — save onboarding data and proceed to verification
   207	      const data: OnboardingData = {
   208	        name: name.trim(),
   209	        email: email.trim(),
   210	        phone,
   211	        qualifications: qualifications.trim(),
   212	        languages: languages.trim(),
   213	        bio: bio.trim(),
   214	        experience: parseInt(experience, 10) || 1,
   215	        specialty: specialty!,
   216	        skills: selectedSkills,
   217	        services: selectedServices,
   218	        certifications,
   219	        availability,
   220	      };
   221	      storeOnboarding(data);
   222	      onComplete();
   223	    }
   224	  };
   225	
   226	  const handleBack = () => {
   227	    if (step > 1) {
   228	      setStep(step - 1);
   229	      window.scrollTo({ top: 0, behavior: 'smooth' });
   230	    }
   231	  };
   232	
   233	  // ─── Step indicator ────────────────────────────────────────────────────
   234	
   235	  const stepLabels = ['Personal', 'Specialist', 'Skills', 'Certifications', 'Availability'];
   236	
   237	  return (
   238	    <div className="min-h-screen bg-gray-50 flex flex-col">
   239	      {/* Top accent bar */}
   240	      <div className="h-1.5 bg-gradient-to-r from-blue-600 to-indigo-600" />
   241	
   242	      {/* Sticky header */}
   243	      <div className="sticky top-0 z-30 bg-white/80 backdrop-blur-lg border-b border-gray-200">
   244	        <div className="max-w-lg mx-auto px-4 py-4">
   245	          <div className="flex items-center justify-between mb-3">
   246	            <div className="flex items-center gap-2">
   247	              <Image src="/nurturee-icon.png" alt="Nurturee" width={28} height={28} className="rounded-lg" />
   248	              <span className="text-lg font-bold text-gray-900 tracking-tight">Nurturee</span>
   249	            </div>
   250	            <Badge variant="outline" className="text-xs text-blue-600 border-blue-200 bg-blue-50">
   251	              Step {step} of {TOTAL_STEPS}
   252	            </Badge>
   253	          </div>
   254	          {/* Progress bar */}
   255	          <div className="flex gap-1">
   256	            {Array.from({ length: TOTAL_STEPS }).map((_, i) => (
   257	              <div
   258	                key={i}
   259	                className={`h-1.5 flex-1 rounded-full transition-colors duration-300 ${
   260	                  i < step ? 'bg-blue-600' : 'bg-gray-200'
   261	                }`}
   262	              />
   263	            ))}
   264	          </div>
   265	          <div className="flex justify-between mt-1.5">
   266	            {stepLabels.map((label, i) => (
   267	              <span
   268	                key={label}
   269	                className={`text-[10px] transition-colors ${
   270	                  i === step - 1 ? 'text-blue-600 font-semibold' : i < step - 1 ? 'text-blue-500' : 'text-gray-300'
   271	                }`}
   272	              >
   273	                {label}
   274	              </span>
   275	            ))}
   276	          </div>
   277	        </div>
   278	      </div>
   279	
   280	      {/* Main content */}
   281	      <main className="flex-1 max-w-lg mx-auto w-full px-4 py-6 pb-32">
   282	        {step === 1 && (
   283	          <div className="space-y-6">
   284	            <div>
   285	              <h2 className="text-xl font-bold text-gray-900">Personal Details</h2>
   286	              <p className="text-sm text-gray-500 mt-1">Tell us about yourself to get started</p>
   287	            </div>
   288	
   289	            <Card>
   290	              <CardContent className="p-5 space-y-4">
   291	                <div className="space-y-2">
   292	                  <Label htmlFor="onb-name" className="text-sm font-medium">Full Name *</Label>
   293	                  <Input
   294	                    id="onb-name"
   295	                    placeholder="Enter your full name"
   296	                    value={name}
   297	                    onChange={(e) => setName(e.target.value)}
   298	                    className="h-10"
   299	                  />
   300	                </div>
   301	
   302	                <div className="space-y-2">
   303	                  <Label htmlFor="onb-email" className="text-sm font-medium">Email Address *</Label>
   304	                  <Input
   305	                    id="onb-email"
   306	                    type="email"
   307	                    placeholder="you@example.com"
   308	                    value={email}
   309	                    onChange={(e) => setEmail(e.target.value)}
   310	                    className="h-10"
   311	                  />
   312	                </div>
   313	
   314	                <div className="space-y-2">
   315	                  <Label htmlFor="onb-phone" className="text-sm font-medium">Phone Number</Label>
   316	                  <Input
   317	                    id="onb-phone"
   318	                    value={phone}
   319	                    disabled
   320	                    className="h-10 bg-gray-50 text-gray-500"
   321	                  />
   322	                  <p className="text-[11px] text-gray-400">Verified via OTP during login</p>
   323	                </div>
   324	
   325	                <div className="space-y-2">
   326	                  <Label htmlFor="onb-qualifications" className="text-sm font-medium">Qualifications *</Label>
   327	                  <Input
   328	                    id="onb-qualifications"
   329	                    placeholder="e.g. B.Sc Nursing, GNM Diploma"
   330	                    value={qualifications}
   331	                    onChange={(e) => setQualifications(e.target.value)}
   332	                    className="h-10"
   333	                  />
   334	                </div>
   335	
   336	                <div className="space-y-2">
   337	                  <Label htmlFor="onb-experience" className="text-sm font-medium">Years of Experience *</Label>
   338	                  <Input
   339	                    id="onb-experience"
   340	                    type="number"
   341	                    placeholder="e.g. 3"
   342	                    value={experience}
   343	                    onChange={(e) => setExperience(e.target.value)}
   344	                    className="h-10"
   345	                    min="0"
   346	                    max="50"
   347	                  />
   348	                </div>
   349	
   350	                <div className="space-y-2">
   351	                  <Label htmlFor="onb-languages" className="text-sm font-medium">Languages Known</Label>
   352	                  <Input
   353	                    id="onb-languages"
   354	                    placeholder="e.g. English, Hindi, Tamil"
   355	                    value={languages}
   356	                    onChange={(e) => setLanguages(e.target.value)}
   357	                    className="h-10"
   358	                  />
   359	                </div>
   360	
   361	                <div className="space-y-2">
   362	                  <Label htmlFor="onb-bio" className="text-sm font-medium">About Yourself</Label>
   363	                  <Textarea
   364	                    id="onb-bio"
   365	                    placeholder="Brief description about your care experience and approach..."
   366	                    value={bio}
   367	                    onChange={(e) => setBio(e.target.value)}
   368	                    className="min-h-[80px] text-sm"
   369	                  />
   370	                </div>
   371	              </CardContent>
   372	            </Card>
   373	          </div>
   374	        )}
   375	
   376	        {step === 2 && (
   377	          <div className="space-y-6">
   378	            <div>
   379	              <h2 className="text-xl font-bold text-gray-900">Specialist Type</h2>
   380	              <p className="text-sm text-gray-500 mt-1">Choose your primary area of care expertise</p>
   381	            </div>
   382	
   383	            <div className="space-y-3">
   384	              {/* Elder Care Option */}
   385	              <button
   386	                type="button"
   387	                onClick={() => setSpecialty('elder_care')}
   388	                className={`w-full rounded-xl border-2 p-5 text-left transition-all ${
   389	                  specialty === 'elder_care'
   390	                    ? 'border-blue-600 bg-blue-50/50 shadow-md ring-2 ring-blue-100'
   391	                    : 'border-gray-200 bg-white hover:border-blue-300 hover:shadow-sm'
   392	                }`}
   393	              >
   394	                <div className="flex items-start gap-4">
   395	                  <div className={`size-14 rounded-xl flex items-center justify-center shrink-0 ${
   396	                    specialty === 'elder_care' ? 'bg-blue-100' : 'bg-gray-100'
   397	                  }`}>
   398	                    <Heart className={`size-7 ${specialty === 'elder_care' ? 'text-blue-600' : 'text-gray-400'}`} />
   399	                  </div>
   400	                  <div className="flex-1">
   401	                    <div className="flex items-center justify-between">
   402	                      <h3 className="text-base font-semibold text-gray-900">Elder Care</h3>
   403	                      {specialty === 'elder_care' && (
   404	                        <CheckCircle className="size-5 text-blue-600" />
   405	                      )}
   406	                    </div>
   407	                    <p className="text-sm text-gray-500 mt-1">
   408	                      Dementia care, geriatric nursing, mobility support, palliative care, and companionship for seniors
   409	                    </p>
   410	                    <div className="flex flex-wrap gap-1.5 mt-2">
   411	                      <Badge variant="outline" className="text-[10px] text-gray-500">Geriatric</Badge>
   412	                      <Badge variant="outline" className="text-[10px] text-gray-500">Dementia</Badge>
   413	                      <Badge variant="outline" className="text-[10px] text-gray-500">Palliative</Badge>
   414	                      <Badge variant="outline" className="text-[10px] text-gray-500">Physiotherapy</Badge>
   415	                    </div>
   416	                  </div>
   417	                </div>
   418	              </button>
   419	
   420	              {/* Child Care Option */}
   421	              <button
   422	                type="button"
   423	                onClick={() => setSpecialty('child_care')}
   424	                className={`w-full rounded-xl border-2 p-5 text-left transition-all ${
   425	                  specialty === 'child_care'
   426	                    ? 'border-blue-600 bg-blue-50/50 shadow-md ring-2 ring-blue-100'
   427	                    : 'border-gray-200 bg-white hover:border-blue-300 hover:shadow-sm'
   428	                }`}
   429	              >
   430	                <div className="flex items-start gap-4">
   431	                  <div className={`size-14 rounded-xl flex items-center justify-center shrink-0 ${
   432	                    specialty === 'child_care' ? 'bg-blue-100' : 'bg-gray-100'
   433	                  }`}>
   434	                    <Baby className={`size-7 ${specialty === 'child_care' ? 'text-blue-600' : 'text-gray-400'}`} />
   435	                  </div>
   436	                  <div className="flex-1">
   437	                    <div className="flex items-center justify-between">
   438	                      <h3 className="text-base font-semibold text-gray-900">Child Care</h3>
   439	                      {specialty === 'child_care' && (
   440	                        <CheckCircle className="size-5 text-blue-600" />
   441	                      )}
   442	                    </div>
   443	                    <p className="text-sm text-gray-500 mt-1">
   444	                      Newborn care, child development, special needs support, feeding, and day care services
   445	                    </p>
   446	                    <div className="flex flex-wrap gap-1.5 mt-2">
   447	                      <Badge variant="outline" className="text-[10px] text-gray-500">Newborn</Badge>
   448	                      <Badge variant="outline" className="text-[10px] text-gray-500">Infant</Badge>
   449	                      <Badge variant="outline" className="text-[10px] text-gray-500">Special Needs</Badge>
   450	                      <Badge variant="outline" className="text-[10px] text-gray-500">Development</Badge>
   451	                    </div>
   452	                  </div>
   453	                </div>
   454	              </button>
   455	            </div>
   456	          </div>
   457	        )}
   458	
   459	        {step === 3 && (
   460	          <div className="space-y-6">
   461	            <div>
   462	              <h2 className="text-xl font-bold text-gray-900">Skills & Services</h2>
   463	              <p className="text-sm text-gray-500 mt-1">Select your core skills and services you can offer</p>
   464	            </div>
   465	
   466	            {/* Skills Section */}
   467	            <Card>
   468	              <CardHeader className="pb-3">
   469	                <div className="flex items-center justify-between">
   470	                  <CardTitle className="text-sm font-semibold text-gray-900 flex items-center gap-2">
   471	                    <Sparkles className="size-4 text-gray-400" />
   472	                    Core Skills
   473	                    {selectedSkills.length > 0 && (
   474	                      <Badge variant="outline" className="text-[10px] text-gray-500 border-gray-200 bg-gray-50 px-1.5 py-0">
   475	                        {selectedSkills.length} selected
   476	                      </Badge>
   477	                    )}
   478	                  </CardTitle>
   479	                </div>
   480	              </CardHeader>
   481	              <CardContent className="p-5 pt-0">
   482	                {selectedSkills.length > 0 && (
   483	                  <div className="flex flex-wrap gap-1.5 mb-4">
   484	                    {selectedSkills.map((skill) => (
   485	                      <div key={skill.name} className="flex items-center gap-1 rounded-lg border border-blue-200 bg-blue-50 px-2 py-1">
   486	                        <BadgeCheck className="size-3 text-blue-600" />
   487	                        <span className="text-[11px] font-medium text-blue-700">{skill.name}</span>
   488	                        <button type="button" onClick={() => toggleSkill(skill.name)} className="ml-0.5">
   489	                          <Minus className="size-3 text-blue-400 hover:text-red-500" />
   490	                        </button>
   491	                      </div>
   492	                    ))}
   493	                  </div>
   494	                )}
   495	
   496	                {Object.entries(skillsByCategory).map(([category, skills]) => {
   497	                  const unselected = skills.filter((s) => !isSkillSelected(s.name));
   498	                  if (unselected.length === 0) return null;
   499	                  return (
   500	                    <div key={category} className="mb-4 last:mb-0">
   501	                      <p className="text-[11px] font-semibold text-gray-400 uppercase tracking-wider mb-2">{category}</p>
   502	                      <div className="flex flex-wrap gap-1.5">
   503	                        {unselected.map((skill) => (
   504	                          <button
   505	                            key={skill.name}
   506	                            type="button"
   507	                            onClick={() => toggleSkill(skill.name)}
   508	                            className="rounded-lg border border-gray-200 bg-white px-2.5 py-1.5 text-[11px] text-gray-600 hover:border-blue-300 hover:text-blue-600 hover:bg-blue-50 transition-colors"
   509	                          >
   510	                            <Plus className="size-2.5 inline mr-0.5 -mt-0.5" />
