    48→    48→
    49→    49→function getAvatarColor(name: string): AvatarColor {
    50→    50→  let hash = 0;
    51→    51→  for (let i = 0; i < name.length; i++) {
    52→    52→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
    53→    53→  }
    54→    54→  return avatarColors[Math.abs(hash) % avatarColors.length];
    55→    55→}

    57→    57→function getInitials(name: string): string {

    61→    61→    .join('')

    64→    64→}

    66→    66→function formatDate(dateStr: string): string {

    69→    69→    day: 'numeric',
    70→    70→    month: 'short',
    71→    71→    year: 'numeric',
    72→    72→  });
