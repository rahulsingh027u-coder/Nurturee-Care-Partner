    36→    36→// ── Skin-tone face detection (pure canvas, works on all browsers) ────────
    37→    37→function detectSkinInOval(
    38→    38→  video: HTMLVideoElement,
    39→    39→  canvas: HTMLCanvasElement,
    40→    40→): boolean {
    41→    41→  const vw = video.videoWidth || 480;
    42→    42→  const vh = video.videoHeight || 640;
    43→    43→  canvas.width = vw;
    44→    44→  canvas.height = vh;
    45→    45→  const ctx = canvas.getContext('2d');
    46→    46→  if (!ctx) return false;
    47→    47→
    48→    48→  ctx.drawImage(video, 0, 0, vw, vh);
    49→    49→
    50→    50→  // Sample the center oval region (face area)
    51→    51→  const cx = vw / 2;
    52→    52→  const cy = vh / 2;
    53→    53→  const rx = vw * 0.2;  // oval horizontal radius
    54→    54→  const ry = vh * 0.28; // oval vertical radius
    55→    55→  const step = 8;        // sample every 8px for speed
    56→    56→
    57→    57→  let skinPixels = 0;
    58→    58→  let totalPixels = 0;
    59→    59→
    60→    60→  for (let y = cy - ry; y <= cy + ry; y += step) {
    61→    61→    for (let x = cx - rx; x <= cx + rx; x += step) {
    62→    62→      // Check if inside ellipse: ((x-cx)/rx)^2 + ((y-cy)/ry)^2 <= 1
    63→    63→      const dx = (x - cx) / rx;
    64→    64→      const dy = (y - cy) / ry;
    65→    65→      if (dx * dx + dy * dy > 1) continue;
    66→    66→
    67→    67→      totalPixels++;
    68→    68→      const pixel = ctx.getImageData(Math.floor(x), Math.floor(y), 1, 1).data;
    69→    69→      const r = pixel[0], g = pixel[1], b = pixel[2];
    70→    70→
    71→    71→      // Skin-tone detection in RGB space (covers wide range of skin tones)
    72→    72→      // Rules: R > G > B, R between 60-255, G between 30-220, B between 15-180
    73→    73→      // And logical skin-tone constraints based on chrominance
    74→    74→      if (
    75→    75→        r > 60 && g > 30 && b > 15 &&
    76→    76→        r > g && r > b &&
    77→    77→        (r - g) > 10 &&
    78→    78→        (r - b) > 15 &&
    79→    79→        Math.abs(r - g) < 100 &&
    80→    80→        b < r * 0.75 &&
    81→    81→        g < r * 0.9
    82→    82→      ) {
    83→    83→        skinPixels++;
    84→    84→      }
    85→    85→    }
    86→    86→  }
    87→    87→
    88→    88→  // Need at least 30% skin-tone pixels in the oval to consider face detected
    89→    89→  return totalPixels > 0 && (skinPixels / totalPixels) > 0.30;
    90→    90→}

   116→   116→function formatTime(time24: string): string {
   117→   117→  const [hours, minutes] = time24.split(':').map(Number);
   118→   118→  const period = hours >= 12 ? 'PM' : 'AM';
   119→   119→  const hours12 = hours % 12 || 12;
   120→   120→  return `${hours12}:${minutes.toString().padStart(2, '0')} ${period}`;
   121→   121→}
