    58→function formatCurrency(amount: number): string {
    59→  return '₹' + amount.toLocaleString('en-IN');
    60→}
    61→
    62→function getGreeting(): string {
    63→  const hour = new Date().getHours();
    64→  if (hour < 12) return 'Good Morning';
    65→  if (hour < 17) return 'Good Afternoon';
    66→  return 'Good Evening';
    67→}
    68→
    69→function formatDate(): string {
    70→  const now = new Date();
    71→  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
    72→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
    73→  return `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
    74→}
