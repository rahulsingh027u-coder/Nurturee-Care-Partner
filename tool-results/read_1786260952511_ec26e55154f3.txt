   181→
   182→function loadDocsFromStorage(): VerificationDoc[] {
   183→  try {
   184→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   185→    if (stored) {
   186→      const parsed = JSON.parse(stored);
   187→      if (Array.isArray(parsed) && parsed.length === 4) return parsed;
   188→    }
   189→  } catch {
   190→    // fall through
   191→  }
   192→  return getInitialDocs();
   193→}
   194→
   195→function saveDocsToStorage(docs: VerificationDoc[]) {
   196→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   197→}
   198→
   199→function getStatusDisplay(status: VerificationStatus) {
   200→  switch (status) {
   201→    case 'approved':
   202→      return { label: 'Verified', color: 'text-emerald-700', bg: 'bg-emerald-100', badgeBg: 'bg-emerald-50 border-emerald-200', icon: CheckCircle };
   203→    case 'submitted':
   204→      return { label: 'Submitted', color: 'text-blue-700', bg: 'bg-blue-100', badgeBg: 'bg-blue-50 border-blue-200', icon: Send };
   205→    case 'under_review':
   206→      return { label: 'Under Review', color: 'text-blue-700', bg: 'bg-blue-100', badgeBg: 'bg-blue-50 border-blue-200', icon: RefreshCw };
   207→    case 'rejected':
   208→      return { label: 'Rejected', color: 'text-red-700', bg: 'bg-red-100', badgeBg: 'bg-red-50 border-red-200', icon: XCircle };
   209→    case 'max_attempts_reached':
   210→      return { label: 'Locked', color: 'text-gray-600', bg: 'bg-gray-100', badgeBg: 'bg-gray-50 border-gray-200', icon: Ban };
   211→    default:
   212→      return { label: 'Pending', color: 'text-amber-700', bg: 'bg-amber-100', badgeBg: 'bg-amber-50 border-amber-200', icon: Clock };
   213→  }

   217→
   218→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   219→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   220→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   221→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   222→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   223→  const [submitting, setSubmitting] = useState(false);
   224→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   225→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   226→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   227→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   228→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   229→
   230→  const currentYear = new Date().getFullYear();
   231→
   232→  // Clean up timers on unmount
   233→  useEffect(() => {
   234→    return () => {
   235→      Object.values(reviewTimersRef.current).forEach((timers) => {
   236→        timers.forEach(clearTimeout);
   237→      });
   238→    };
   239→  }, []);
   240→
   241→  // Persist docs to localStorage whenever they change
   242→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   243→    setDocs((prev) => {
   244→      const updated = updater(prev);
   245→      saveDocsToStorage(updated);
   246→      return updated;
   247→    });
   248→  }, []);
   249→
   250→  // ─── Backend Review Simulation ───────────────────────────────────────────
   251→
   252→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   253→    // Clear any existing timers for this type
   254→    if (reviewTimersRef.current[type]) {
   255→      reviewTimersRef.current[type].forEach(clearTimeout);
   256→    }
   257→    reviewTimersRef.current[type] = [];
   258→
   259→    const doc = docs.find((d) => d.type === type);
   260→    if (!doc) return;
   261→
   262→    // Timer 1: After 3s → under_review
   263→    const t1 = setTimeout(() => {
   264→      setReviewingType(type);
   265→      updateDocs((prev) =>
   266→        prev.map((d) =>
   267→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   268→        )
   269→      );
   270→    }, 3000);
   271→
   272→    // Timer 2: After 5-7s total → resolve
   273→    const resolveDelay = 5000 + Math.random() * 2000;
   274→    const t2 = setTimeout(() => {
   275→      const currentDoc = docs.find((d) => d.type === type);
   276→      if (!currentDoc) return;
   277→
   278→      const currentAttempts = currentDoc.attempts.length;
   279→      let willApprove: boolean;
   280→
   281→      if (type === 'aadhaar') {
   282→        // Aadhaar always approve on first attempt (demo-friendly)
   283→        willApprove = true;
   284→      } else {
   285→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   286→        if (attemptNumber === 1) {
   287→          willApprove = Math.random() < 0.4;
   288→        } else if (attemptNumber === 2) {
   289→          willApprove = Math.random() < 0.7;
   290→        } else {
   291→          willApprove = true;
   292→        }
   293→      }
   294→
   295→      if (willApprove) {
   296→        // Approve
   297→        setApprovingType(type);
   298→        updateDocs((prev) =>
   299→          prev.map((d) => {
   300→            if (d.type !== type) return d;
   301→            return {
   302→              ...d,
   303→              status: 'approved' as VerificationStatus,
   304→              attempts: d.attempts.map((a) =>
   305→                a.attemptNumber === attemptNumber
   306→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   307→                  : a
   308→              ),
   309→            };
   310→          })
   311→        );
   312→        setReviewingType(null);
   313→
   314→        // If Aadhaar approved and onComplete exists, call it after delay
   315→        if (type === 'aadhaar' && onComplete) {
   316→          setTimeout(() => {
   317→            setApprovingType(null);
   318→            setSubmittedType(null);
   319→            onComplete();
   320→          }, 1000);
   321→        } else {
   322→          // For optional: show success briefly, then allow going back
   323→          setTimeout(() => {
   324→            setApprovingType(null);
   325→            setSubmittedType(null);
   326→            setActiveType(null);
   327→          }, 2000);
   328→        }
   329→      } else {
   330→        // Reject
   331→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   332→        updateDocs((prev) =>
   333→          prev.map((d) => {
   334→            if (d.type !== type) return d;
   335→            const updatedAttempts = d.attempts.map((a) =>
   336→              a.attemptNumber === attemptNumber
   337→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   338→                : a
   339→            );
   340→            // Check if max attempts reached
   341→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   342→            return {
   343→              ...d,
   344→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   345→              attempts: updatedAttempts,
   346→              currentDraft: { documents: [], formFields: {} },
   347→            };
   348→          })
   349→        );
   350→        setReviewingType(null);
   351→        // Transition from submitted view to allow re-upload
   352→        setTimeout(() => {
   353→          setSubmittedType(null);
   354→        }, 300);
   355→      }
   356→    }, resolveDelay);
   357→
   358→    reviewTimersRef.current[type] = [t1, t2];
   359→  }, [docs, onComplete, updateDocs]);
   360→
   361→  // ─── Handlers ────────────────────────────────────────────────────────────
   362→
   363→  const handleSelectType = (type: VerificationType) => {
   364→    const doc = docs.find((d) => d.type === type);
   365→    if (!doc) return;
   366→    // If approved or max_attempts_reached, don't open form (but allow viewing)
   367→    setActiveType(type);
   368→    // Restore draft if exists, otherwise use empty
   369→    setFormFields(doc.currentDraft?.formFields || {});
   370→    setUploadedFiles(doc.currentDraft?.documents || []);
   371→    setSubmittedType(null);
   372→  };
   373→
   374→  const handleBack = () => {
   375→    // Save draft
   376→    if (activeType) {
   377→      updateDocs((prev) =>
   378→        prev.map((d) =>
   379→          d.type === activeType
   380→            ? { ...d, currentDraft: { documents: uploadedFiles, formFields } }
   381→            : d
   382→        )
   383→      );
   384→    }
   385→    // Clean up timers for this type if we navigate back
   386→    if (activeType && reviewTimersRef.current[activeType]) {
   387→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   388→      if (!submittedType && !reviewingType) {
   389→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   390→        delete reviewTimersRef.current[activeType];
   391→      }
   392→    }
   393→    setActiveType(null);
   394→    setFormFields({});
   395→    setUploadedFiles([]);
   396→    setSubmittedType(null);
   397→    setReviewingType(null);
   398→  };
   399→
   400→  const handleFileUpload = () => {
   401→    const mockFile = {
   402→      id: `file-${Date.now()}`,
   403→      name: 'document.jpg',
   404→      size: 245000,
   405→      type: 'image/jpeg',
   406→    };
   407→    setUploadedFiles((prev) => [...prev, mockFile]);
   408→  };
   409→
   410→  const handleRemoveFile = (fileId: string) => {
   411→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   412→  };
   413→
   414→  const handleSubmit = async () => {
   415→    if (!activeType) return;
   416→    setSubmitting(true);
   417→
   418→    // Simulate submission delay
   419→    await new Promise((resolve) => setTimeout(resolve, 1500));
   420→
   421→    const doc = docs.find((d) => d.type === activeType);
   422→    const nextAttemptNumber = (doc?.attempts?.length || 0) + 1;
   423→
   424→    const newAttempt = {
   425→      attemptNumber: nextAttemptNumber,
   426→      submittedAt: new Date().toISOString(),
   427→      documents: [...uploadedFiles],
   428→      formFields: { ...formFields },
   429→      status: 'submitted' as const,
   430→    };
   431→
   432→    updateDocs((prev) =>
   433→      prev.map((d) => {
   434→        if (d.type !== activeType) return d;
   435→        return {
   436→          ...d,
   437→          status: 'submitted' as VerificationStatus,
   438→          attempts: [...d.attempts, newAttempt],
   439→          currentDraft: { documents: [], formFields: {} },
   440→        };
   441→      })
   442→    );
   443→
   444→    setSubmitting(false);
   445→    setSubmittedType(activeType);
   446→
   447→    // Start the backend review simulation
   448→    startReviewSimulation(activeType, nextAttemptNumber);
   449→  };
   450→
   451→  // ─── Derived State ────────────────────────────────────────────────────────
   452→
   453→  const aadhaarDoc = docs.find((d) => d.type === 'aadhaar');
   454→  const aadhaarApproved = aadhaarDoc?.status === 'approved';
   455→  const showOptionalSection = aadhaarApproved || optionalMode;
   456→
   457→  // ─── View 1: Overview ────────────────────────────────────────────────────
   458→
   459→  if (!activeType) {
   460→    const optionalTypes = (Object.keys(verificationConfig) as VerificationType[]).filter(
   461→      (type) => type !== 'aadhaar'
   462→    );
   463→
   464→    return (
   465→      <div className="min-h-screen bg-gray-50 flex flex-col">
   466→        <div className="flex-1 flex flex-col items-center justify-center p-4">
   467→          <div className="w-full max-w-2xl space-y-6">
   468→            {/* Header */}
   469→            <div className="text-center space-y-2">
   470→              <div className="inline-flex items-center justify-center size-16 rounded-2xl bg-gradient-to-br from-blue-600 to-indigo-600 text-white mb-2">
   471→                <Shield className="size-8" />
   472→              </div>
   473→              <h1 className="text-2xl font-bold text-gray-900">Document Verification</h1>
   474→              <p className="text-sm text-gray-500 max-w-md mx-auto">
   475→                Complete your verification to start accepting assignments. Aadhaar verification is mandatory.
   476→              </p>
   477→            </div>
   478→
   479→            {/* Mandatory Section */}
   480→            {aadhaarDoc && aadhaarDoc.status !== 'approved' && (
   481→              <Card className="border-blue-200 bg-blue-50/30">
   482→                <CardContent className="p-4 flex items-center gap-3">
   483→                  <div className="size-10 rounded-lg bg-blue-100 flex items-center justify-center shrink-0">
   484→                    <Shield className="size-5 text-blue-600" />
   485→                  </div>
   486→                  <div className="flex-1">
   487→                    <p className="text-sm font-semibold text-blue-900">Aadhaar Verification — Required</p>
   488→                    <p className="text-xs text-blue-600">
   489→                      {aadhaarDoc.status === 'max_attempts_reached'
   490→                        ? 'Maximum attempts reached. Contact support.'
   491→                        : 'You must complete Aadhaar verification to access the portal.'}
   492→                    </p>
   493→                  </div>
   494→                  {aadhaarDoc.status !== 'max_attempts_reached' && (
   495→                    <Button
   496→                      size="sm"
   497→                      className="bg-blue-600 hover:bg-blue-700 text-white"
   498→                      onClick={() => handleSelectType('aadhaar')}
   499→                    >
   500→                      Verify Now
   501→                    </Button>
   502→                  )}
   503→                </CardContent>
   504→              </Card>
   505→            )}
   506→
   507→            {/* Optional Section */}
   508→            {showOptionalSection && (
   509→              <div className="space-y-3">
   510→                <div className="flex items-center gap-2">
   511→                  <span className="text-sm font-semibold text-gray-700">Additional Verifications</span>
   512→                  <Badge variant="outline" className="text-[10px] text-gray-500 border-gray-300 bg-gray-50">
   513→                    Optional
   514→                  </Badge>
   515→                </div>
   516→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   517→                  {optionalTypes.map((type) => {
   518→                    const config = verificationConfig[type];
   519→                    const Icon = config.icon;
   520→                    const doc = docs.find((d) => d.type === type);
   521→                    const statusDisplay = getStatusDisplay(doc?.status || 'not_started');
   522→                    const StatusIcon = statusDisplay.icon;
   523→                    const isReviewing = reviewingType === type;
   524→
   525→                    return (
   526→                      <button
   527→                        key={type}
   528→                        onClick={() => handleSelectType(type)}
   529→                        className={`rounded-xl border border-gray-100 bg-white p-4 flex items-center gap-3 text-left hover:shadow-sm hover:border-gray-200 transition-all cursor-pointer w-full group ${
   530→                          doc?.status === 'approved' ? 'opacity-70' : ''
   531→                        }`}
   532→                      >
   533→                        <div className={`size-10 rounded-lg flex items-center justify-center ${statusDisplay.bg}`}>
   534→                          {isReviewing ? (
   535→                            <Loader2 className={`size-5 ${statusDisplay.color} animate-spin`} />
   536→                          ) : (
   537→                            <Icon className={`size-5 ${statusDisplay.color}`} />
   538→                          )}
   539→                        </div>
   540→                        <div className="flex-1 min-w-0">
   541→                          <p className="text-sm font-medium text-gray-900 group-hover:text-blue-700 transition-colors">
   542→                            {config.title}
   543→                          </p>

   554→                </div>
   555→              </div>

   557→          </div>
   558→        </div>
   559→
   560→        {/* Sticky Bottom Bar */}
   561→        {aadhaarApproved && (onComplete || optionalMode) && (

   563→            <div className="w-full max-w-2xl mx-auto">

   566→                  className="w-full bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-700 hover:to-indigo-700 text-white h-12 text-sm font-medium rounded-xl shadow-md"
   567→                  onClick={onComplete}

   569→                  <ArrowLeft className="size-4 mr-2" />
   570→                  Back to Profile
   571→                </Button>
   572→              ) : (
   573→                <Button
   574→                  className="w-full bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-700 hover:to-indigo-700 text-white h-12 text-sm font-medium rounded-xl shadow-md"
   575→                  onClick={onComplete}
   576→                >
   577→                  <Sparkles className="size-4 mr-2" />
   578→                  Go to Dashboard
   579→                </Button>
   580→              )}
   581→            </div>
   582→          </div>
   583→        )}
   584→      </div>
   585→    );
   586→  }
   587→
   588→  // ─── View 2: Upload Form ────────────────────────────────────────────────
   589→
   590→  const config = verificationConfig[activeType];
   591→  const Icon = config.icon;
   592→  const doc = docs.find((d) => d.type === activeType);
   593→  const attemptsUsed = doc?.attempts?.length || 0;
   594→  const nextAttempt = attemptsUsed + 1;
   595→  const isSubmitted = submittedType === activeType;
   596→  const isReviewing = reviewingType === activeType;
   597→  const isApproved = approvingType === activeType;
   598→  const isLocked = doc?.status === 'max_attempts_reached';
   599→  const isRejected = doc?.status === 'rejected';
   600→  const lastRejectionReason = doc?.attempts?.length
   601→    ? doc.attempts[doc.attempts.length - 1]?.rejectionReason
   602→    : null;
   603→
   604→  const allFieldsFilled = config.fields
   605→    .filter((f) => f.required)
   606→    .every((f) => formFields[f.key]?.trim());
   607→
   608→  const isSubmitDisabled = !allFieldsFilled || uploadedFiles.length === 0 || submitting || isSubmitted || isReviewing || isLocked || isApproved;
   609→
   610→  return (
   611→    <div className="min-h-screen bg-gray-50">
   612→      {/* Sticky Top Bar */}
   613→      <div className="sticky top-0 z-30 bg-white border-b border-gray-200">
   614→        <div className="max-w-2xl mx-auto flex items-center gap-3 px-4 py-3">
   615→          <button
   616→            onClick={handleBack}
   617→            className="flex items-center gap-1.5 text-sm text-gray-600 hover:text-gray-900 transition-colors"
   618→          >
   619→            <ArrowLeft className="size-4" />
   620→            Back
   621→          </button>
   622→          <div className="flex-1 flex items-center gap-2">
   623→            <Icon className="size-5 text-blue-600" />
   624→            <h1 className="text-sm font-semibold text-gray-900">{config.title}</h1>
   625→          </div>
   626→          {config.mandatory && (
   627→            <Badge className="bg-red-50 text-red-600 border border-red-200 text-[10px]">Required</Badge>
   628→          )}
   629→        </div>
   630→      </div>
   631→
   632→      <div className="max-w-2xl mx-auto p-4 space-y-5 pb-8">
   633→        {/* Approved State */}
   634→        {isApproved && (
   635→          <Card className="border-emerald-200 bg-emerald-50/50">
   636→            <CardContent className="p-6 text-center">
   637→              <div className="inline-flex items-center justify-center size-12 rounded-full bg-emerald-100 mb-3">
   638→                <CheckCircle className="size-6 text-emerald-600" />
   639→              </div>
   640→              <h2 className="text-lg font-bold text-emerald-900">Verification Approved!</h2>
   641→              <p className="text-sm text-emerald-600 mt-1">
   642→                {activeType === 'aadhaar'
   643→                  ? 'Redirecting to your dashboard...'
   644→                  : 'Your document has been verified successfully.'}
   645→              </p>
   646→              {activeType !== 'aadhaar' && (
   647→                <div className="mt-3">
   648→                  <Loader2 className="size-4 text-emerald-500 animate-spin mx-auto" />
   649→                </div>
   650→              )}
   651→            </CardContent>
   652→          </Card>
   653→        )}
   654→
   655→        {/* Submitted State (waiting for review) */}
   656→        {isSubmitted && !isApproved && (
   657→          <Card className="border-blue-200 bg-blue-50/50">
   658→            <CardContent className="p-6 text-center">
   659→              <div className="inline-flex items-center justify-center size-12 rounded-full bg-blue-100 mb-3">
   660→                <Send className="size-6 text-blue-600" />
   661→              </div>
   662→              <h2 className="text-lg font-bold text-blue-900">Document Submitted!</h2>
   663→              <div className="flex items-center justify-center gap-2 mt-2">
   664→                <Loader2 className="size-4 text-blue-500 animate-spin" />
   665→                <p className="text-sm text-blue-600">Under Review...</p>
   666→              </div>
   667→              <p className="text-xs text-blue-400 mt-1">This usually takes a few seconds in demo mode</p>
   668→            </CardContent>
   669→          </Card>
   670→        )}
   671→
   672→        {/* Locked State (max attempts reached) */}
   673→        {isLocked && !isSubmitted && (
   674→          <Card className="border-red-200 bg-red-50/50">
   675→            <CardContent className="p-6 text-center">
   676→              <div className="inline-flex items-center justify-center size-12 rounded-full bg-red-100 mb-3">
   677→                <Ban className="size-6 text-red-600" />
   678→              </div>
   679→              <h2 className="text-lg font-bold text-red-900">Maximum Attempts Reached</h2>
   680→              <p className="text-sm text-red-600 mt-2">
   681→                You have used all 3 verification attempts for this document. Please contact support for assistance.
   682→              </p>
   683→              <Separator className="my-4" />
   684→              <div className="flex items-center justify-center gap-2 text-xs text-red-500">
   685→                <AlertCircle className="size-3.5" />
   686→                <span>Contact Nurturee support at support@nurturee.com</span>
   687→              </div>
   688→            </CardContent>
   689→          </Card>
   690→        )}
   691→
   692→        {/* Rejection Alert */}
   693→        {isRejected && !isSubmitted && !isLocked && (
   694→          <Card className="border-red-200 bg-red-50/50">
   695→            <CardContent className="p-4">
   696→              <div className="flex items-start gap-3">
   697→                <div className="size-8 rounded-full bg-red-100 flex items-center justify-center shrink-0 mt-0.5">
   698→                  <XCircle className="size-4 text-red-600" />
   699→                </div>
   700→                <div className="flex-1">
   701→                  <p className="text-sm font-semibold text-red-900">Verification Rejected</p>
   702→                  <p className="text-sm text-red-700 mt-1">{lastRejectionReason}</p>
   703→                  <p className="text-xs text-red-500 mt-2 font-medium">
   704→                    Re-upload (Attempt {nextAttempt} of {MAX_ATTEMPTS})
   705→                  </p>
   706→                </div>
   707→              </div>
   708→            </CardContent>
   709→          </Card>
   710→        )}
   711→
   712→        {/* Description Card with Attempt Tracker */}
   713→        {!isLocked && !isApproved && (
   714→          <Card>
   715→            <CardContent className="p-4">
   716→              <p className="text-sm text-gray-600">{config.description}</p>
   717→              <div className="flex items-center gap-2 mt-3">
   718→                {Array.from({ length: MAX_ATTEMPTS }).map((_, i) => (
   719→                  <div
   720→                    key={i}
   721→                    className={`size-2.5 rounded-full transition-colors ${
   722→                      i < attemptsUsed
   723→                        ? 'bg-blue-500'
   724→                        : 'bg-gray-200'
   725→                    }`}
   726→                  />
   727→                ))}
   728→                <span className="text-xs text-gray-500 ml-1">
   729→                  {attemptsUsed === 0
   730→                    ? 'No attempts used'
   731→                    : `${attemptsUsed} of ${MAX_ATTEMPTS} attempts used`}

   733→              </div>
   734→            </CardContent>
   735→          </Card>
   736→        )}
   737→
   738→        {/* Form Fields — only when not locked and not in approved state */}
   739→        {!isLocked && !isApproved && (

   750→                  const parsedDate = dateValue ? (() => {
   751→                    const parts = dateValue.split('/');
   752→                    if (parts.length === 3) {
   753→                      const [day, month, year] = parts.map(Number);
   754→                      if (!isNaN(day) && !isNaN(month) && !isNaN(year)) {

   759→                  })() : undefined;
   760→
   761→                  return (
   762→                    <div key={field.key} className="space-y-2">
   763→                      <Label className="text-xs font-medium text-gray-500">

   765→                      </Label>
   766→                      <Popover
   767→                        open={calendarOpen[field.key] || false}
   768→                        onOpenChange={(open) => setCalendarOpen((prev) => ({ ...prev, [field.key]: open }))}
   769→                      >
   770→                        <PopoverTrigger asChild>
   771→                          <Button
   772→                            variant="outline"
   773→                            className="w-full justify-start text-left font-normal text-sm"
   774→                            disabled={isSubmitted || isReviewing}
   775→                          >
   776→                            <CalendarIcon className="mr-2 h-4 w-4" />
   777→                            {dateValue || `Select ${field.label}`}
   778→                          </Button>
   779→                        </PopoverTrigger>
   780→                        <PopoverContent className="w-auto p-0" align="start">
   781→                          <Calendar
   782→                            mode="single"
   783→                            selected={parsedDate}
   784→                            onSelect={(date) => {
   785→                              if (date) {
   786→                                const formatted = format(date, 'dd/MM/yyyy');
   787→                                setFormFields((prev) => ({ ...prev, [field.key]: formatted }));
   788→                                setCalendarOpen((prev) => ({ ...prev, [field.key]: false }));
   789→                              }
   790→                            }}
   791→                            captionLayout="dropdown"
   792→                            fromYear={1940}
   793→                            toYear={currentYear}
   794→                            defaultMonth={parsedDate || new Date(currentYear - 30, 0)}
   795→                          />
   796→                        </PopoverContent>
   797→                      </Popover>
   798→                    </div>
   799→                  );
   800→                }
   801→
   802→                if (field.type === 'select') {
   803→                  return (
   804→                    <div key={field.key} className="space-y-2">
   805→                      <Label className="text-xs font-medium text-gray-500">
   806→                        {field.label} {field.required && <span className="text-red-500">*</span>}
   807→                      </Label>
   808→                      <select
   809→                        value={formFields[field.key] || ''}
   810→                        onChange={(e) => setFormFields((prev) => ({ ...prev, [field.key]: e.target.value }))}
   811→                        disabled={isSubmitted || isReviewing}
   812→                        className="w-full rounded-lg border border-gray-200 bg-white px-3 py-2 text-sm text-gray-900 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent disabled:opacity-50 disabled:cursor-not-allowed"
   813→                      >
   814→                        <option value="">Select {field.label}</option>
   815→                        {field.options?.map((opt) => (
   816→                          <option key={opt} value={opt}>{opt}</option>
   817→                        ))}
   818→                      </select>
   819→                    </div>
   820→                  );
   821→                }
   822→
   823→                // text field
   824→                return (
   825→                  <div key={field.key} className="space-y-2">
   826→                    <Label className="text-xs font-medium text-gray-500">
   827→                      {field.label} {field.required && <span className="text-red-500">*</span>}
   828→                    </Label>
   829→                    <Input
   830→                      type="text"
   831→                      placeholder={field.placeholder}
   832→                      value={formFields[field.key] || ''}
   833→                      onChange={(e) => setFormFields((prev) => ({ ...prev, [field.key]: e.target.value }))}
   834→                      disabled={isSubmitted || isReviewing}
   835→                      className="text-sm"
   836→                    />
   837→                  </div>
   838→                );
   839→              })}
   840→            </CardContent>
   841→          </Card>
   842→        )}
   843→
   844→        {/* Document Upload — only when not locked and not in approved state */}
   845→        {!isLocked && !isApproved && (
   846→          <Card>
   847→            <CardHeader className="pb-3">
   848→              <CardTitle className="text-base font-semibold text-gray-900">
   849→                Upload Documents
   850→              </CardTitle>
   851→            </CardHeader>
   852→            <CardContent className="space-y-4">
   853→              <div
   854→                className={`border-2 border-dashed rounded-xl p-6 text-center transition-colors cursor-pointer ${
   855→                  isSubmitted || isReviewing
   856→                    ? 'border-gray-200 bg-gray-50/50 cursor-not-allowed'
   857→                    : 'border-gray-200 hover:border-blue-300 hover:bg-blue-50/30'

   859→                onClick={() => {

   864→                <p className="text-sm font-medium text-gray-700">
   865→                  {isSubmitted || isReviewing ? 'Upload locked during review' : 'Click to upload document'}
   866→                </p>
   867→                <p className="text-xs text-gray-400 mt-1">
   868→                  JPG, PNG, HEIC or PDF. Max {config.maxFileSize / 1024 / 1024}MB

   871→
   872→              {uploadedFiles.length > 0 && (

   874→                  {uploadedFiles.map((file) => (

   876→                      <FileText className="size-4 text-gray-400" />

   880→                      </div>
   881→                      {!(isSubmitted || isReviewing) && (

   884→                          className="text-gray-400 hover:text-red-500 transition-colors"
   885→                        >
   886→                          <XCircle className="size-4" />
   887→                        </button>
   888→                      )}
   889→                    </div>
   890→                  ))}
   891→                </div>
   892→              )}
   893→            </CardContent>
   894→          </Card>
   895→        )}
   896→
   897→        {/* Submit Button */}
   898→        {!isLocked && !isApproved && (
   899→          <Button
   900→            className={`w-full h-12 text-sm font-medium rounded-xl shadow-md ${
   901→              isSubmitDisabled
   902→                ? 'bg-gray-200 text-gray-400 cursor-not-allowed'
   903→                : 'bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-700 hover:to-indigo-700 text-white'
   904→            }`}
   905→            disabled={isSubmitDisabled}
   906→            onClick={handleSubmit}
   907→          >
   908→            {submitting ? (
   909→              <>
   910→                <Loader2 className="size-4 mr-2 animate-spin" />
   911→                Submitting...
   912→              </>
   913→            ) : isRejected ? (
   914→              <>
   915→                <RefreshCw className="size-4 mr-2" />
   916→                Re-submit for Verification (Attempt {nextAttempt} of {MAX_ATTEMPTS})
   917→              </>
   918→            ) : (
   919→              <>
   920→                <Send className="size-4 mr-2" />
   921→                Submit for Verification
   922→              </>
   923→            )}
   924→          </Button>
   925→        )}
   926→      </div>
   927→    </div>
   928→  );
