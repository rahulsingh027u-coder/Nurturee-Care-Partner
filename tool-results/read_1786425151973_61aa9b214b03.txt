   184→function loadDocsFromStorage(): VerificationDoc[] {
   185→  try {
   186→    // Version check — if storage format is outdated, clear and start fresh
   187→    const storedVersion = localStorage.getItem(STORAGE_VERSION_KEY);
   188→    if (storedVersion !== String(CURRENT_STORAGE_VERSION)) {
   189→      localStorage.removeItem(VERIFICATION_STORAGE_KEY);
   190→      localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   191→      return getInitialDocs();
   192→    }
   193→    const stored = localStorage.getItem(VERIFICATION_STORAGE_KEY);
   194→    if (stored) {
   195→      const parsed = JSON.parse(stored);
   196→      if (Array.isArray(parsed) && parsed.length === 4) {
   197→        // Validate structure — reject data from old format
   198→        const isValid = parsed.every((d: VerificationDoc) => d.id && d.type && d.status && Array.isArray(d.attempts));
   199→        if (isValid) return parsed;
   200→      }
   201→    }
   202→  } catch {
   203→    // fall through
   204→  }
   205→  return getInitialDocs();
   206→}
   207→
   208→function saveDocsToStorage(docs: VerificationDoc[]) {
   209→  localStorage.setItem(STORAGE_VERSION_KEY, String(CURRENT_STORAGE_VERSION));
   210→  localStorage.setItem(VERIFICATION_STORAGE_KEY, JSON.stringify(docs));
   211→}

   230→// ─── Optional Deadline Helpers ─────────────────────────────────────────────────
   231→
   232→const OPTIONAL_DEADLINE_DAYS = 15;
   233→const REGISTERED_AT_KEY = 'nurturee_care_partner_registered_at';
   234→
   235→function getDaysRemaining(): number | null {
   236→  try {
   237→    const registeredAt = localStorage.getItem(REGISTERED_AT_KEY);
   238→    if (!registeredAt) return null;
   239→    const registered = new Date(registeredAt).getTime();
   240→    const deadline = registered + OPTIONAL_DEADLINE_DAYS * 24 * 60 * 60 * 1000;
   241→    const remaining = Math.ceil((deadline - Date.now()) / (24 * 60 * 60 * 1000));
   242→    return Math.max(0, remaining);
   243→  } catch {
   244→    return null;
   245→  }
   246→}

   250→export default function VerificationPage({ onComplete, optionalMode = false }: VerificationPageProps) {
   251→  const [docs, setDocs] = useState<VerificationDoc[]>(loadDocsFromStorage);
   252→  const [activeType, setActiveType] = useState<VerificationType | null>(null);
   253→  const [formFields, setFormFields] = useState<Record<string, string>>({});
   254→  const [uploadedFiles, setUploadedFiles] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   255→  const [uploadedFront, setUploadedFront] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   256→  const [uploadedBack, setUploadedBack] = useState<Array<{ id: string; name: string; size: number; type: string }>>([]);
   257→  const [submitting, setSubmitting] = useState(false);
   258→  const [submittedType, setSubmittedType] = useState<VerificationType | null>(null);
   259→  const [reviewingType, setReviewingType] = useState<VerificationType | null>(null);
   260→  const [calendarOpen, setCalendarOpen] = useState<Record<string, boolean>>({});
   261→  const [approvingType, setApprovingType] = useState<VerificationType | null>(null);
   262→  const reviewTimersRef = useRef<Record<string, NodeJS.Timeout[]>>({});
   263→
   264→  const currentYear = new Date().getFullYear();
   265→
   266→  // Clean up timers on unmount
   267→  useEffect(() => {
   268→    return () => {
   269→      Object.values(reviewTimersRef.current).forEach((timers) => {
   270→        timers.forEach(clearTimeout);
   271→      });
   272→    };
   273→  }, []);
   274→
   275→  // Guard against stale submitted/reviewing state if docs changed (e.g., page refresh, storage cleared)
   276→  useEffect(() => {
   277→    if (submittedType) {
   278→      const doc = docs.find((d) => d.type === submittedType);
   279→      if (!doc || (doc.status !== 'submitted' && doc.status !== 'under_review')) {
   280→        setSubmittedType(null);
   281→        setReviewingType(null);
   282→      }
   283→    }
   284→    if (reviewingType) {
   285→      const doc = docs.find((d) => d.type === reviewingType);
   286→      if (!doc || doc.status !== 'under_review') {
   287→        setReviewingType(null);
   288→      }
   289→    }
   290→  }, [docs, submittedType, reviewingType]);
   291→
   292→  // Persist docs to localStorage whenever they change
   293→  const updateDocs = useCallback((updater: (prev: VerificationDoc[]) => VerificationDoc[]) => {
   294→    setDocs((prev) => {
   295→      const updated = updater(prev);
   296→      saveDocsToStorage(updated);
   297→      return updated;
   298→    });
   299→  }, []);
   300→
   301→  // ─── Backend Review Simulation ───────────────────────────────────────────
   302→
   303→  const startReviewSimulation = useCallback((type: VerificationType, attemptNumber: number) => {
   304→    // Clear any existing timers for this type
   305→    if (reviewTimersRef.current[type]) {
   306→      reviewTimersRef.current[type].forEach(clearTimeout);
   307→    }
   308→    reviewTimersRef.current[type] = [];
   309→
   310→    const doc = docs.find((d) => d.type === type);
   311→    if (!doc) return;
   312→
   313→    // Timer 1: After 3s → under_review
   314→    const t1 = setTimeout(() => {
   315→      setReviewingType(type);
   316→      updateDocs((prev) =>
   317→        prev.map((d) =>
   318→          d.type === type ? { ...d, status: 'under_review' as VerificationStatus } : d
   319→        )
   320→      );
   321→    }, 3000);
   322→
   323→    // Timer 2: After 5-7s total → resolve
   324→    const resolveDelay = 5000 + Math.random() * 2000;
   325→    const t2 = setTimeout(() => {
   326→      const currentDoc = docs.find((d) => d.type === type);
   327→      if (!currentDoc) return;
   328→
   329→      let willApprove: boolean;
   330→
   331→      if (type === 'aadhaar') {
   332→        // Aadhaar always approve on first attempt (demo-friendly)
   333→        willApprove = true;
   334→      } else {
   335→        // Others: 60% reject on attempt 1, 30% on attempt 2, always approve on attempt 3
   336→        if (attemptNumber === 1) {
   337→          willApprove = Math.random() < 0.4;
   338→        } else if (attemptNumber === 2) {
   339→          willApprove = Math.random() < 0.7;
   340→        } else {
   341→          willApprove = true;
   342→        }
   343→      }
   344→
   345→      if (willApprove) {
   346→        // Approve
   347→        setApprovingType(type);
   348→        updateDocs((prev) =>
   349→          prev.map((d) => {
   350→            if (d.type !== type) return d;
   351→            return {
   352→              ...d,
   353→              status: 'approved' as VerificationStatus,
   354→              attempts: d.attempts.map((a) =>
   355→                a.attemptNumber === attemptNumber
   356→                  ? { ...a, status: 'approved' as const, reviewedAt: new Date().toISOString() }
   357→                  : a
   358→              ),
   359→            };
   360→          })
   361→        );
   362→        setReviewingType(null);
   363→
   364→        // If Aadhaar approved and onComplete exists, call it after delay
   365→        if (type === 'aadhaar' && onComplete) {
   366→          setTimeout(() => {
   367→            setApprovingType(null);
   368→            setSubmittedType(null);
   369→            onComplete();
   370→          }, 1000);
   371→        } else {
   372→          // For optional: show success briefly, then allow going back
   373→          setTimeout(() => {
   374→            setApprovingType(null);
   375→            setSubmittedType(null);
   376→            setActiveType(null);
   377→          }, 2000);
   378→        }
   379→      } else {
   380→        // Reject
   381→        const reason = REJECTION_REASONS[Math.floor(Math.random() * REJECTION_REASONS.length)];
   382→        updateDocs((prev) =>
   383→          prev.map((d) => {
   384→            if (d.type !== type) return d;
   385→            const updatedAttempts = d.attempts.map((a) =>
   386→              a.attemptNumber === attemptNumber
   387→                ? { ...a, status: 'rejected' as const, reviewedAt: new Date().toISOString(), rejectionReason: reason }
   388→                : a
   389→            );
   390→            // Check if max attempts reached
   391→            const isMaxAttempts = updatedAttempts.length >= MAX_ATTEMPTS && updatedAttempts[updatedAttempts.length - 1].status === 'rejected';
   392→            return {
   393→              ...d,
   394→              status: isMaxAttempts ? ('max_attempts_reached' as VerificationStatus) : ('rejected' as VerificationStatus),
   395→              attempts: updatedAttempts,
   396→              currentDraft: { documents: [], formFields: {} },
   397→            };
   398→          })
   399→        );
   400→        setReviewingType(null);
   401→        // Transition from submitted view to allow re-upload
   402→        setTimeout(() => {
   403→          setSubmittedType(null);
   404→        }, 300);
   405→      }
   406→    }, resolveDelay);
   407→
   408→    reviewTimersRef.current[type] = [t1, t2];
   409→  }, [docs, onComplete, updateDocs]);
   410→
   411→  // ─── Handlers ────────────────────────────────────────────────────────────
   412→
   413→  const handleSelectType = (type: VerificationType) => {
   414→    const doc = docs.find((d) => d.type === type);
   415→    if (!doc) return;
   416→    setActiveType(type);
   417→    // Restore draft for editable docs; approved/locked docs show read-only
   418→    const isFrozenDoc = doc.status === 'max_attempts_reached' || doc.status === 'approved';
   419→    if (!isFrozenDoc) {
   420→      setFormFields(doc.currentDraft?.formFields || {});
   421→      if (type === 'aadhaar') {
   422→        const draftDocs = doc.currentDraft?.documents || [];
   423→        setUploadedFront(draftDocs.filter(f => f.id.startsWith('front-')));
   424→        setUploadedBack(draftDocs.filter(f => f.id.startsWith('back-')));
   425→        setUploadedFiles([]);
   426→      } else {
   427→        setUploadedFiles(doc.currentDraft?.documents || []);
   428→        setUploadedFront([]);
   429→        setUploadedBack([]);
   430→      }
   431→    } else {
   432→      setFormFields({});
   433→      setUploadedFiles([]);
   434→      setUploadedFront([]);
   435→      setUploadedBack([]);
   436→    }
   437→    setSubmittedType(null);
   438→  };
   439→
   440→  const handleBack = () => {
   441→    // Save draft
   442→    if (activeType) {
   443→      // For Aadhaar, combine front + back into documents array
   444→      const isAadhaar = activeType === 'aadhaar';
   445→      const combinedDocs = isAadhaar
   446→        ? [...uploadedFront.map(f => ({ ...f, id: `front-${f.id}` })), ...uploadedBack.map(f => ({ ...f, id: `back-${f.id}` }))]
   447→        : uploadedFiles;
   448→      updateDocs((prev) =>
   449→        prev.map((d) =>
   450→          d.type === activeType
   451→            ? { ...d, currentDraft: { documents: combinedDocs, formFields } }
   452→            : d
   453→        )
   454→      );
   455→    }
   456→    // Clean up timers for this type if we navigate back
   457→    if (activeType && reviewTimersRef.current[activeType]) {
   458→      // Don't clear if review is in progress (submittedType or reviewingType is set)
   459→      if (!submittedType && !reviewingType) {
   460→        reviewTimersRef.current[activeType].forEach(clearTimeout);
   461→        delete reviewTimersRef.current[activeType];
   462→      }
   463→    }
   464→    setActiveType(null);
   465→    setFormFields({});
   466→    setUploadedFiles([]);
   467→    setUploadedFront([]);
   468→    setUploadedBack([]);
   469→    setSubmittedType(null);
   470→    setReviewingType(null);
   471→  };
   472→
   473→  const validateAndCreateFileEntries = (files: FileList, config: typeof verificationConfig[VerificationType] | null) => {
   474→    const result: Array<{ id: string; name: string; size: number; type: string }> = [];
   475→    for (let i = 0; i < files.length; i++) {
   476→      const file = files[i];
   477→      if (config && !config.allowedTypes.includes(file.type)) {
   478→        alert(`Invalid file type: ${file.name}. Allowed: JPG, PNG, HEIC, PDF`);
   479→        continue;
   480→      }
   481→      if (config && file.size > config.maxFileSize) {
   482→        alert(`File too large: ${file.name}. Max ${config.maxFileSize / 1024 / 1024}MB`);
   483→        continue;
   484→      }
   485→      result.push({ id: `file-${Date.now()}-${i}`, name: file.name, size: file.size, type: file.type });
   486→    }
   487→    return result;
   488→  };
   489→
   490→  // Generic file change (for non-Aadhaar docs)
   491→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   492→    const files = e.target.files;
   493→    if (!files || files.length === 0) return;
   494→    const config = activeType ? verificationConfig[activeType] : null;
   495→    const newFiles = validateAndCreateFileEntries(files, config);
   496→    if (newFiles.length > 0) {
   497→      setUploadedFiles((prev) => [...prev, ...newFiles]);
   498→    }
   499→    e.target.value = '';
   500→  };
   501→
   502→  // Aadhaar-specific: handle front/back file uploads
   503→  const handleAadhaarSideChange = (side: 'front' | 'back') => (e: React.ChangeEvent<HTMLInputElement>) => {
   504→    const files = e.target.files;
   505→    if (!files || files.length === 0) return;
   506→    const config = verificationConfig['aadhaar'];
   507→    const newFiles = validateAndCreateFileEntries(files, config);
   508→    const setter = side === 'front' ? setUploadedFront : setUploadedBack;
   509→    if (newFiles.length > 0) {
   510→      setter(newFiles); // Replace — only one file per side
   511→    }
   512→    e.target.value = '';
   513→  };
   514→
   515→  const handleRemoveFile = (fileId: string) => {
   516→    setUploadedFiles((prev) => prev.filter((f) => f.id !== fileId));
   517→  };
   518→  const handleRemoveAadhaarSide = (side: 'front' | 'back') => () => {
   519→    if (side === 'front') setUploadedFront([]);
   520→    else setUploadedBack([]);
   521→  };
   522→
   523→  const handleSubmit = async () => {
   524→    if (!activeType) return;
   525→    setSubmitting(true);
   526→
   527→    // Simulate submission delay
