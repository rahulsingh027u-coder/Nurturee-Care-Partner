     1→'use client';

     3→import { useState, useMemo } from 'react';
     4→import { Badge } from '@/components/ui/badge';
     5→import { Button } from '@/components/ui/button';
     6→import {

    33→interface ShiftsViewProps {

    38→  confirmed: {

    41→  },
    42→  in_progress: {

    45→  },
    46→  completed: {
    47→    label: 'Completed',

    62→
    63→function getDateGroupLabel(dateStr: string): string {
    64→  const date = parseISO(dateStr);
    65→  if (isToday(date)) return 'Today';
    66→  if (isTomorrow(date)) return 'Tomorrow';
    67→  if (isYesterday(date)) return 'Yesterday';
    68→  return format(date, 'EEE, MMM d, yyyy');
    69→}
    70→
    71→function getDateGroupSortKey(dateStr: string): number {
    72→  const date = parseISO(dateStr);
    73→  if (isToday(date)) return 0;
    74→  if (isTomorrow(date)) return 1;
    75→  if (isYesterday(date)) return -1;
    76→  return date.getTime();
    77→}

    82→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('Upcoming');
    83→  const [startShift, setStartShift] = useState<Shift | null>(null);
    84→  const [shiftStarted, setShiftStarted] = useState(false);
    85→  const [completeShift, setCompleteShift] = useState<Shift | null>(null);
    86→  const [shiftCompleted, setShiftCompleted] = useState(false);

    88→
    89→  // Shifts in the active date range (used for stats counts)
    90→  const dateRangeShifts = useMemo(() => {
    91→    return mockShifts.filter((shift) => {
    92→      const shiftDate = parseISO(shift.date);
    93→      if (startDate || endDate) {
    94→        // User picked a custom range
    95→        if (startDate && shiftDate < startDate && !isSameDay(shiftDate, startDate)) return false;
    96→        if (endDate && shiftDate > endDate && !isSameDay(shiftDate, endDate)) return false;

    98→        // Default: current month
    99→        const now = new Date();
   100→        const monthStart = startOfMonth(now);
   101→        const monthEnd = endOfMonth(now);
   102→        if (!isWithinInterval(shiftDate, { start: monthStart, end: monthEnd })) return false;
   103→      }
   104→      return true;
   105→    });
   106→  }, [startDate, endDate]);
   107→

   109→    const total = dateRangeShifts.length;

   111→    const upcoming = dateRangeShifts.filter(
   112→      (s) => s.status === 'confirmed' || s.status === 'in_progress'
   113→    ).length;
   114→    const cancelled = dateRangeShifts.filter((s) => s.status === 'cancelled').length;
   115→    return { total, completed, upcoming, cancelled };
   116→  }, [dateRangeShifts]);
   117→

   119→    return mockShifts.filter((shift) => {
   120→      const shiftDate = parseISO(shift.date);
   121→

   123→      if (startDate && shiftDate < startDate && !isSameDay(shiftDate, startDate)) return false;
   124→      if (endDate && shiftDate > endDate && !isSameDay(shiftDate, endDate)) return false;
   125→
   126→      // Status filter
   127→      if (statusFilter === 'Confirmed' && shift.status !== 'confirmed') return false;
   128→      if (statusFilter === 'In Progress' && shift.status !== 'in_progress') return false;
   129→      if (statusFilter === 'Completed' && shift.status !== 'completed') return false;
   130→      if (statusFilter === 'Cancelled' && shift.status !== 'cancelled') return false;
   131→      if (statusFilter === 'Upcoming' && shift.status !== 'confirmed' && shift.status !== 'in_progress') return false;
   132→
   133→      return true;
   134→    });
   135→  }, [startDate, endDate, statusFilter, dateRangeShifts]);
   136→

   138→  const groupedShifts = useMemo(() => {
   139→    const groups: Record<string, typeof mockShifts> = {};

   141→    for (const shift of filteredShifts) {

   147→    // Sort groups: Today first, then Tomorrow, then other future dates, then past

   150→      const keyB = getDateGroupSortKey(b[0].date);

   152→    });

   154→    return sortedEntries;
   155→  }, [filteredShifts]);
   156→

   159→    { key: 'Upcoming', label: 'Upcoming', icon: Clock, count: stats.upcoming, color: 'text-blue-500', activeColor: 'text-blue-600 border-blue-300 bg-blue-50 ring-1 ring-blue-100' },
   160→    { key: 'Completed', label: 'Completed', icon: CheckCircle, count: stats.completed, color: 'text-emerald-500', activeColor: 'text-emerald-600 border-emerald-300 bg-emerald-50 ring-1 ring-emerald-100' },
   161→    { key: 'Cancelled', label: 'Cancelled', icon: XCircle, count: stats.cancelled, color: 'text-red-400', activeColor: 'text-red-600 border-red-300 bg-red-50 ring-1 ring-red-100' },
   162→  ];
   163→
   164→  return (
   165→    <div className="space-y-5">

   167→      <div>
   168→        <div className="flex items-center gap-2.5 mb-4">
   169→          <div className="flex items-center justify-center w-9 h-9 rounded-lg bg-blue-50">
   170→            <Briefcase className="w-5 h-5 text-blue-600" />
   171→          </div>

   175→        {/* Unified filter bar: Status pills + Date filters in one row */}

   181→              <button

   183→                key={pill.key}
   184→                onClick={() => setStatusFilter(pill.key)}
   185→                className={`flex items-center gap-2 rounded-lg border px-3 py-2 transition-all hover:shadow-md hover:-translate-y-0.5 active:scale-[0.98] cursor-pointer ${
   186→                  isActive ? pill.activeColor : 'border-gray-100 bg-white hover:border-gray-300'
   187→                }`}
   188→              >
   189→                <Icon className={`w-3.5 h-3.5 ${isActive ? '' : pill.color} transition-colors`} />
   190→                <span className={`text-xs font-medium ${isActive ? '' : 'text-gray-500'} transition-colors`}>{pill.label}</span>

   196→          {/* Divider */}
   197→          <div className="w-px h-6 bg-gray-200 mx-1 hidden sm:block" />

   199→          {/* Start Date Picker */}

   203→                className={`flex items-center gap-1.5 rounded-lg border px-3 py-1.5 text-xs font-medium transition-all hover:shadow-sm ${
   204→                  startDate
   205→                    ? 'bg-blue-50 text-blue-700 border-blue-200'
   206→                    : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'
   207→                }`}
   208→              >
   209→                <CalendarIcon className="w-3.5 h-3.5" />
   210→                {startDate ? format(startDate, 'MMM d, yyyy') : 'Start Date'}
   211→              </button>
   212→            </PopoverTrigger>
   213→            <PopoverContent className="w-auto p-0" align="start">
   214→              <Calendar
   215→                mode="single"
   216→                selected={startDate}

   218→              />
   219→            </PopoverContent>
   220→          </Popover>
   221→
   222→          {/* End Date Picker */}
   223→          <Popover>
   224→            <PopoverTrigger asChild>
   225→              <button
   226→                className={`flex items-center gap-1.5 rounded-lg border px-3 py-1.5 text-xs font-medium transition-all hover:shadow-sm ${
   227→                  endDate
   228→                    ? 'bg-blue-50 text-blue-700 border-blue-200'
   229→                    : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'
   230→                }`}
   231→              >
   232→                <CalendarIcon className="w-3.5 h-3.5" />
   233→                {endDate ? format(endDate, 'MMM d, yyyy') : 'End Date'}
   234→              </button>
   235→            </PopoverTrigger>

   237→              <Calendar
   238→                mode="single"
   239→                selected={endDate}
   240→                onSelect={(date) => setEndDate(date ?? undefined)}
   241→              />
   242→            </PopoverContent>
   243→          </Popover>
   244→
   245→          {/* Clear date filter */}
   246→          {(startDate || endDate) && (

   248→              onClick={() => { setStartDate(undefined); setEndDate(undefined); }}
   249→              className="text-[11px] text-gray-400 hover:text-gray-600 transition-colors"
   250→            >
   251→              ✕ Clear
   252→            </button>
   253→          )}
   254→        </div>
   255→      </div>
   256→
   257→      {/* Section 3: Shifts List */}
   258→      {groupedShifts.length === 0 ? (
   259→        <div className="flex flex-col items-center justify-center py-16 text-center">
   260→          <div className="w-14 h-14 rounded-full bg-gray-100 flex items-center justify-center mb-4">
   261→            <Briefcase className="w-7 h-7 text-gray-400" />
   262→          </div>
   263→          <p className="text-sm font-medium text-gray-500">No shifts found</p>
   264→          <p className="text-xs text-gray-400 mt-1">Try adjusting your filters</p>
   265→        </div>

   267→        <div className="space-y-5">
   268→          {groupedShifts.map(([groupLabel, shifts]) => (
   269→            <div key={groupLabel}>
   270→              <div className="flex items-center gap-2 mb-2">
   271→                <CalendarIcon className="w-3.5 h-3.5 text-gray-500" />
   272→                <span className="text-sm font-semibold text-gray-700">{groupLabel}</span>
   273→                <span className="text-xs text-gray-400">({shifts.length})</span>
   274→              </div>
   275→
   276→              <div className="space-y-3">
   277→                {shifts.map((shift) => {
   278→                  const badgeInfo = statusBadgeMap[shift.status];
   279→                  const borderClass = leftBorderMap[shift.status];
   280→                  const isInProgress = shift.status === 'in_progress';
   281→                  const isCancelled = shift.status === 'cancelled';
   282→
   283→                  return (
   284→                    <div
   285→                      key={shift.id}
   286→                      className={`rounded-xl border border-gray-100 bg-white p-4 border-l-[3px] ${borderClass} ${isCancelled ? 'opacity-60' : ''}`}
   287→                    >
   288→                      {/* Row 1: Time + Status + Earnings */}
   289→                      <div className="flex items-center justify-between flex-wrap gap-2 mb-2">
   290→                        <div className="flex items-center gap-2 flex-wrap">
   291→                          <div className="flex items-center gap-1.5 text-gray-600">
   292→                            <Clock className="w-3.5 h-3.5 text-gray-400" />
   293→                            <span className="text-sm font-medium">
   294→                              {shift.startTime} - {shift.endTime}
   295→                            </span>
   296→                          </div>
   297→                          <Badge
   298→                            variant="outline"
   299→                            className={`text-[11px] px-2 py-0 ${badgeInfo.className}`}
   300→                          >
   301→                            {isInProgress && (
   302→                              <span className="relative flex h-2 w-2 mr-1.5">
   303→                                <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-blue-400 opacity-75"></span>
   304→                                <span className="relative inline-flex rounded-full h-2 w-2 bg-blue-500"></span>
   305→                              </span>
   306→                            )}
   307→                            {badgeInfo.label}
   308→                          </Badge>
   309→                        </div>
   310→                        <div className="flex items-center gap-1 text-emerald-600">
   311→                          <IndianRupee className="w-3.5 h-3.5" />
   312→                          <span className="text-sm font-semibold">
   313→                            {shift.earnings > 0
   314→                              ? `₹${shift.earnings.toLocaleString('en-IN')}`
   315→                              : '—'}
   316→                          </span>
   317→                        </div>
   318→                      </div>
   319→
   320→                      {/* Row 2: Patient name + Service type */}
   321→                      <div className="flex items-center gap-2 mb-2">
   322→                        <span className="text-sm font-semibold text-gray-900">
   323→                          {shift.patientName}
   324→                        </span>
   325→                        <span className="text-xs text-gray-400">•</span>

   327→                      </div>
   328→
   329→                      {/* Row 3: Mode badge + Booking ID */}
   330→                      <div className="flex items-center gap-2 mb-2">

   332→                          variant="outline"
   333→                          className={
   334→                            shift.consultationMode === 'in_home'
   335→                              ? 'bg-emerald-50 text-emerald-700 border-emerald-200 hover:bg-emerald-50 text-[11px] px-2 py-0'
   336→                              : 'bg-blue-50 text-blue-700 border-blue-200 hover:bg-blue-50 text-[11px] px-2 py-0'
   337→                          }
   338→                        >
   339→                          {shift.consultationMode === 'in_home' ? '🏠 In-Home' : '💻 Online'}
   340→                        </Badge>
   341→                        <span className="text-xs text-gray-400 font-mono">{shift.bookingId}</span>
   342→                      </div>
   343→
   344→                      {/* Row 4: Address (if in_home) */}
   345→                      {shift.consultationMode === 'in_home' && shift.address && (
   346→                        <div className="flex items-start gap-1.5 mb-2">
   347→                          <MapPin className="w-3.5 h-3.5 text-gray-400 mt-0.5 shrink-0" />
   348→                          <span className="text-xs text-gray-500">{shift.address}</span>
   349→                        </div>
   350→                      )}
   351→
   352→                      {/* Row 5: Notes */}
   353→                      {shift.notes && (
   354→                        <div className="bg-gray-50 rounded-lg p-2 mb-2">
   355→                          <p className="text-xs text-gray-600">{shift.notes}</p>

   357→                      )}
   358→
   359→                      {/* Row 6: Special Instructions */}
   360→                      {shift.specialInstructions && (
   361→                        <div className="bg-amber-50 border border-amber-200 rounded-lg p-2 mb-3">
   362→                          <div className="flex items-start gap-1.5">
   363→                            <AlertCircle className="w-3.5 h-3.5 text-amber-600 mt-0.5 shrink-0" />
   364→                            <p className="text-xs text-amber-700">{shift.specialInstructions}</p>

   366→                        </div>

   369→                      {/* Action Buttons */}
   370→                      <div className="flex items-center gap-2 pt-2 border-t border-gray-50">

   373→                            <Button

   375→                              className="bg-gradient-to-r from-blue-600 to-blue-500 hover:from-blue-700 hover:to-blue-600 text-white rounded-lg text-xs h-8 px-3"

   377→                            >
   378→                              Start Shift

   381→                              variant="outline"

   383→                              className="rounded-lg text-xs h-8 px-3 text-gray-600"
   384→                              onClick={() => onNavigate('patients')}
   385→                            >
   386→                              View Care Member
   387→                            </Button>
   388→                          </>
   389→                        )}
   390→                        {shift.status === 'in_progress' && (
   391→                          <>
   392→                            <Button
   393→                              size="sm"
   394→                              className="bg-gradient-to-r from-emerald-600 to-emerald-500 hover:from-emerald-700 hover:to-emerald-600 text-white rounded-lg text-xs h-8 px-3"
   395→                              onClick={() => setCompleteShift(shift)}
   396→                            >
   397→                              <CheckCircle2 className="w-3 h-3 mr-1" />
   398→                              Complete Shift

   400→                            <Button
   401→                              variant="outline"
   402→                              size="sm"
   403→                              className="rounded-lg text-xs h-8 px-3 text-gray-600"
   404→                            >
   405→                              Report Issue
   406→                            </Button>
   407→                          </>
   408→                        )}
   409→                        {shift.status === 'completed' && (
   410→                          <>
   411→                            <Button
   412→                              variant="outline"
   413→                              size="sm"
   414→                              className="rounded-lg text-xs h-8 px-3 text-blue-600 border-blue-200 hover:bg-blue-50"
   415→                              onClick={() => setDetailShift(shift)}
   416→                            >
   417→                              View Details
   418→                            </Button>
   419→                            <div className="flex items-center gap-1 text-emerald-600 ml-auto">
   420→                              <IndianRupee className="w-3.5 h-3.5" />
   421→                              <span className="text-sm font-semibold">

   433→                              <XCircle className="w-3 h-3 mr-1" />

   437→                              <span className="text-xs text-gray-400 italic ml-2">
   438→                                {shift.notes}
   439→                              </span>
   440→                            )}

   443→                      </div>
   444→                    </div>

   447→              </div>
