   179→// ── Status Display Config ───────────────────────────────────────────
   180→function getStatusDisplay(status: VerificationStatus) {
   181→  switch (status) {
   182→    case 'approved':
   183→      return { label: 'Verified', color: 'text-emerald-600', bg: 'bg-emerald-50', border: 'border-emerald-200', icon: CheckCircle };
   184→    case 'submitted':
   185→      return { label: 'Submitted', color: 'text-indigo-600', bg: 'bg-indigo-50', border: 'border-indigo-200', icon: Send };
   186→    case 'under_review':
   187→      return { label: 'Under Review', color: 'text-amber-600', bg: 'bg-amber-50', border: 'border-amber-200', icon: Clock };
   188→    case 'rejected':
   189→      return { label: 'Rejected', color: 'text-red-600', bg: 'bg-red-50', border: 'border-red-200', icon: AlertCircle };
   190→    case 'in_progress':
   191→      return { label: 'Draft', color: 'text-blue-600', bg: 'bg-blue-50', border: 'border-blue-200', icon: CircleDot };
   192→    case 'max_attempts_reached':
   193→      return { label: 'Locked', color: 'text-gray-500', bg: 'bg-gray-100', border: 'border-gray-300', icon: Ban };
   194→    default:
   195→      return { label: 'Not Started', color: 'text-gray-400', bg: 'bg-gray-50', border: 'border-gray-200', icon: XCircle };
   196→  }
   197→}

   199→// ── Component ────────────────────────────────────────────────────────
   200→interface VerificationFlowProps {
   201→  open: boolean;
   202→  onOpenChange: (open: boolean) => void;
   203→  verifications: VerificationDoc[];
   204→  onVerificationUpdate: (updated: VerificationDoc[]) => void;
   205→}
   206→
   207→export default function VerificationFlow({
   208→  open,
   209→  onOpenChange,
   210→  verifications,
   211→  onVerificationUpdate,
   212→}: VerificationFlowProps) {
   213→  const [view, setView] = useState<'overview' | 'upload'>('overview');
   214→  const [selectedType, setSelectedType] = useState<VerificationType | null>(null);
   215→  const [localDocs, setLocalDocs] = useState<VerificationDoc[]>(verifications);
   216→  const [isSubmitting, setIsSubmitting] = useState(false);
   217→  const [submitSuccess, setSubmitSuccess] = useState(false);
   218→  const [dragOver, setDragOver] = useState(false);
   219→  const fileInputRef = useRef<HTMLInputElement>(null);
   220→
   221→  // Sync with parent
   222→  const docs = open ? localDocs : verifications;
   223→  const setDocs = setLocalDocs;
   224→
   225→  const approvedCount = docs.filter((d) => d.status === 'approved').length;
   226→  const lockedCount = docs.filter((d) => d.status === 'max_attempts_reached').length;
   227→  const activeCount = docs.filter(
   228→    (d) =>
   229→      d.status === 'submitted' ||
   230→      d.status === 'under_review'
   231→  ).length;
   232→  const progressPercent = Math.round((approvedCount / docs.length) * 100);
   233→
   234→  // ── Helpers ──
   235→  const getCurrentDraft = () => {
   236→    if (!selectedType) return { documents: [] as UploadedFile[], formFields: {} as Record<string, string> };
   237→    return docs.find((d) => d.type === selectedType)?.currentDraft || { documents: [], formFields: {} };
   238→  };
   239→
   240→  const updateDraft = (updates: { documents?: UploadedFile[]; formFields?: Record<string, string> }) => {
   241→    setDocs((prev) =>
   242→      prev.map((d) => {
   243→        if (d.type !== selectedType) return d;
   244→        return {
   245→          ...d,
   246→          status: 'in_progress',
   247→          currentDraft: {
   248→            documents: updates.documents ?? d.currentDraft.documents,
   249→            formFields: updates.formFields ?? d.currentDraft.formFields,
   250→          },
   251→        };
   252→      })
   253→    );
   254→  };
   255→
   256→  const handleFileUpload = useCallback(
   257→    (files: FileList | null) => {
   258→      if (!files || !selectedType) return;
   259→      const newFiles: UploadedFile[] = Array.from(files).map((f) => ({
   260→        id: `${Date.now()}-${Math.random().toString(36).substring(2, 9)}`,
   261→        name: f.name,
   262→        size: f.size,
   263→        type: f.type,
   264→      }));
   265→      const draft = getCurrentDraft();
   266→      updateDraft({ documents: [...draft.documents, ...newFiles] });
   267→    },
   268→    [selectedType, docs]
   269→  );
   270→
   271→  const handleRemoveFile = (fileId: string) => {
   272→    const draft = getCurrentDraft();
   273→    updateDraft({ documents: draft.documents.filter((f) => f.id !== fileId) });
   274→  };
   275→
   276→  const handleFieldChange = (key: string, value: string) => {
   277→    const draft = getCurrentDraft();
   278→    updateDraft({ formFields: { ...draft.formFields, [key]: value } });
   279→  };
   280→
   281→  const handleSelectVerification = (type: VerificationType) => {
   282→    setSelectedType(type);
   283→    setView('upload');
   284→    setSubmitSuccess(false);
   285→  };
   286→
   287→  const handleBackToOverview = () => {
   288→    setView('overview');
   289→    setSelectedType(null);
   290→    setSubmitSuccess(false);
   291→  };
   292→
   293→  const handleClose = () => {
   294→    onVerificationUpdate(docs);
   295→    setView('overview');
   296→    setSelectedType(null);
   297→    setSubmitSuccess(false);
   298→    onOpenChange(false);
   299→  };
   300→
   301→  const handleSubmit = async () => {
   302→    if (!selectedType) return;
   303→    const config = verificationConfig[selectedType];
   304→    const current = docs.find((d) => d.type === selectedType);
   305→    const draft = current?.currentDraft || { documents: [], formFields: {} };
   306→
   307→    // Validate required fields
   308→    const allRequiredFilled = config.formFields
   309→      .filter((f) => f.required)
   310→      .every((f) => draft.formFields[f.key]?.trim());
   311→    if (!allRequiredFilled || draft.documents.length === 0) return;
   312→
   313→    setIsSubmitting(true);
   314→    // Simulate API call to backend portal
   315→    await new Promise((r) => setTimeout(r, 2000));
   316→
   317→    const attemptNumber = (current?.attempts?.length || 0) + 1;
   318→    const newAttempt: VerificationAttempt = {
   319→      attemptNumber,
   320→      submittedAt: new Date().toISOString(),
   321→      documents: [...draft.documents],
   322→      formFields: { ...draft.formFields },
   323→      status: 'submitted',
   324→    };
   325→
   326→    setDocs((prev) =>
   327→      prev.map((d) => {
   328→        if (d.type !== selectedType) return d;
   329→        const newAttempts = [...(d.attempts || []), newAttempt];
   330→        const isLastAttempt = attemptNumber >= MAX_ATTEMPTS;
   331→        return {
   332→          ...d,
   333→          status: 'submitted',
   334→          attempts: newAttempts,
   335→          currentDraft: { documents: [], formFields: {} },
   336→          // If this was the 3rd attempt, we still set submitted —
   337→          // backend will decide approved or rejected
   338→        };
   339→      })
   340→    );
   341→
   342→    setIsSubmitting(false);
   343→    setSubmitSuccess(true);
   344→
   345→    // Simulate backend moving to under_review after a brief delay
   346→    setTimeout(() => {
   347→      setDocs((prev) =>
   348→        prev.map((d) => {
   349→          if (d.type !== selectedType) return d;
   350→          return { ...d, status: 'under_review' };
   351→        })
   352→      );
   353→    }, 2500);
   354→  };
   355→
   356→  const formatFileSize = (bytes: number) => {
   357→    if (bytes < 1024) return bytes + ' B';
   358→    if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB';
   359→    return (bytes / 1048576).toFixed(1) + ' MB';
   360→  };
   361→
   362→  // ── RENDER: Overview ───────────────────────────────────────────────
   363→  const renderOverview = () => (
   364→    <div className="space-y-5">
   365→      {/* Progress Banner */}
   366→      <div className="rounded-xl bg-gradient-to-r from-blue-50 via-indigo-50 to-violet-50 border border-blue-100 p-4">
   367→        <div className="flex items-center justify-between mb-2">
   368→          <div>
   369→            <p className="text-sm font-semibold text-gray-900">Verification Progress</p>
   370→            <p className="text-xs text-gray-500 mt-0.5">
   371→              {approvedCount} of {docs.length} verified
   372→              {activeCount > 0 && ` · ${activeCount} under review`}
   373→              {lockedCount > 0 && ` · ${lockedCount} locked`}
   374→            </p>
   375→          </div>
   376→          <span className="text-lg font-bold text-blue-700">{progressPercent}%</span>
   377→        </div>
   378→        <Progress value={progressPercent} className="h-2" />
   379→      </div>
   380→
   381→      {/* 3-Attempt Policy Banner */}
   382→      <div className="rounded-xl bg-amber-50 border border-amber-100 p-3.5 flex items-start gap-3">
   383→        <Info className="size-5 text-amber-500 shrink-0 mt-0.5" />
   384→        <div>
   385→          <p className="text-sm font-medium text-amber-800">3-Attempt Approval Policy</p>
   386→          <p className="text-xs text-amber-600 mt-1 leading-relaxed">
   387→            Each verification allows up to <span className="font-bold">3 upload attempts</span>. If documents are rejected, you can re-upload corrected documents. After 3 failed attempts, that verification will be <span className="font-bold text-red-600">permanently locked</span> and you will need to contact Nurturee support for assistance.
   388→          </p>
   389→        </div>
   390→      </div>
   391→
   392→      {/* Verification Cards */}
   393→      <div className="space-y-3">
   394→        {docs.map((verDoc) => {
   395→          const config = verificationConfig[verDoc.type];
   396→          const status = getStatusDisplay(verDoc.status);
   397→          const ConfigIcon = config.icon;
   398→          const StatusIcon = status.icon;
   399→          const attemptsUsed = verDoc.attempts.length;
   400→          const attemptsLeft = MAX_ATTEMPTS - attemptsUsed;
   401→          const isLocked = verDoc.status === 'max_attempts_reached';
   402→          const isApproved = verDoc.status === 'approved';
   403→          const isActive = verDoc.status === 'submitted' || verDoc.status === 'under_review';
   404→          const isRejected = verDoc.status === 'rejected';
   405→          const latestAttempt = verDoc.attempts[verDoc.attempts.length - 1];
   406→
   407→          return (
   408→            <div
   409→              key={verDoc.type}
   410→              className={`rounded-xl border ${status.border} bg-white p-4 transition-all hover:shadow-sm`}
   411→            >
   412→              <div className="flex items-start gap-3">
   413→                <div className={`size-11 rounded-lg ${config.bgColor} flex items-center justify-center shrink-0`}>
   414→                  <ConfigIcon className={`size-5 ${config.textColor}`} />
   415→                </div>
   416→                <div className="flex-1 min-w-0">
   417→                  <div className="flex items-center justify-between">
   418→                    <p className="text-sm font-semibold text-gray-900">{config.title}</p>
   419→                    <div
   420→                      className={`inline-flex items-center gap-1 rounded-full px-2.5 py-0.5 text-xs font-medium ${status.bg} ${status.color} border ${status.border}`}
   421→                    >
   422→                      <StatusIcon className="size-3" />
   423→                      {status.label}
   424→                    </div>
   425→                  </div>
   426→
   427→                  {/* Attempt Tracker (dots) */}
   428→                  {!isApproved && !isLocked && (
   429→                    <div className="flex items-center gap-1.5 mt-2">
   430→                      {Array.from({ length: MAX_ATTEMPTS }).map((_, i) => {
   431→                        const attempt = verDoc.attempts[i];
   432→                        let dotClass = 'bg-gray-200 border-gray-300';
   433→                        let dotIcon: React.ReactNode = null;
   434→                        if (attempt) {
   435→                          if (attempt.status === 'approved') {
   436→                            dotClass = 'bg-emerald-100 border-emerald-300';
   437→                            dotIcon = <CheckCircle className="size-3 text-emerald-500" />;
   438→                          } else if (attempt.status === 'rejected') {
   439→                            dotClass = 'bg-red-100 border-red-300';
   440→                            dotIcon = <XCircle className="size-3 text-red-500" />;
   441→                          } else {
   442→                            dotClass = 'bg-blue-100 border-blue-300';
   443→                            dotIcon = <Loader2 className="size-3 text-blue-500 animate-spin" />;
   444→                          }
   445→                        }
   446→                        return (
   447→                          <div
   448→                            key={i}
   449→                            className={`size-6 rounded-full border-2 flex items-center justify-center ${dotClass}`}
   450→                            title={`Attempt ${i + 1}${attempt ? ` — ${attempt.status}` : ' — unused'}`}
   451→                          >
   452→                            {dotIcon}
   453→                          </div>
   454→                        );
   455→                      })}
   456→                      <span className="text-xs text-gray-500 ml-1">
   457→                        {attemptsUsed === 0
   458→                          ? `${MAX_ATTEMPTS} attempts available`
   459→                          : `${attemptsUsed}/${MAX_ATTEMPTS} used` +
   460→                            (attemptsLeft > 0 ? ` · ${attemptsLeft} remaining` : ' · No attempts left')}
   461→                      </span>
   462→                    </div>
   463→                  )}
   464→
   465→                  {isLocked && (
   466→                    <div className="mt-2 rounded-lg bg-gray-100 border border-gray-200 p-2 flex items-start gap-2">
   467→                      <Ban className="size-4 text-gray-500 shrink-0 mt-0.5" />
   468→                      <div>
   469→                        <p className="text-xs font-medium text-gray-700">Maximum attempts reached</p>
   470→                        <p className="text-xs text-gray-500">
   471→                          All 3 attempts were rejected. Please contact{' '}
   472→                          <span className="text-blue-600 font-medium">Nurturee Support</span> for further assistance.
   473→                        </p>
   474→                      </div>
   475→                    </div>
   476→                  )}
   477→
   478→                  {/* Rejection Reason */}
   479→                  {isRejected && latestAttempt?.rejectionReason && (
   480→                    <div className="mt-2 rounded-lg bg-red-50 border border-red-100 p-2">
   481→                      <p className="text-xs text-red-600">
   482→                        <span className="font-medium">Reason: </span>
   483→                        {latestAttempt.rejectionReason}
   484→                      </p>
   485→                    </div>
   486→                  )}
   487→
   488→                  {/* Last submitted date */}
   489→                  {latestAttempt?.submittedAt && !isApproved && (
   490→                    <p className="text-xs text-gray-400 mt-1.5">
   491→                      Last submitted:{' '}
   492→                      {new Date(latestAttempt.submittedAt).toLocaleDateString('en-IN', {
   493→                        day: 'numeric',
   494→                        month: 'short',
   495→                        year: 'numeric',
   496→                      })}
   497→                    </p>
   498→                  )}
   499→
   500→                  {/* Action Buttons */}
   501→                  <div className="mt-3 flex items-center gap-2">
   502→                    {(verDoc.status === 'not_started' || verDoc.status === 'in_progress') && (
   503→                      <Button size="sm" onClick={() => handleSelectVerification(verDoc.type)} className="gap-1.5 text-xs h-8">
   504→                        <Upload className="size-3.5" />
   505→                        {verDoc.status === 'not_started' ? 'Start Verification' : 'Continue Upload'}
   506→                      </Button>
   507→                    )}
   508→                    {isRejected && attemptsLeft > 0 && (
   509→                      <Button
   510→                        size="sm"
   511→                        variant="outline"
   512→                        onClick={() => handleSelectVerification(verDoc.type)}
   513→                        className="gap-1.5 text-xs h-8 text-orange-600 border-orange-200 hover:bg-orange-50"
   514→                      >
   515→                        <RotateCcw className="size-3.5" />
   516→                        Re-upload (Attempt {attemptsUsed + 1} of {MAX_ATTEMPTS})
   517→                      </Button>
   518→                    )}
   519→                    {isRejected && attemptsLeft <= 0 && (
   520→                      <Button size="sm" variant="outline" disabled className="gap-1.5 text-xs h-8">
   521→                        <Ban className="size-3.5" />
   522→                        Max Attempts Reached
   523→                      </Button>
   524→                    )}
   525→                    {isActive && (
   526→                      <Badge className="rounded-full text-xs bg-indigo-50 text-indigo-600 border border-indigo-200">
   527→                        <Loader2 className="size-3 mr-1 animate-spin" />
   528→                        Sent to backend for review
   529→                      </Badge>

   531→                    {isApproved && (

   568→
